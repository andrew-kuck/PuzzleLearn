Below is a complete single-file activity. Save it as `index.html` and open it in a browser.

It uses:

- **Leaflet** for interactive maps
- **Natural Earth GeoJSON** for wordless continent/country shapes
- **geoBoundaries** for province/state boundaries
- **OpenStreetMap Overpass + Nominatim** for a random real-world street address, with a built-in fallback list if public APIs are slow/rate-limited
- **Turf.js** for distance and geometry calculations

> Note: Public map/geocoding APIs have rate limits. For a production classroom app, cache results or use your own API key/proxy.

Key updates:

1. **Province/state is omitted entirely** if it is not meaningfully distinct from the country or city.
   - Example: `Asia › Singapore › Singapore › 1 Bayfront Avenue`
   - Example: `Asia › Japan › Tokyo › 1-1 Chiyoda`
   - Not: `Asia › Japan › Tokyo › Tokyo › 1-1 Chiyoda`

2. **After the city-location guess**, the map:
   - draws the error line from the user’s guess to the actual city location,
   - shows the distance,
   - briefly pauses,
   - removes that temporary city-guess line/marker,
   - then zooms to show the whole city.

3. **Street-address guessing happens after the city is already fully visible.**
   - The map does not move after the street guess.
   - It simply draws the error line between the guessed location and the actual address.

    async function getRandomAddress() {
      for (let attempt = 0; attempt < 3; attempt++) {
        try {
          const place =
            GLOBAL_SAMPLE_PLACES[
              Math.floor(Math.random() * GLOBAL_SAMPLE_PLACES.length)

          const addr = reverse.address || {};

          const countryCode = (addr.country_code || "").toUpperCase();
          if (!countryCode || !addr.country) continue;

          const countryInfo = await getCountryInfo(countryCode);
          const continent = continentFromCountryInfo(countryInfo);

          const city =
            addr.city ||
            addr.town ||
            addr.village ||
            addr.municipality ||
            addr.suburb ||
            addr.county;

            addr.state ||
            addr.province ||
            addr.region ||
            addr.state_district ||
            null;

          const road = addr.road || addr.pedestrian || addr.footway;
          const house = addr.house_number;

          if (!continent || !city || !road) continue;

          if (!isMeaningfulRegion(state, city, addr.country)) {
            state = null;
          }

          const street = house ? `${house} ${road}` : road;

          const cityInfo = await geocodeCity(city, state, addr.country);

          return {
            continent,
            country: addr.country,
            iso3: countryInfo.cca3,
            state,
            city,
            street,
            lat: Number(osmAddress.lat),
            lng: Number(osmAddress.lng),
            cityLat: cityInfo?.lat ?? Number(osmAddress.lat),
            cityLng: cityInfo?.lng ?? Number(osmAddress.lng),
            cityBbox: cityInfo?.bbox ?? null,
            cityGeojson: cityInfo?.geojson ?? null
          };
        } catch (err) {
          console.warn("Random OSM address attempt failed:", err);
        }
      }

      return FALLBACK_ADDRESSES[
        Math.floor(Math.random() * FALLBACK_ADDRESSES.length)
      ];
    }

    async function randomAddressFromOverpass(place) {
      const query = `
        [out:json][timeout:20];
        (
          node["addr:housenumber"]["addr:street"](around:${place.radius},${place.lat},${place.lng});
          way["addr:housenumber"]["addr:street"](around:${place.radius},${place.lat},${place.lng});
        );
        out center 80;
      `;

      const url = `${OVERPASS_URL}?data=${encodeURIComponent(query)}`;
      const data = await fetchJson(url);

      if (!data.elements || data.elements.length === 0) {
        throw new Error("No OSM address elements found.");
      }

      const element =
        data.elements[Math.floor(Math.random() * data.elements.length)];

      const lat = element.lat ?? element.center?.lat;
      const lng = element.lon ?? element.center?.lon;

      if (lat == null || lng == null) {
        throw new Error("OSM element has no usable coordinate.");
      }

      return { lat, lng };
    }

    async function reverseGeocode(lat, lng) {
      return fetchJson(`${NOMINATIM_REVERSE}&lat=${lat}&lon=${lng}`);
    }

    async function geocodeCity(city, state, country) {
      try {
        const parts = new URLSearchParams();
        parts.set("city", city);
        if (state) parts.set("state", state);
        parts.set("country", country);
        parts.set("polygon_geojson", "1");

        const data = await fetchJson(`${NOMINATIM_SEARCH}&${parts.toString()}`);

        if (!Array.isArray(data) || !data[0]) return null;

        const item = data[0];

        return {
          lat: Number(item.lat),
          lng: Number(item.lon),
          bbox: item.boundingbox
            ? item.boundingbox.map(Number)
            : null,
          geojson: item.geojson || null
        };
      } catch {
        return null;
      }
    }

    async function getCountryInfo(alpha2) {

        `${REST_COUNTRIES}${alpha2}?fields=cca3,region,subregion,name`
      );
      return Array.isArray(data) ? data[0] : data;
    }

    function continentFromCountryInfo(info) {
      if (!info) return null;

      if (info.region === "Europe") return "Europe";
      if (info.region === "Asia") return "Asia";
      if (info.region === "Africa") return "Africa";
      if (info.region === "Oceania") return "Oceania";
      if (info.region === "Antarctic") return "Antarctica";

      if (info.region === "Americas") {
        if ((info.subregion || "").includes("South")) return "South America";
        return "North America";
      }

      return info.region || null;
    }

    function normalizeTargetAddress(address) {
      const copy = { ...address };

      if (!isMeaningfulRegion(copy.state, copy.city, copy.country)) {
        copy.state = null;
      }

      return copy;
    }

    function isMeaningfulRegion(region, city, country) {
      if (!region) return false;

      const r = normalizeName(region);
      const c = normalizeName(city);
      const countryNorm = normalizeName(country);

      if (!r) return false;
      if (r === c) return false;
      if (r === countryNorm) return false;

      return true;
    }

      }
    }

    /************************************************************
     * GUESS HANDLERS
     ************************************************************/

    function handleContinentGuess(clickedCountryFeature) {
      const guessedContinent = clickedCountryFeature.properties.CONTINENT;

      if (guessedContinent !== target.continent) {
        const guessedCenter = centerOfFeatures(
          getFeaturesByContinent(guessedContinent)
        );

        const correctCenter = centerOfFeatures(
          getFeaturesByContinent(target.continent)
        );

        addErrorLine(
          guessedCenter,
          correctCenter,
          `${guessedContinent} → ${target.continent}`,
          drawingLayer
        );
      }

      renderContinent();
    }

    function handleCountryGuess(clickedCountryFeature) {
      if (!sameCountry(clickedCountryFeature, correctCountryFeature)) {
        addErrorLine(
          centerOfFeature(clickedCountryFeature),
          centerOfFeature(correctCountryFeature),
          `${countryName(clickedCountryFeature)} → ${target.country}`,
          drawingLayer
        );
      }

      renderCountryOrProvinceStep();
    }

    function handleStateGuess(clickedStateFeature) {
      if (!sameAdminFeature(clickedStateFeature, correctStateFeature)) {
        addErrorLine(
          centerOfFeature(clickedStateFeature),
          centerOfFeature(correctStateFeature),
          `${adminName(clickedStateFeature)} → ${adminName(correctStateFeature)}`,
          drawingLayer
        );
      }

      renderCityGuess();
    }

    async function handleCityPointGuess(e) {
      const guess = { lat: e.latlng.lat, lng: e.latlng.lng };
      const actual = { lat: target.cityLat, lng: target.cityLng };

      addGuessMarker(guess, temporaryLayer, "Your city guess");

      L.circleMarker([actual.lat, actual.lng], {
        radius: 7,
        color: "#00274c",
        fillColor: "#ffcb05",
        fillOpacity: 1,
        weight: 3
      })
        .addTo(temporaryLayer)
        .bindTooltip(target.city, { permanent: true, direction: "top" });

      addErrorLine(
        guess,
        actual,
        `Your city guess → ${target.city}`,
        temporaryLayer
      );

      updateBanner(
        `City error added. Zooming to ${target.city} in a moment…`,
        false
      );

      await sleep(CITY_GUESS_PAUSE_MS);

      temporaryLayer.clearLayers();
      renderStreetGuess();
    }

    function handleStreetPointGuess(e) {
      const guess = { lat: e.latlng.lat, lng: e.latlng.lng };
      const actual = { lat: target.lat, lng: target.lng };

      addGuessMarker(guess, drawingLayer, "Your address guess");

      L.circleMarker([actual.lat, actual.lng], {
        radius: 7,
        color: "#00274c",
        fillColor: "#ffcb05",
        fillOpacity: 1,
        weight: 3
      })
        .addTo(drawingLayer)
        .bindTooltip("Actual address", { permanent: true, direction: "top" });

      addErrorLine(
        guess,
        actual,
        "Your address guess → actual address",
        drawingLayer
      );

      activePart = null;
      finalMode = true;

      updateBanner(
        `Final score: ${formatKm(totalErrorKm)} total error. Refresh the page to play again.`,
        true
      );

      /*
        Important: no map movement here.
        The city was already fitted to the screen before this guess.
      */
    }

    /************************************************************
     * GEOMETRY, DISTANCE, AND DATA HELPERS
     ************************************************************/

    function addErrorLine(from, to, label, layerGroup) {
      const a = normalizeLatLng(from);
      const b = normalizeLatLng(to);

      const km = turf.distance(
        turf.point([a.lng, a.lat]),
        turf.point([b.lng, b.lat]),
        { units: "kilometers" }
      );

      totalErrorKm += km;

      L.polyline(
        [
          [a.lat, a.lng],
          [b.lat, b.lng]
        ],
        {
          color: "#e11d48",
          weight: 4,
          opacity: 0.9,
          dashArray: "8 8"
        }
      ).addTo(layerGroup);

      const mid = {
        lat: (a.lat + b.lat) / 2,
        lng: midpointLng(a.lng, b.lng)
      };

      L.marker([mid.lat, mid.lng], {
        icon: L.divIcon({
          className: "",
          html: `<div class="distance-label">${formatKm(km)}</div>`,
          iconSize: null
        })
      })
        .addTo(layerGroup)
        .bindTooltip(label);

      updateBanner(null, finalMode);

      return km;
    }

    function addGuessMarker(point, layerGroup, label) {
      L.circleMarker([point.lat, point.lng], {
        radius: 6,
        color: "#e11d48",
        fillColor: "#e11d48",
        fillOpacity: 0.9,
        weight: 2
      })
        .addTo(layerGroup)
        .bindTooltip(label, { permanent: false });
    }

    function midpointLng(lng1, lng2) {
      let diff = lng2 - lng1;

      if (Math.abs(diff) > 180) {
        if (diff > 0) lng1 += 360;
        else lng2 += 360;
      }

      let mid = (lng1 + lng2) / 2;

      if (mid > 180) mid -= 360;
      if (mid < -180) mid += 360;

      return mid;
    }

    function normalizeLatLng(value) {
      if (Array.isArray(value)) return { lat: value[0], lng: value[1] };
      return value;
    }

    function centerOfFeature(feature) {
      const c = turf.centroid(feature).geometry.coordinates;
      return { lat: c[1], lng: c[0] };
    }

    function centerOfFeatures(features) {
      const collection = {
        type: "FeatureCollection",
        features
      };

      const c = turf.centroid(collection).geometry.coordinates;
      return { lat: c[1], lng: c[0] };
    }

    function findCountryFeature(iso3, country) {
      return countries.features.find((f) => {
        const p = f.properties || {};

        return (
          p.ISO_A3 === iso3 ||
          p.ADM0_A3 === iso3 ||
          p.SOV_A3 === iso3 ||
          normalizeName(p.ADMIN) === normalizeName(country) ||
          normalizeName(p.NAME) === normalizeName(country)
        );
      });
    }

    function getFeaturesByContinent(continent) {
      return countries.features.filter(
        (f) => f.properties && f.properties.CONTINENT === continent
      );
    }

    async function loadAdm1(iso3) {
      if (adm1Cache.has(iso3)) return adm1Cache.get(iso3);

      try {
        const meta = await fetchJson(`${GEOBOUNDARIES_API}/${iso3}/ADM1/`);
        const geojsonUrl =
          meta.simplifiedGeometryGeoJSON || meta.geometryGeoJSON;

        if (!geojsonUrl) throw new Error("No ADM1 GeoJSON URL.");

        const fc = await fetchJson(geojsonUrl);
        adm1Cache.set(iso3, fc);
        return fc;
      } catch (err) {
        console.warn("ADM1 boundary load failed:", err);
        adm1Cache.set(iso3, null);
        return null;
      }
    }

    function findContainingAdminFeature(fc, lat, lng) {
      if (!fc || !fc.features) return null;

      const pt = turf.point([lng, lat]);

      return (
        fc.features.find((feature) => {
          try {
            return turf.booleanPointInPolygon(pt, feature);
          } catch {
            return false;
          }
        }) || null
      );
    }

    function findAdminFeatureByName(fc, name) {
      if (!fc || !fc.features || !name) return null;

      const needle = normalizeName(name);

      return (
        fc.features.find((feature) => {
          const n = normalizeName(adminName(feature));
          return n === needle || n.includes(needle) || needle.includes(n);
        }) || null
      );
    }

    function sameCountry(a, b) {
      const ap = a.properties || {};
      const bp = b.properties || {};

      const aIso = ap.ISO_A3 || ap.ADM0_A3 || ap.SOV_A3;
      const bIso = bp.ISO_A3 || bp.ADM0_A3 || bp.SOV_A3;

      if (aIso && bIso && aIso === bIso) return true;

      return normalizeName(countryName(a)) === normalizeName(countryName(b));
    }

    function sameAdminFeature(a, b) {
      if (!a || !b) return false;

      const ap = a.properties || {};
      const bp = b.properties || {};

      if (ap.shapeID && bp.shapeID) return ap.shapeID === bp.shapeID;
      if (ap.shapeISO && bp.shapeISO) return ap.shapeISO === bp.shapeISO;

      return normalizeName(adminName(a)) === normalizeName(adminName(b));
    }

    function countryName(feature) {
      const p = feature.properties || {};
      return p.ADMIN || p.NAME || p.NAME_EN || "Unknown country";
    }

    function adminName(feature) {
      const p = feature.properties || {};

      return (
        p.shapeName ||
        p.name ||
        p.NAME_1 ||
        p.NAME ||
        p.admin1Name ||
        "Unknown region"
      );
    }

    function normalizeName(s) {
      return String(s || "")
        .toLowerCase()
        .normalize("NFD")
        .replace(/\p{Diacritic}/gu, "")
        .replace(/[^a-z0-9]+/g, " ")
        .trim();
    }

    function fitFeature(feature) {
      const temp = L.geoJSON(feature);
      const bounds = temp.getBounds();
      if (bounds.isValid()) map.fitBounds(bounds.pad(0.08));
    }

    function fitFeatures(features) {
      const temp = L.geoJSON({
        type: "FeatureCollection",
        features
      });

      const bounds = temp.getBounds();
      if (bounds.isValid()) map.fitBounds(bounds.pad(0.08));
    }

    /************************************************************
     * UI HELPERS
     ************************************************************/

    function updateBanner(instructionText, isFinal) {
      const parts = [
        { key: "continent", value: target?.continent },
        { key: "country", value: target?.country }
      ];

      if (stateAvailable && target?.state) {
        parts.push({ key: "state", value: target.state });
      }

      parts.push(
        { key: "city", value: target?.city },
        { key: "street", value: target?.street }
      );

      document.getElementById("addressLine").innerHTML = parts
        .filter((p) => p.value)
        .map((p) => {
          const cls = p.key === activePart ? "part active" : "part";
          return `<span class="${cls}">${escapeHtml(p.value)}</span>`;
        })
        .join(" &nbsp;›&nbsp; ");

      if (instructionText) {
        document.getElementById("instruction").textContent = instructionText;
      }

      document.getElementById("score").textContent = isFinal
        ? `Final score: ${formatKm(totalErrorKm)}`
        : `Running error: ${formatKm(totalErrorKm)}`;
    }

    function formatKm(km) {
      if (km >= 100) return `${Math.round(km).toLocaleString()} km`;
      if (km >= 10) return `${km.toFixed(1)} km`;
      return `${km.toFixed(2)} km`;
    }

    function clearInteractiveMap() {
      map.off("click");

      if (regionLayer) {
        map.removeLayer(regionLayer);
        regionLayer = null;
      }

      if (tileLayer) {
        map.removeLayer(tileLayer);
        tileLayer = null;
      }
    }

    function escapeHtml(str) {
      return String(str)
        .replaceAll("&", "&amp;")
        .replaceAll("<", "&lt;")
        .replaceAll(">", "&gt;")
        .replaceAll('"', "&quot;")
        .replaceAll("'", "&#039;");
    }

    function sleep(ms) {
      return new Promise((resolve) => setTimeout(resolve, ms));
    }

    async function fetchJson(url) {
      const response = await fetch(url);

      if (!response.ok) {
        throw new Error(`${response.status} ${response.statusText}: ${url}`);
      }

      return response.json();
    }
