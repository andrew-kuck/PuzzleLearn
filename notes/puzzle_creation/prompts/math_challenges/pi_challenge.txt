create an activity using html, css, and javascript. there is javascript constant for a string that is a long sequence of digits. initially, the constant can be set to the number 8 1000 times. initially the page is filled with the string. the string should be in size 32 font and centered on the page vertically and horizontally. it should extend with 10 digits on each line, and then break midword to the next line, so that the only scroll would be a vertical scroll. at the very top of the page is a small header that informs the user to click enter when they are ready to begin. when the user clicks enter, the header and the number disappear. instead there is a sequence of boxes moving right to left along a line. each box corresponds to a number in the sequence. the boxes are moving towards a dropoff that is near the left side of the screen. if the user enters the number of the next box, the box turns into that number and then fades away. if the user enters a number that is different from the next box, then the box turns red without revealing the number, and the user is prevented from further input for time it takes a box to move two spaces along the line. there is a constant to set the speed of the boxes moving along the line. there is a counter that increments for every correct entry (for every box that disappears). if a box reaches the dropoff, it falls over the cliff. at that point a modal pops up to tell the user Game Over and to show them the number of boxes they got correct. there is a button on the modal to restart, at which point the screen returns to its original state showing the whole number, and the process repeats.

the font for the numbers on the first page should be 52, and there should always be 10 digits in a row before breaking to the next line. the dropoff does not need to be labeled. no horizontal line should continue to the left of the dropoff. also, the vertical line marking the dropoff extends slightly above the horizontal line; it should touch the horizontal line but not extend above it.

actually, have the row labels appear just to the right of the row, with only 15 pixels of separation. and make sure that the actual row of digits is centered on the screen, so that the space between the 5th and 6th digits of a given row are exactly at the center of the screen

if the last digit of the digit_sequence is correctly guessed, call the function activityComplete()

Below is a complete single-file HTML activity. Save it as `index.html` and open it in a browser.

You can adjust the main constants near the top of the JavaScript:

- `DIGIT_SEQUENCE`
- `BOX_SPEED`
- `BOX_SPACING`
- `DROPOFF_X`

  <script>
    const DIGIT_SEQUENCE = "8".repeat(1000);
    const BOX_SPEED = 140;
    const BOX_SPACING = 76;
    const BOX_SIZE = 52;
    const DROPOFF_X = 120;
    const SPAWN_BUFFER = 900;

    const PENALTY_MS = Math.round((BOX_SPACING * 2 / BOX_SPEED) * 1000);

    document.documentElement.style.setProperty("--box-size", `${BOX_SIZE}px`);
    document.documentElement.style.setProperty("--dropoff-x", `${DROPOFF_X}px`);

    const startHeader = document.getElementById("startHeader");
    const initialScreen = document.getElementById("initialScreen");
    const numberDisplay = document.getElementById("numberDisplay");
    const gameScreen = document.getElementById("gameScreen");
    const track = document.getElementById("track");
    const scoreEl = document.getElementById("score");
    const inputStatus = document.getElementById("inputStatus");
    const modalOverlay = document.getElementById("modalOverlay");
    const modalTitle = document.getElementById("modalTitle");
    const modalMessage = document.getElementById("modalMessage");
    const restartButton = document.getElementById("restartButton");

    let activeBoxes = [];
    let nextSpawnIndex = 0;
    let nextExpectedIndex = 0;
    let score = 0;
    let gameRunning = false;
    let inputLocked = false;
    let lastFrameTime = 0;
    let animationId = null;
    let lockTimer = null;

    function formatSequenceInRows(sequence, rowLength) {
      const rows = [];

      for (let i = 0; i < sequence.length; i += rowLength) {
        rows.push(sequence.slice(i, i + rowLength));
      }

      return rows.join("\n");
    }

    function showInitialScreen() {
      document.body.classList.remove("game-active");
      startHeader.style.display = "block";
      initialScreen.style.display = "grid";
      gameScreen.style.display = "none";
      modalOverlay.style.display = "none";

      numberDisplay.textContent = formatSequenceInRows(DIGIT_SEQUENCE, 10);

      clearGameObjects();
      window.scrollTo({ top: 0, behavior: "instant" });
    }

    function clearGameObjects() {
      gameRunning = false;
      inputLocked = false;

      if (animationId) {
        cancelAnimationFrame(animationId);
        animationId = null;
      }

      if (lockTimer) {
        clearTimeout(lockTimer);
        lockTimer = null;
      }

      activeBoxes.forEach(box => box.el.remove());
      activeBoxes = [];

      nextSpawnIndex = 0;
      nextExpectedIndex = 0;
      score = 0;
      scoreEl.textContent = "0";
      inputStatus.textContent = "Type the next digit.";
    }

    function startGame() {
      clearGameObjects();

      document.body.classList.add("game-active");
      startHeader.style.display = "none";
      initialScreen.style.display = "none";
      gameScreen.style.display = "block";
      modalOverlay.style.display = "none";

      gameRunning = true;
      spawnBoxesIfNeeded();

      lastFrameTime = performance.now();
      animationId = requestAnimationFrame(gameLoop);
    }

    function spawnBox(x) {
      if (nextSpawnIndex >= DIGIT_SEQUENCE.length) return;

      const el = document.createElement("div");
      el.className = "box";
      el.textContent = "";
      el.setAttribute("aria-label", `Box ${nextSpawnIndex + 1}`);
      track.appendChild(el);

      const box = {
        index: nextSpawnIndex,
        digit: DIGIT_SEQUENCE[nextSpawnIndex],
        x,
        el,
        state: "normal"
      };

      activeBoxes.push(box);
      updateBoxPosition(box);

      nextSpawnIndex++;
    }

    function getRightmostBoxX() {
      if (activeBoxes.length === 0) {
        return window.innerWidth - 160;
      }

      return Math.max(...activeBoxes.map(box => box.x));
    }

    function spawnBoxesIfNeeded() {
      let rightmostX = getRightmostBoxX();

      while (
        nextSpawnIndex < DIGIT_SEQUENCE.length &&
        rightmostX < window.innerWidth + SPAWN_BUFFER
      ) {
        const newX = activeBoxes.length === 0 && nextSpawnIndex === 0
          ? window.innerWidth - 160
          : rightmostX + BOX_SPACING;

        spawnBox(newX);
        rightmostX = newX;
      }
    }

    function updateBoxPosition(box) {
      box.el.style.setProperty("--x", `${box.x}px`);
      box.el.style.transform = `translate(${box.x}px, 0)`;
    }

    function gameLoop(timestamp) {
      if (!gameRunning) return;

      const dt = (timestamp - lastFrameTime) / 1000;
      lastFrameTime = timestamp;

      for (const box of activeBoxes) {
        if (box.state !== "done") {
          box.x -= BOX_SPEED * dt;
          updateBoxPosition(box);
        }
      }

      const nextBox = activeBoxes.find(
        box => box.index === nextExpectedIndex && box.state !== "done"
      );

      if (nextBox && nextBox.x <= DROPOFF_X) {
        makeBoxFall(nextBox);
        return;
      }

      spawnBoxesIfNeeded();

      animationId = requestAnimationFrame(gameLoop);
    }

    function handleDigitInput(key) {
      if (!gameRunning || inputLocked) return;
      if (!/^[0-9]$/.test(key)) return;

      const expectedDigit = DIGIT_SEQUENCE[nextExpectedIndex];

      const targetBox = activeBoxes.find(
        box => box.index === nextExpectedIndex && box.state !== "done"
      );

      if (!targetBox) return;

      if (key === expectedDigit) {
        handleCorrectInput(targetBox, key);
      } else {
        handleWrongInput(targetBox);
      }
    }

    function handleCorrectInput(box, digit) {
      box.state = "done";
      box.el.textContent = digit;
      box.el.classList.remove("wrong");
      box.el.classList.add("correct");

      score++;
      nextExpectedIndex++;
      scoreEl.textContent = score;

      setTimeout(() => {
        box.el.remove();
        activeBoxes = activeBoxes.filter(active => active !== box);
      }, 300);

      if (nextExpectedIndex >= DIGIT_SEQUENCE.length) {
        endGame(true);
      }
    }

    function handleWrongInput(box) {
      inputLocked = true;
      inputStatus.textContent = `Wrong digit. Input locked for ${PENALTY_MS} ms.`;

      box.state = "wrong";
      box.el.textContent = "";
      box.el.classList.add("wrong");

      lockTimer = setTimeout(() => {
        if (!gameRunning) return;

        inputLocked = false;
        inputStatus.textContent = "Type the next digit.";

        if (box.state === "wrong") {
          box.state = "normal";
          box.el.classList.remove("wrong");
        }
      }, PENALTY_MS);
    }

    function makeBoxFall(box) {
      gameRunning = false;
      inputLocked = true;

      if (animationId) {
        cancelAnimationFrame(animationId);
        animationId = null;
      }

      if (lockTimer) {
        clearTimeout(lockTimer);
        lockTimer = null;
      }

      box.el.classList.remove("wrong");
      box.el.classList.add("falling");

      setTimeout(() => {
        endGame(false);
      }, 650);
    }

    function endGame(won) {
      gameRunning = false;
      inputLocked = true;

      if (animationId) {
        cancelAnimationFrame(animationId);
        animationId = null;
      }

      if (lockTimer) {
        clearTimeout(lockTimer);
        lockTimer = null;
      }

      modalTitle.textContent = won ? "You Finished!" : "Game Over";
      modalMessage.textContent = `You got ${score} box${score === 1 ? "" : "es"} correct.`;
      modalOverlay.style.display = "flex";
    }

    document.addEventListener("keydown", event => {
      if (!gameRunning && modalOverlay.style.display !== "flex") {
        if (event.key === "Enter") {
          event.preventDefault();
          startGame();
        }

        return;
      }

      if (gameRunning) {
        handleDigitInput(event.key);
      }
    });

    restartButton.addEventListener("click", () => {
      showInitialScreen();
    });

    showInitialScreen();
  </script>
</body>
</html>
