create an activity using html, css, and javascript. create an n-gon with between 8 and 14 sides, and determine the area of the shape. The sides must all be horizontal or vertical. each of the sides will have an integer length. Label the length of each side, with the following exceptions: first, of all the horizontal sides choose one at random not to label; of all the vertical sides choose one at random not to label; if group of sides are all horizontal or vertical translations of each other, only label one of those sides. consider only labeled sides when determining which sides are horizontal or vertical translations of each other, when removing additional labels based on horizontal or vertical translation. the user must supply the area of the shape. if correct, a progress bar will advance such that after 10 correct answers the progress bar is full and a function activityComplete() is called. the shape should appear large on the screen, but the labels should be small enough to be clearly associated with a particular side. the border of the shape should be thin with no rounded edges.

the labels should remain where they are, but always oriented with the bottom of the number towards the bottom of the screen (not rotated for vertical lines)

remove some titles and text, remove references to scoreText

Below is a complete single-file HTML activity. Save it as `index.html` and open it in a browser.

It generates a random orthogonal 8–14-gon, labels side lengths according to your rules, checks the user’s area answer, advances a progress bar, and calls `activityComplete()` after 10 correct answers.

comment out the side.orientation bit to not have rotated sides

  <script>
    /*
      If the host page already defines activityComplete(), this will not overwrite it.
      After 10 correct answers, activityComplete() is called.
    */
    window.activityComplete = window.activityComplete || function () {
      console.log("activityComplete() called");
      alert("Activity complete!");
    };

    const TARGET_CORRECT = 10;

    const svg = document.getElementById("shapeSvg");
    const form = document.getElementById("answerForm");
    const input = document.getElementById("areaInput");
    const feedback = document.getElementById("feedback");
    const progressFill = document.getElementById("progressFill");
    const scoreText = document.getElementById("scoreText");
    const problemTitle = document.getElementById("problemTitle");
    const submitBtn = document.getElementById("submitBtn");
    const progressShell = document.querySelector(".progress-shell");

    let correctCount = 0;
    let currentArea = 0;
    let currentProblem = null;

    function randInt(min, max) {
      return Math.floor(Math.random() * (max - min + 1)) + min;
    }

    function shuffle(array) {
      const copy = array.slice();
      for (let i = copy.length - 1; i > 0; i--) {
        const j = randInt(0, i);
        [copy[i], copy[j]] = [copy[j], copy[i]];
      }
      return copy;
    }

    function choice(array) {
      return array[randInt(0, array.length - 1)];
    }

    function randomPartition(total, parts) {
      if (parts === 1) return [total];

      const cuts = new Set();
      while (cuts.size < parts - 1) {
        cuts.add(randInt(1, total - 1));
      }

      const sorted = [0, ...Array.from(cuts).sort((a, b) => a - b), total];
      const widths = [];

      for (let i = 0; i < sorted.length - 1; i++) {
        widths.push(sorted[i + 1] - sorted[i]);
      }

      return widths;
    }

    function randomStepHeights(count, min, max) {
      const values = [];

      for (let i = 0; i < count; i++) {
        let value = randInt(min, max);

        // Avoid consecutive equal heights so that every generated side is real.
        while (i > 0 && value === values[i - 1]) {
          value = randInt(min, max);
        }

        values.push(value);
      }

      return values;
    }

    function signedArea(points) {
      let sum = 0;

      for (let i = 0; i < points.length; i++) {
        const a = points[i];
        const b = points[(i + 1) % points.length];
        sum += a.x * b.y - b.x * a.y;
      }

      return sum / 2;
    }

    function makeOrthogonalPolygon() {
      // Total side count is 2 * (top horizontal segments + bottom horizontal segments).
      // This gives 8, 10, 12, or 14 sides.
      const halfSideCount = randInt(4, 7);
      const n = halfSideCount * 2;

      const topSegments = randInt(1, halfSideCount - 1);
      const bottomSegments = halfSideCount - topSegments;

      const width = randInt(Math.max(12, halfSideCount + 7), 20);

      const topWidths = randomPartition(width, topSegments);
      const bottomWidths = randomPartition(width, bottomSegments);

      // Top is always above bottom, guaranteeing a simple orthogonal polygon.
      const topYs = randomStepHeights(topSegments, 8, 15);
      const bottomYs = randomStepHeights(bottomSegments, 0, 5);

      const topXs = [0];
      for (const w of topWidths) {
        topXs.push(topXs[topXs.length - 1] + w);
      }

      const bottomXs = [0];
      for (const w of bottomWidths) {
        bottomXs.push(bottomXs[bottomXs.length - 1] + w);
      }

      const points = [];

      // Start at upper-left and trace clockwise along the top chain.
      points.push({ x: 0, y: topYs[0] });

      let x = 0;
      for (let i = 0; i < topSegments; i++) {
        x += topWidths[i];
        points.push({ x, y: topYs[i] });

        if (i < topSegments - 1) {
          points.push({ x, y: topYs[i + 1] });
        }
      }

      // Right vertical side.
      points.push({ x: width, y: bottomYs[bottomSegments - 1] });

      // Trace the bottom chain from right to left.
      for (let i = bottomSegments - 1; i >= 0; i--) {
        points.push({ x: bottomXs[i], y: bottomYs[i] });

        if (i > 0) {
          points.push({ x: bottomXs[i], y: bottomYs[i - 1] });
        }
      }

      return {
        points,
        n,
        area: Math.abs(signedArea(points))
      };
    }

    function getSides(points) {
      return points.map((p1, i) => {
        const p2 = points[(i + 1) % points.length];
        const dx = p2.x - p1.x;
        const dy = p2.y - p1.y;
        const orientation = dy === 0 ? "H" : "V";
        const length = Math.abs(dx || dy);

        return {
          index: i,
          p1,
          p2,
          dx,
          dy,
          orientation,
          length,
          minX: Math.min(p1.x, p2.x),
          maxX: Math.max(p1.x, p2.x),
          minY: Math.min(p1.y, p2.y),
          maxY: Math.max(p1.y, p2.y),
          label: true
        };
      });
    }

    function areHorizontalOrVerticalTranslations(a, b) {
      if (a.orientation !== b.orientation) return false;
      if (a.length !== b.length) return false;

      if (a.orientation === "H") {
        const sameHorizontalLine = a.p1.y === b.p1.y;
        const sameXSpan = a.minX === b.minX && a.maxX === b.maxX;

        // Horizontal sides can be horizontal translations or vertical translations.
        return sameHorizontalLine || sameXSpan;
      } else {
        const sameVerticalLine = a.p1.x === b.p1.x;
        const sameYSpan = a.minY === b.minY && a.maxY === b.maxY;

        // Vertical sides can be vertical translations or horizontal translations.
        return sameVerticalLine || sameYSpan;
      }
    }

    function applyLabelRules(sides) {
      sides.forEach(side => {
        side.label = true;
      });

      const horizontalSides = sides.filter(side => side.orientation === "H");
      const verticalSides = sides.filter(side => side.orientation === "V");

      // Exception 1: Hide one random horizontal side and one random vertical side.
      if (horizontalSides.length > 0) {
        choice(horizontalSides).label = false;
      }

      if (verticalSides.length > 0) {
        choice(verticalSides).label = false;
      }

      /*
        Exception 2:
        Among the sides that are still labeled, if sides are horizontal or vertical
        translations of each other, keep only one label.

        This intentionally considers only currently labeled sides, so the two randomly
        removed sides above do not affect this duplicate-removal pass.
      */
      const stillLabeled = shuffle(sides.filter(side => side.label));
      const kept = [];

      for (const side of stillLabeled) {
        const duplicateOfKeptSide = kept.some(keptSide =>
          areHorizontalOrVerticalTranslations(side, keptSide)
        );

        if (duplicateOfKeptSide) {
          side.label = false;
        } else {
          kept.push(side);
        }
      }
    }

    function clearSvg() {
      while (svg.firstChild) {
        svg.removeChild(svg.firstChild);
      }
    }

    function createSvgElement(tag) {
      return document.createElementNS("http://www.w3.org/2000/svg", tag);
    }

    function renderProblem(problem) {
      clearSvg();

      const points = problem.points;
      const sides = getSides(points);
      applyLabelRules(sides);

      const minX = Math.min(...points.map(p => p.x));
      const maxX = Math.max(...points.map(p => p.x));
      const minY = Math.min(...points.map(p => p.y));
      const maxY = Math.max(...points.map(p => p.y));

      const width = maxX - minX;
      const height = maxY - minY;
      const margin = Math.max(1.4, Math.min(width, height) * 0.12);

      svg.setAttribute(
        "viewBox",
        `${minX - margin} ${minY - margin} ${width + margin * 2} ${height + margin * 2}`
      );

      const polygon = createSvgElement("polygon");
      polygon.setAttribute("points", points.map(p => `${p.x},${p.y}`).join(" "));
      polygon.setAttribute("class", "shape-outline");
      svg.appendChild(polygon);

      const orientationSign = Math.sign(signedArea(points));
      const clockwise = orientationSign < 0;

      const fontSize = Math.max(0.34, Math.min(0.56, Math.min(width, height) / 22));
      const labelOffset = Math.max(0.42, fontSize * 1.25);

      for (const side of sides) {
        if (!side.label) continue;

        const mx = (side.p1.x + side.p2.x) / 2;
        const my = (side.p1.y + side.p2.y) / 2;

        const len = side.length;
        const ux = side.dx / len;
        const uy = side.dy / len;

        // Interior normal: right normal for clockwise polygons, left normal otherwise.
        let nx;
        let ny;

        if (clockwise) {
          nx = uy;
          ny = -ux;
        } else {
          nx = -uy;
          ny = ux;
        }

        const tx = mx + nx * labelOffset;
        const ty = my + ny * labelOffset;

        const text = createSvgElement("text");
        text.setAttribute("x", tx);
        text.setAttribute("y", ty);
        text.setAttribute("class", "side-label");
        text.setAttribute("font-size", fontSize);
        text.textContent = side.length;

        if (side.orientation === "V") {
          text.setAttribute("transform", `rotate(-90 ${tx} ${ty})`);
        }

        svg.appendChild(text);
      }
    }

    function updateProgress() {
      scoreText.textContent = correctCount;
      progressFill.style.width = `${(correctCount / TARGET_CORRECT) * 100}%`;
      progressShell.setAttribute("aria-valuenow", String(correctCount));
    }

    function newProblem() {
      currentProblem = makeOrthogonalPolygon();
      currentArea = currentProblem.area;

      problemTitle.textContent = `Find the area of this ${currentProblem.n}-gon`;
      input.value = "";
      feedback.textContent = "";
      feedback.className = "feedback";

      renderProblem(currentProblem);
      input.focus();
    }

    form.addEventListener("submit", event => {
      event.preventDefault();

      const userAnswer = Number(input.value);

      if (!Number.isFinite(userAnswer)) {
        feedback.textContent = "Please enter a number.";
        feedback.className = "feedback incorrect";
        return;
      }

      if (Math.abs(userAnswer - currentArea) < 1e-9) {
        correctCount++;
        updateProgress();

        if (correctCount >= TARGET_CORRECT) {
          feedback.textContent = "Correct! Progress complete.";
          feedback.className = "feedback correct";
          input.disabled = true;
          submitBtn.disabled = true;

          setTimeout(() => {
            window.activityComplete();
          }, 200);

          return;
        }

        feedback.textContent = "Correct! Loading a new shape...";
        feedback.className = "feedback correct";

        setTimeout(newProblem, 450);
      } else {
        feedback.textContent = "Not quite. Try again.";
        feedback.className = "feedback incorrect";
      }
    });

    updateProgress();
    newProblem();
  </script>
</body>
</html>
