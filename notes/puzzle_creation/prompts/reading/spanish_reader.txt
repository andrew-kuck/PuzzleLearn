create an activity using html, css, and javascript. the user is prompted to paste spanish language text into a large text area and click submit. the spanish text is then broken into sentences. each sentence is converted into english, so that there exists a spanish and an english version of each sentence. the pasted text should disappear. the new page should have a very clean interface that consists entirely of the first spanish sentence in the top half of the screen and the corresponding first english sentence in the bottom half of the screen. one word of the spanish sentence (at random) should be removed and replaced with an input box for user typed input. only words with 5 or more characters should be candidates for removal, unless there are not such words in the sentence. the user must supply the missing word in the input box and hit enter. if they are correct, the input box turns green, and after an a-second pause, the next sentence is presented (in spanish and english). a is a javascript constant initially set to 1 second. if the user is incorrect, the input box turns red and the user is prevented from further input for b seconds. b is a javascript constant initially set to 2 seconds. after answering incorrectly c times, the correct answer is inserted into the input for the user, and after the usual a-second pause, the screen advances to the next sentence. continue in this way through all the sentences. after advancing beyond the last sentence in the text, all the text on the screen should disappear and the user should be given the option to repeat the same text or use a new text. if they repeat the same text, repeat the process from the first sentence, but picking different random words to remove. if they pick to use a new text, then the page can simply refresh so that the entire process starts over.

the display for showing sentences is good, but make sure the sentence fits on its half of the screen, with some padding. the default text size should be slightly smaller, and have the text shrink if necessary, even if this means subsequent sentences are displayed at different font sizes

Below is a complete single-file activity using **HTML, CSS, and JavaScript**. Save it as something like `spanish-cloze.html` and open it in a browser.

Important note: the translation is done using the public MyMemory translation API. For private/sensitive text, replace the translation function with your own approved/private translation service.

  <script>
    /*
      Timing constants.

      A_SECONDS:
      Pause after a correct answer before advancing.

      B_SECONDS:
      Lockout time after an incorrect answer.

      C_MAX_INCORRECT:
      Number of incorrect attempts before showing the answer.
    */
    const A_SECONDS = 1;
    const B_SECONDS = 2;
    const C_MAX_INCORRECT = 3;

    const setupScreen = document.getElementById("setupScreen");
    const loadingScreen = document.getElementById("loadingScreen");
    const activityScreen = document.getElementById("activityScreen");
    const endScreen = document.getElementById("endScreen");

    const textInput = document.getElementById("textInput");
    const submitBtn = document.getElementById("submitBtn");

    const spanishSentenceEl = document.getElementById("spanishSentence");
    const englishSentenceEl = document.getElementById("englishSentence");

    const repeatBtn = document.getElementById("repeatBtn");
    const newTextBtn = document.getElementById("newTextBtn");

    let sentencePairs = [];
    let currentIndex = 0;
    let currentAnswer = "";
    let currentIncorrectCount = 0;
    let locked = false;

    submitBtn.addEventListener("click", startActivity);
    repeatBtn.addEventListener("click", repeatSameText);
    newTextBtn.addEventListener("click", () => location.reload());

    async function startActivity() {
      const rawText = textInput.value.trim();

      if (!rawText) {
        alert("Please paste some Spanish text first.");
        return;
      }

      const spanishSentences = splitIntoSentences(rawText);

      if (spanishSentences.length === 0) {
        alert("No sentences were found. Please try a longer text.");
        return;
      }

      setupScreen.classList.add("hidden");
      loadingScreen.classList.remove("hidden");

      try {
        const translatedSentences = [];

        for (const sentence of spanishSentences) {
          const translation = await translateSpanishToEnglish(sentence);
          translatedSentences.push(translation);
        }

        sentencePairs = spanishSentences.map((spanish, index) => ({
          spanish,
          english: translatedSentences[index],
          usedRemovalStarts: new Set()
        }));

        currentIndex = 0;

        loadingScreen.classList.add("hidden");
        activityScreen.classList.remove("hidden");

        presentSentence();
      } catch (error) {
        console.error(error);
        alert("There was a problem translating the text. Please try again.");
        location.reload();
      }
    }

    function splitIntoSentences(text) {
      const cleanText = text.replace(/\s+/g, " ").trim();

      if ("Segmenter" in Intl) {
        const segmenter = new Intl.Segmenter("es", { granularity: "sentence" });
        return Array.from(segmenter.segment(cleanText))
          .map(item => item.segment.trim())
          .filter(Boolean);
      }

      return cleanText
        .match(/[^.!?]+[.!?]+["')\]]*|[^.!?]+$/g)
        ?.map(sentence => sentence.trim())
        .filter(Boolean) || [];
    }

    async function translateSpanishToEnglish(sentence) {
      /*
        This uses MyMemory's public API for convenience.

        For a production app, especially if users may paste private text,
        replace this function with a call to your own backend or an approved
        translation provider.
      */

      const url =
        "https://api.mymemory.translated.net/get?q=" +
        encodeURIComponent(sentence) +
        "&langpair=es|en";

      const response = await fetch(url);

      if (!response.ok) {
        throw new Error("Translation request failed.");
      }

      const data = await response.json();

      const translated =
        data?.responseData?.translatedText ||
        "[Translation unavailable]";

      return decodeHtmlEntities(translated);
    }

    function decodeHtmlEntities(text) {
      const textarea = document.createElement("textarea");
      textarea.innerHTML = text;
      return textarea.value;
    }

    function presentSentence() {
      if (currentIndex >= sentencePairs.length) {
        finishActivity();
        return;
      }

      currentIncorrectCount = 0;
      locked = false;

      const pair = sentencePairs[currentIndex];
      const cloze = createClozeSentence(pair);

      currentAnswer = cloze.answer;

      spanishSentenceEl.innerHTML = cloze.html;
      englishSentenceEl.textContent = pair.english;

      const input = document.getElementById("answerInput");
      input.style.width = `${Math.max(currentAnswer.length + 1, 4)}ch`;
      input.focus();

      input.addEventListener("keydown", event => {
        if (event.key === "Enter") {
          event.preventDefault();
          checkAnswer(input);
        }
      });
    }

    function createClozeSentence(pair) {
      const sentence = pair.spanish;

      /*
        This regex captures words made of letters, including accented characters.
        It also allows internal apostrophes or hyphens.
      */
      const wordRegex = /\p{L}+(?:[’'-]\p{L}+)*/gu;
      const matches = Array.from(sentence.matchAll(wordRegex));

      if (matches.length === 0) {
        return {
          html: escapeHtml(sentence),
          answer: ""
        };
      }

      let candidates = matches.filter(match => match[0].length >= 5);

      if (candidates.length === 0) {
        candidates = matches;
      }

      /*
        Try to pick a different removed word on repeats.
        If all possible words have already been used, allow repeats.
      */
      let unusedCandidates = candidates.filter(match => {
        return !pair.usedRemovalStarts.has(match.index);
      });

      if (unusedCandidates.length === 0) {
        unusedCandidates = candidates;
      }

      const chosen =
        unusedCandidates[Math.floor(Math.random() * unusedCandidates.length)];

      const answer = chosen[0];
      const start = chosen.index;
      const end = start + answer.length;

      pair.usedRemovalStarts.add(start);

      const before = sentence.slice(0, start);
      const after = sentence.slice(end);

      const inputHtml =
        `<input 
          id="answerInput" 
          class="cloze-input" 
          type="text" 
          autocomplete="off" 
          autocapitalize="off" 
          spellcheck="false"
          aria-label="Missing Spanish word"
        />`;

      return {
        html: escapeHtml(before) + inputHtml + escapeHtml(after),
        answer
      };
    }

    function checkAnswer(input) {
      if (locked) return;

      const userAnswer = normalizeAnswer(input.value);
      const correctAnswer = normalizeAnswer(currentAnswer);

      if (userAnswer === correctAnswer) {
        handleCorrect(input);
      } else {
        handleIncorrect(input);
      }
    }

    function normalizeAnswer(value) {
      /*
        Case-insensitive but accent-sensitive.

        For example:
        "Canción" and "canción" match.
        "cancion" and "canción" do not.
      */
      return value
        .trim()
        .normalize("NFC")
        .toLocaleLowerCase("es");
    }

    function handleCorrect(input) {
      locked = true;
      input.value = currentAnswer;
      input.classList.remove("wrong");
      input.classList.add("correct");
      input.disabled = true;

      setTimeout(() => {
        currentIndex++;
        presentSentence();
      }, A_SECONDS * 1000);
    }

    function handleIncorrect(input) {
      currentIncorrectCount++;
      locked = true;

      input.classList.remove("correct");
      input.classList.add("wrong");
      input.disabled = true;

      if (currentIncorrectCount >= C_MAX_INCORRECT) {
        input.value = currentAnswer;
        input.classList.remove("wrong");
        input.classList.add("correct");

        setTimeout(() => {
          currentIndex++;
          presentSentence();
        }, A_SECONDS * 1000);

        return;
      }

      setTimeout(() => {
        input.disabled = false;
        input.classList.remove("wrong");
        input.focus();
        input.select();
        locked = false;
      }, B_SECONDS * 1000);
    }

    function finishActivity() {
      activityScreen.classList.add("hidden");
      endScreen.classList.remove("hidden");

      spanishSentenceEl.textContent = "";
      englishSentenceEl.textContent = "";
    }

    function repeatSameText() {
      endScreen.classList.add("hidden");
      activityScreen.classList.remove("hidden");

      currentIndex = 0;
      currentAnswer = "";
      currentIncorrectCount = 0;
      locked = false;

      presentSentence();
    }

    function escapeHtml(text) {
      return text
        .replaceAll("&", "&amp;")
        .replaceAll("<", "&lt;")
        .replaceAll(">", "&gt;")
        .replaceAll('"', "&quot;")
        .replaceAll("'", "&#039;");
    }
  </script>
</body>
</html>
