create an activity using html, css, and javascript. there are two bars on the screen, one above the other, that fill most of the width of the screen. each bar extends from 0 (when empty) to 20 (when full), but there are no tick marks or labels along the bars. the bottom bar is filled with orange to some integer o between 2 and 19 (inclusive). the top bar is filled with blue to some integer b between o and 20 (inclusive). both bars have their fill value displayed in the center of the filled region, the blue bar to two decimal places and the orange bar as an integer. the right side of the filled region of both bars can be dragged to change the fill quantity, and the displayed number for that bar changes accordingly. the blue bar can be dragged to any value between 0 and 20 (inclusive) while the orange bar snaps only to integer values while being dragged, also between 0 and 20 (inclusive). the user must drag the orange bar to 1 and blue bar to a value, such that ratio between the blue fill and the orange fill is within p% of the originally shown ratio. p is a javascript constant that is initially set to 10. as soon as the user moves drags one of the fill areas, a countdown timer starts. the countdown timer starting time is displayed to 1 decimal place, it is a javascript constant, and it is initially set to 4. if the user creates the unit ratio within the given percentage before the time runs out, then a green checkmark appears over the screen, a new problem is presented, and a progress bar is advanced one unit. if time runs out before the user creates a correct solution, a red x appears over the screen, a new problem is presented, and the progress bar goes back one unit. the progress bar extends from -10 to 10 (without labels or tick marks except for a gray vertical line in the middle indicating 0). 0 displays as nothing, negative numbers display as red fill extending to the left of 0, and positive numbers display as green fill extending to the right of 0. progress cannot go below -10. when progress reaches 10 a function activityComplete() is called.

  <script>
    /***********************
     * Activity constants
     ***********************/
    const PERCENT_TOLERANCE = 10;
    const TIMER_START_SECONDS = 4.0;
    const BAR_MIN = 0;
    const BAR_MAX = 20;
    const PROGRESS_MIN = -10;
    const PROGRESS_MAX = 10;

    /***********************
     * State
     ***********************/
    let originalOrange = 0;
    let originalBlue = 0;
    let targetRatio = 0;

    let orangeValue = 0;
    let blueValue = 0;

    let progress = 0;

    let timerRunning = false;
    let timerRemaining = TIMER_START_SECONDS;
    let timerInterval = null;
    let problemLocked = false;

    /***********************
     * Elements
     ***********************/
    const blueBar = document.getElementById("blueBar");
    const orangeBar = document.getElementById("orangeBar");

    const blueFill = document.getElementById("blueFill");
    const orangeFill = document.getElementById("orangeFill");

    const blueValueLabel = document.getElementById("blueValueLabel");
    const orangeValueLabel = document.getElementById("orangeValueLabel");

    const timerEl = document.getElementById("timer");
    const overlay = document.getElementById("overlay");

    const negativeProgress = document.getElementById("negativeProgress");
    const positiveProgress = document.getElementById("positiveProgress");
    const completeMessage = document.getElementById("completeMessage");

    /***********************
     * Utility functions
     ***********************/
    function randomInteger(min, max) {
      return Math.floor(Math.random() * (max - min + 1)) + min;
    }

    function clamp(value, min, max) {
      return Math.min(max, Math.max(min, value));
    }

    function valueToPercent(value) {
      return (value / BAR_MAX) * 100;
    }

    function getValueFromPointer(event, barElement) {
      const rect = barElement.getBoundingClientRect();
      const x = event.clientX - rect.left;
      const proportion = clamp(x / rect.width, 0, 1);
      return proportion * BAR_MAX;
    }

    /***********************
     * Rendering
     ***********************/
    function renderBars() {
      blueFill.style.width = `${valueToPercent(blueValue)}%`;
      orangeFill.style.width = `${valueToPercent(orangeValue)}%`;

      blueValueLabel.textContent = blueValue.toFixed(2);
      orangeValueLabel.textContent = String(Math.round(orangeValue));

      // Keep labels readable when fill is tiny.
      blueValueLabel.style.display = blueValue <= 0.05 ? "none" : "block";
      orangeValueLabel.style.display = orangeValue <= 0.05 ? "none" : "block";
    }

    function renderTimer() {
      timerEl.textContent = timerRemaining.toFixed(1);
      timerEl.classList.toggle("warning", timerRunning && timerRemaining <= 1.2);
    }

    function renderProgress() {
      const halfTrackPercent = 50;

      if (progress > 0) {
        positiveProgress.style.width = `${(progress / PROGRESS_MAX) * halfTrackPercent}%`;
        negativeProgress.style.width = "0%";
      } else if (progress < 0) {
        positiveProgress.style.width = "0%";
        negativeProgress.style.width = `${(Math.abs(progress) / Math.abs(PROGRESS_MIN)) * halfTrackPercent}%`;
      } else {
        positiveProgress.style.width = "0%";
        negativeProgress.style.width = "0%";
      }
    }

    /***********************
     * Problem lifecycle
     ***********************/
    function newProblem() {
      problemLocked = false;
      timerRunning = false;
      clearInterval(timerInterval);
      timerInterval = null;

      timerRemaining = TIMER_START_SECONDS;
      renderTimer();

      originalOrange = randomInteger(2, 19);
      originalBlue = randomInteger(originalOrange, 20);

      orangeValue = originalOrange;
      blueValue = originalBlue;

      targetRatio = originalBlue / originalOrange;

      renderBars();
    }

    function startTimerIfNeeded() {
      if (timerRunning || problemLocked) return;

      timerRunning = true;
      timerRemaining = TIMER_START_SECONDS;
      renderTimer();

      const startTime = performance.now();

      timerInterval = setInterval(() => {
        const elapsed = (performance.now() - startTime) / 1000;
        timerRemaining = Math.max(0, TIMER_START_SECONDS - elapsed);
        renderTimer();

        if (timerRemaining <= 0) {
          handleFailure();
        }
      }, 50);
    }

    function isCorrect() {
      if (orangeValue !== 1) return false;

      const currentRatio = blueValue / orangeValue;
      const percentDifference = Math.abs(currentRatio - targetRatio) / targetRatio * 100;

      return percentDifference <= PERCENT_TOLERANCE;
    }

    function checkSolution() {
      if (problemLocked) return;

      if (isCorrect()) {
        handleSuccess();
      }
    }

    function showOverlay(symbol, className) {
      overlay.textContent = symbol;
      overlay.className = `overlay-result show ${className}`;

      setTimeout(() => {
        overlay.className = "overlay-result";
      }, 650);
    }

    function handleSuccess() {
      if (problemLocked) return;
      problemLocked = true;

      clearInterval(timerInterval);
      timerInterval = null;
      timerRunning = false;

      progress = clamp(progress + 1, PROGRESS_MIN, PROGRESS_MAX);
      renderProgress();

      showOverlay("✓", "success");

      if (progress >= PROGRESS_MAX) {
        setTimeout(() => {
          activityComplete();
        }, 700);
      } else {
        setTimeout(newProblem, 750);
      }
    }

    function handleFailure() {
      if (problemLocked) return;
      problemLocked = true;

      clearInterval(timerInterval);
      timerInterval = null;
      timerRunning = false;
      timerRemaining = 0;
      renderTimer();

      progress = clamp(progress - 1, PROGRESS_MIN, PROGRESS_MAX);
      renderProgress();

      showOverlay("✕", "failure");

      setTimeout(newProblem, 750);
    }

    /***********************
     * Dragging
     ***********************/
    function enableDragging(barElement, setValueCallback) {
      let dragging = false;

      function move(event) {
        if (!dragging || problemLocked) return;

        event.preventDefault();

        const rawValue = getValueFromPointer(event, barElement);
        setValueCallback(rawValue);

        renderBars();
        checkSolution();
      }

      function stopDrag() {
        dragging = false;
        window.removeEventListener("pointermove", move);
        window.removeEventListener("pointerup", stopDrag);
        window.removeEventListener("pointercancel", stopDrag);
      }

      barElement.addEventListener("pointerdown", event => {
        if (problemLocked) return;

        dragging = true;
        barElement.setPointerCapture?.(event.pointerId);

        startTimerIfNeeded();

        const rawValue = getValueFromPointer(event, barElement);
        setValueCallback(rawValue);

        renderBars();
        checkSolution();

        window.addEventListener("pointermove", move);
        window.addEventListener("pointerup", stopDrag);
        window.addEventListener("pointercancel", stopDrag);
      });
    }

    enableDragging(blueBar, rawValue => {
      blueValue = clamp(rawValue, BAR_MIN, BAR_MAX);
    });

    enableDragging(orangeBar, rawValue => {
      orangeValue = clamp(Math.round(rawValue), BAR_MIN, BAR_MAX);
    });

    /***********************
     * Completion hook
     ***********************/
    function activityComplete() {
      completeMessage.classList.add("show");
      console.log("Activity complete!");
    }

    /***********************
     * Initialize
     ***********************/
    renderProgress();
    newProblem();
  </script>
</body>
</html>
