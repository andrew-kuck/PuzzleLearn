create an activity using html, css, and javascript. show 2 large numbers, one on the left side of the screen and one on the right side of the screen. these numbers must be between 1 and 100 and must have a common factor greater than 1. each starting number a has two branches extending down, with empty circle nodes b and c at the end of each branch. the user can enter a number into b or c. after entering a number for b and clicking away, the c circle takes the value a/b. after entering the number for c and clicking away, the b circle takes the value a/c. If the calculated value is an integer, display in black without decimals. if the calculated value is not an integer, display in red with 2 decimal places of accuracy. when b and c both have values, branches and empty nodes should appear below them as well, 2 branches for b and 2 for c, so that the same process can repeat in the next row. right-clicking any node (including the top number a) makes both of the children nodes blank and completely removes any nodes/branches that extend below the children nodes. double-clicking a node makes the node background color green. on each side of the screen (each factor tree) there can only be one node colored green. if the number from the green node on the left matches the number from the green node on the right, and they are both equal to the greatest common factor of the two top node numbers, then a green checkmark appears over the screen and a new question is presented. a progress bar advances with each correct answer so that it is full after 10 correct answers, at which point the function activityComplete() is called. make everything on the tree such a size that it can fit in the window without lots of scrolling. make it so that double-clicking a green node makes it not green anymore. make sure the the children nodes are centered below the parent node. make sure the lines from the parent to the children are connected. side scroll should still be available when content overflows the container, but there should not be any side scroll if the tree fits in the container.

make the text font smaller if the number is 3 digits and cannot fit

  <script>
    /************************************************************
     * If your LMS or host page already defines activityComplete,
     * remove this placeholder.
     ************************************************************/
    function activityComplete() {
      alert("Activity complete!");
    }

    const GOAL = 10;

    let correctCount = 0;
    let currentGCF = 1;
    let locked = false;

    let leftRoot = null;
    let rightRoot = null;

    let selectedGreen = {
      left: null,
      right: null
    };

    const els = {
      leftTree: document.getElementById("leftTree"),
      rightTree: document.getElementById("rightTree"),
      leftContent: document.getElementById("leftContent"),
      rightContent: document.getElementById("rightContent"),
      leftLines: document.getElementById("leftLines"),
      rightLines: document.getElementById("rightLines"),
      progressBar: document.getElementById("progressBar"),
      scoreText: document.getElementById("scoreText"),
      checkmark: document.getElementById("checkmark")
    };

    function gcd(a, b) {
      a = Math.abs(a);
      b = Math.abs(b);
      while (b) {
        [a, b] = [b, a % b];
      }
      return a;
    }

    function randomInt(min, max) {
      return Math.floor(Math.random() * (max - min + 1)) + min;
    }

    function makeNode(value = null, side, parent = null) {
      return {
        id: crypto.randomUUID(),
        value,
        side,
        parent,
        children: null
      };
    }

    function generateNumbers() {
      let a, b, g;

      do {
        a = randomInt(2, 100);
        b = randomInt(2, 100);
        g = gcd(a, b);
      } while (g <= 1);

      return { a, b, g };
    }

    function newQuestion() {
      locked = false;
      selectedGreen.left = null;
      selectedGreen.right = null;

      const nums = generateNumbers();
      currentGCF = nums.g;

      leftRoot = makeNode(nums.a, "left");
      rightRoot = makeNode(nums.b, "right");

      ensureChildren(leftRoot);
      ensureChildren(rightRoot);

      renderAll();
    }

    function ensureChildren(node) {
      if (!node.children) {
        node.children = [
          makeNode(null, node.side, node),
          makeNode(null, node.side, node)
        ];
      }
    }

    function clearSelectedIfInside(node) {
      for (const side of ["left", "right"]) {
        if (selectedGreen[side] && containsNode(node, selectedGreen[side])) {
          selectedGreen[side] = null;
        }
      }
    }

    function containsNode(root, target) {
      if (!root || !target) return false;
      if (root === target) return true;
      if (!root.children) return false;
      return root.children.some(child => containsNode(child, target));
    }

    function resetChildren(node) {
      clearSelectedIfInsideChildren(node);

      node.children = [
        makeNode(null, node.side, node),
        makeNode(null, node.side, node)
      ];
    }

    function clearSelectedIfInsideChildren(node) {
      if (!node.children) return;

      for (const child of node.children) {
        clearSelectedIfInside(child);
      }
    }

    function pruneDescendants(node) {
      if (!node) return;
      clearSelectedIfInside(node);
      node.children = null;
    }

    function parseNodeValue(text) {
      if (text === null || text === undefined) return null;
      const trimmed = String(text).trim();
      if (trimmed === "") return null;

      const value = Number(trimmed);
      return Number.isFinite(value) ? value : null;
    }

    function isIntegerValue(value) {
      return Number.isInteger(value);
    }

    function displayValue(value) {
      if (value === null || value === undefined || !Number.isFinite(value)) {
        return "";
      }

      return isIntegerValue(value) ? String(value) : value.toFixed(2);
    }

    function valueClass(value) {
      if (value === null || value === undefined || !Number.isFinite(value)) {
        return "";
      }

      return isIntegerValue(value) ? "" : "red";
    }

    function renderAll() {
      renderTree(leftRoot, els.leftTree);
      renderTree(rightRoot, els.rightTree);

      requestAnimationFrame(() => {
        drawLines(leftRoot, els.leftContent, els.leftLines);
        drawLines(rightRoot, els.rightContent, els.rightLines);
      });
    }

    function renderTree(root, container) {
      container.innerHTML = "";
      container.appendChild(renderNode(root));
    }

    function renderNode(node) {
      const wrap = document.createElement("div");
      wrap.className = "tree-node-wrap";

      const circle = document.createElement("div");
      circle.className = [
        "node",
        node.parent ? "" : "root",
        node.value === null ? "blank" : "",
        valueClass(node.value),
        selectedGreen[node.side] === node ? "green" : ""
      ].join(" ").trim();

      circle.dataset.nodeId = node.id;

      const input = document.createElement("input");
      input.value = displayValue(node.value);
      input.placeholder = "";
      input.inputMode = "decimal";
      input.autocomplete = "off";
      input.spellcheck = false;

      if (!node.parent) {
        input.readOnly = true;
        input.tabIndex = -1;
      }

      input.addEventListener("focus", () => {
        if (!input.readOnly) input.select();
      });

      input.addEventListener("blur", () => {
        if (locked || !node.parent) return;
        handleNodeBlur(node, input.value);
      });

      input.addEventListener("keydown", event => {
        if (event.key === "Enter") {
          input.blur();
        }
      });

      circle.addEventListener("contextmenu", event => {
        event.preventDefault();
        if (locked) return;

        resetChildren(node);
        renderAll();
      });

      circle.addEventListener("dblclick", event => {
        event.preventDefault();
        if (locked) return;

        toggleGreen(node);
      });

      circle.appendChild(input);
      wrap.appendChild(circle);

      if (node.children) {
        const children = document.createElement("div");
        children.className = "children";

        for (const child of node.children) {
          children.appendChild(renderNode(child));
        }

        wrap.appendChild(children);
      }

      return wrap;
    }

    function handleNodeBlur(node, rawValue) {
      const newValue = parseNodeValue(rawValue);

      node.value = newValue;

      /*
       * Editing a node changes the math below it, so remove any old
       * descendants from this node.
       */
      if (node.children) {
        pruneDescendants(node);
      }

      const parent = node.parent;
      if (!parent || !parent.children || parent.value === null) {
        renderAll();
        return;
      }

      const sibling = parent.children[0] === node
        ? parent.children[1]
        : parent.children[0];

      if (newValue !== null && newValue !== 0) {
        sibling.value = parent.value / newValue;

        /*
         * The sibling's old descendants are no longer trustworthy after
         * recalculation.
         */
        if (sibling.children) {
          pruneDescendants(sibling);
        }
      }

      /*
       * Once both children have values, each child gets its own blank pair
       * so the process can continue on the next row.
       */
      if (
        parent.children[0].value !== null &&
        parent.children[1].value !== null
      ) {
        ensureChildren(parent.children[0]);
        ensureChildren(parent.children[1]);
      }

      renderAll();
    }

    function toggleGreen(node) {
      const side = node.side;

      if (selectedGreen[side] === node) {
        selectedGreen[side] = null;
      } else {
        selectedGreen[side] = node;
      }

      renderAll();
      checkAnswer();
    }

    function checkAnswer() {
      const left = selectedGreen.left;
      const right = selectedGreen.right;

      if (!left || !right) return;
      if (left.value === null || right.value === null) return;

      const leftMatches = Math.abs(left.value - currentGCF) < 0.000001;
      const rightMatches = Math.abs(right.value - currentGCF) < 0.000001;
      const sameValue = Math.abs(left.value - right.value) < 0.000001;

      if (leftMatches && rightMatches && sameValue) {
        handleCorrect();
      }
    }

    function handleCorrect() {
      if (locked) return;
      locked = true;

      correctCount++;
      updateProgress();

      els.checkmark.classList.add("show");

      setTimeout(() => {
        els.checkmark.classList.remove("show");

        if (correctCount >= GOAL) {
          activityComplete();
        } else {
          newQuestion();
        }
      }, 800);
    }

    function updateProgress() {
      const percent = Math.min(100, (correctCount / GOAL) * 100);
      els.progressBar.style.width = `${percent}%`;
      els.scoreText.textContent = `${correctCount} / ${GOAL} correct`;
    }

    function drawLines(root, contentEl, svgEl) {
      const width = contentEl.scrollWidth;
      const height = contentEl.scrollHeight;

      svgEl.setAttribute("width", width);
      svgEl.setAttribute("height", height);
      svgEl.setAttribute("viewBox", `0 0 ${width} ${height}`);
      svgEl.innerHTML = "";

      const contentRect = contentEl.getBoundingClientRect();

      function centerBottom(el) {
        const r = el.getBoundingClientRect();
        return {
          x: r.left - contentRect.left + r.width / 2,
          y: r.bottom - contentRect.top
        };
      }

      function centerTop(el) {
        const r = el.getBoundingClientRect();
        return {
          x: r.left - contentRect.left + r.width / 2,
          y: r.top - contentRect.top
        };
      }

      function nodeElement(node) {
        return contentEl.querySelector(`[data-node-id="${node.id}"]`);
      }

      function walk(node) {
        if (!node.children) return;

        const parentEl = nodeElement(node);
        if (!parentEl) return;

        const p = centerBottom(parentEl);

        for (const child of node.children) {
          const childEl = nodeElement(child);
          if (!childEl) continue;

          const c = centerTop(childEl);

          const line = document.createElementNS(
            "http://www.w3.org/2000/svg",
            "line"
          );

          line.setAttribute("x1", p.x);
          line.setAttribute("y1", p.y);
          line.setAttribute("x2", c.x);
          line.setAttribute("y2", c.y);
          line.setAttribute("stroke", "var(--line-color)");
          line.setAttribute("stroke-width", "3");
          line.setAttribute("stroke-linecap", "round");

          svgEl.appendChild(line);

          walk(child);
        }
      }

      walk(root);
    }

    window.addEventListener("resize", () => {
      requestAnimationFrame(() => {
        drawLines(leftRoot, els.leftContent, els.leftLines);
        drawLines(rightRoot, els.rightContent, els.rightLines);
      });
    });

    updateProgress();
    newQuestion();
  </script>
</body>
</html>
