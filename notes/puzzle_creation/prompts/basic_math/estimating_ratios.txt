create an activity using html, css, and javascript. show a bar with some percent colored red and some percent colored blue, at random. the percentage of blue should be between 10 and 90. there is an input for the user to guess the ratio of blue to red, entered as in the form b:r. when the user submits their guess, proceed as follows. if the entered ratio b:r is greater than the actual ratio: sequentially superimpose b congruent rectangles over the blue area such that the entire blue area is filled; then sequentially superimpose r rectangles of the same size over the red area, which will leave some amount of red area uncovered; take the percent that is uncovered (out of the entire bar) and add this to an accumulating error count. if the entered ratio b:r is less than the actual ratio: sequentially superimpose r congruent rectangles over the red area such that the entire red area is filled; then sequentially superimpose b rectangles of the same size over the blue area, which will leave some amount of blue area uncovered; take the percent that is uncovered (out of the entire bar) and add this to an accumulating error count. the user may only submit one guess for a question, after which they should be able to see their error for the question and their accumulated error. they can then click a button to be presented with a new question. After they have completed 10 questions, a function activityComplete() is called.

Below is a complete single-file HTML activity. Save it as `index.html` and open it in a browser.

It randomly generates a blue/red bar, lets the user guess the ratio as `b:r`, animates the rectangle-overlay comparison, tracks question error and accumulated error, and calls `activityComplete()` after 10 questions.

  <script>
    const TOTAL_QUESTIONS = 10;

    let questionNumber = 1;
    let accumulatedError = 0;

    let bluePercent = 0;
    let redPercent = 0;

    let guessSubmitted = false;

    const bar = document.getElementById("bar");
    const blueArea = document.getElementById("blueArea");
    const redArea = document.getElementById("redArea");

    const questionCounter = document.getElementById("questionCounter");
    const accumulatedErrorDisplay = document.getElementById("accumulatedError");
    const feedback = document.getElementById("feedback");

    const guessForm = document.getElementById("guessForm");
    const ratioInput = document.getElementById("ratioInput");
    const submitBtn = document.getElementById("submitBtn");
    const nextBtn = document.getElementById("nextBtn");

    // Default completion function.
    // Replace this with your LMS/platform completion code if needed.
    window.activityComplete = window.activityComplete || function () {
      feedback.innerHTML += `
        <p class="success">
          Activity complete! Final accumulated error:
          ${accumulatedError.toFixed(2)}%.
        </p>
      `;
      console.log("activityComplete() called");
    };

    function startQuestion() {
      guessSubmitted = false;

      clearOverlays();

      bluePercent = randomInteger(10, 90);
      redPercent = 100 - bluePercent;

      blueArea.style.width = bluePercent + "%";
      redArea.style.width = redPercent + "%";

      questionCounter.textContent = `Question ${questionNumber} of ${TOTAL_QUESTIONS}`;
      accumulatedErrorDisplay.textContent =
        `Accumulated error: ${accumulatedError.toFixed(2)}%`;

      ratioInput.value = "";
      ratioInput.disabled = false;
      submitBtn.disabled = false;
      nextBtn.classList.add("hidden");

      feedback.textContent = "Make your estimate, then submit one guess.";
      ratioInput.focus();
    }

    function clearOverlays() {
      document
        .querySelectorAll(".overlay-rect, .uncovered-marker")
        .forEach(el => el.remove());
    }

    function randomInteger(min, max) {
      return Math.floor(Math.random() * (max - min + 1)) + min;
    }

    function parseRatio(value) {
      const trimmed = value.trim();
      const match = trimmed.match(/^(\d+)\s*:\s*(\d+)$/);

      if (!match) return null;

      const b = Number(match[1]);
      const r = Number(match[2]);

      if (!Number.isInteger(b) || !Number.isInteger(r) || b <= 0 || r <= 0) {
        return null;
      }

      return { b, r };
    }

    function sleep(ms) {
      return new Promise(resolve => setTimeout(resolve, ms));
    }

    function addOverlayRect(leftPercent, widthPercent) {
      const rect = document.createElement("div");
      rect.className = "overlay-rect";
      rect.style.left = leftPercent + "%";
      rect.style.width = widthPercent + "%";
      bar.appendChild(rect);

      requestAnimationFrame(() => {
        rect.classList.add("show");
      });
    }

    function addUncoveredMarker(leftPercent, widthPercent) {
      if (widthPercent <= 0.0001) return;

      const marker = document.createElement("div");
      marker.className = "uncovered-marker";
      marker.style.left = leftPercent + "%";
      marker.style.width = widthPercent + "%";
      bar.appendChild(marker);

      requestAnimationFrame(() => {
        marker.classList.add("show");
      });
    }

    async function animateRectangles(count, startLeft, rectWidth, delay) {
      for (let i = 0; i < count; i++) {
        addOverlayRect(startLeft + i * rectWidth, rectWidth);
        await sleep(delay);
      }
    }

    async function handleGuess(b, r) {
      clearOverlays();

      /*
        Compare guessed ratio b:r with actual ratio bluePercent:redPercent.

        b:r > blue:red  iff  b / r > bluePercent / redPercent
                         iff  b * redPercent > r * bluePercent
      */
      const comparison = b * redPercent - r * bluePercent;

      let questionError = 0;
      let explanation = "";

      const totalRects = b + r;
      const delay = Math.max(35, Math.min(180, 2200 / totalRects));

      if (comparison > 0) {
        // Guess ratio is greater than the actual ratio.
        // First, fill blue with b congruent rectangles.
        const rectWidth = bluePercent / b;

        await animateRectangles(b, 0, rectWidth, delay);

        // Then place r rectangles of the same size over red.
        await animateRectangles(r, bluePercent, rectWidth, delay);

        const redCovered = r * rectWidth;
        questionError = Math.max(0, redPercent - redCovered);

        const uncoveredLeft = bluePercent + redCovered;
        addUncoveredMarker(uncoveredLeft, questionError);

        explanation = `
          Your guessed ratio <strong>${b}:${r}</strong> is
          <strong>greater</strong> than the actual blue:red ratio.
          The blue area was divided into ${b} congruent rectangles.
          Then ${r} same-size rectangles were placed over the red area,
          leaving <span class="error">${questionError.toFixed(2)}%</span>
          of the whole bar uncovered.
        `;
      } else if (comparison < 0) {
        // Guess ratio is less than the actual ratio.
        // First, fill red with r congruent rectangles.
        const rectWidth = redPercent / r;

        await animateRectangles(r, bluePercent, rectWidth, delay);

        // Then place b rectangles of the same size over blue.
        await animateRectangles(b, 0, rectWidth, delay);

        const blueCovered = b * rectWidth;
        questionError = Math.max(0, bluePercent - blueCovered);

        const uncoveredLeft = blueCovered;
        addUncoveredMarker(uncoveredLeft, questionError);

        explanation = `
          Your guessed ratio <strong>${b}:${r}</strong> is
          <strong>less</strong> than the actual blue:red ratio.
          The red area was divided into ${r} congruent rectangles.
          Then ${b} same-size rectangles were placed over the blue area,
          leaving <span class="error">${questionError.toFixed(2)}%</span>
          of the whole bar uncovered.
        `;
      } else {
        // Exact ratio.
        const rectWidth = bluePercent / b;

        await animateRectangles(b, 0, rectWidth, delay);
        await animateRectangles(r, bluePercent, rectWidth, delay);

        questionError = 0;

        explanation = `
          <span class="success">Exact!</span>
          Your guessed ratio <strong>${b}:${r}</strong> matches the actual
          blue:red ratio. No area is left uncovered, so your error is
          <strong>0.00%</strong>.
        `;
      }

      accumulatedError += questionError;

      feedback.innerHTML = `
        ${explanation}
        <p>
          Question error:
          <strong>${questionError.toFixed(2)}%</strong>
        </p>
        <p>
          Accumulated error:
          <strong>${accumulatedError.toFixed(2)}%</strong>
        </p>
      `;

      accumulatedErrorDisplay.textContent =
        `Accumulated error: ${accumulatedError.toFixed(2)}%`;

      if (questionNumber >= TOTAL_QUESTIONS) {
        nextBtn.classList.add("hidden");
        submitBtn.disabled = true;
        ratioInput.disabled = true;
        activityComplete();
      } else {
        nextBtn.classList.remove("hidden");
      }
    }

    guessForm.addEventListener("submit", async function (event) {
      event.preventDefault();

      if (guessSubmitted) return;

      const parsed = parseRatio(ratioInput.value);

      if (!parsed) {
        feedback.innerHTML = `
          <span class="error">
            Please enter a valid ratio using positive whole numbers, like 3:2.
          </span>
        `;
        return;
      }

      guessSubmitted = true;
      ratioInput.disabled = true;
      submitBtn.disabled = true;
      nextBtn.classList.add("hidden");

      feedback.textContent = "Animating your comparison...";

      await handleGuess(parsed.b, parsed.r);
    });

    nextBtn.addEventListener("click", function () {
      if (questionNumber < TOTAL_QUESTIONS) {
        questionNumber++;
        startQuestion();
      }
    });

    startQuestion();
  </script>
</body>
</html>
