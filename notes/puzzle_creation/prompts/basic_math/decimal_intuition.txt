/*
  This activity avoids JavaScript floating-point weirdness by storing all
  changing numbers as exact rational values using BigInt numerators and
  denominators.
*/

class Rational {
  constructor(num, den = 1n) {
    num = BigInt(num);
    den = BigInt(den);

    if (den === 0n) {
      throw new Error("Denominator cannot be zero.");
    }

    if (den < 0n) {
      num = -num;
      den = -den;
    }

    if (num === 0n) {
      this.num = 0n;
      this.den = 1n;
      return;
    }

    const g = Rational.gcd(Rational.absBig(num), den);
    this.num = num / g;
    this.den = den / g;
  }

  static absBig(n) {
    return n < 0n ? -n : n;
  }

  static gcd(a, b) {
    while (b !== 0n) {
      const t = b;
      b = a % b;
      a = t;
    }
    return a;
  }

  static fromInt(n) {
    return new Rational(BigInt(n), 1n);
  }

  static fromBigInt(n) {
    return new Rational(n, 1n);
  }

  static fromTenths(tenths) {
    return new Rational(BigInt(tenths), 10n);
  }

  add(other) {
    return new Rational(
      this.num * other.den + other.num * this.den,
      this.den * other.den
    );
  }

  sub(other) {
    return new Rational(
      this.num * other.den - other.num * this.den,
      this.den * other.den
    );
  }

  mul(other) {
    return new Rational(this.num * other.num, this.den * other.den);
  }

  div(other) {
    if (other.num === 0n) {
      throw new Error("Cannot divide by zero.");
    }
    return new Rational(this.num * other.den, this.den * other.num);
  }

  abs() {
    return new Rational(Rational.absBig(this.num), this.den);
  }

  neg() {
    return new Rational(-this.num, this.den);
  }

  truncBigInt() {
    return this.num / this.den;
  }

  isZero() {
    return this.num === 0n;
  }

  lessThanOrEqualZero() {
    return this.num <= 0n;
  }
}

/*
  Main number formatting:
  - Maximum of 3 decimal places.
  - Trailing zeros are removed.
  - If the value is an integer, the decimal point remains with no digits.
  - If more than 3 decimal places are needed and the hidden digits are not all
    zero, a one-digit-width ellipsis is shown.
*/
function formatMainRational(r) {
  const maxPlaces = 3;

  const negative = r.num < 0n;
  let n = Rational.absBig(r.num);
  const d = r.den;

  const integerPart = n / d;
  let remainder = n % d;

  let digits = [];

  for (let i = 0; i < maxPlaces && remainder !== 0n; i++) {
    remainder *= 10n;
    const digit = remainder / d;
    remainder %= d;
    digits.push(digit.toString());
  }

  const needsEllipsis = remainder !== 0n;

  if (!needsEllipsis) {
    while (digits.length && digits[digits.length - 1] === "0") {
      digits.pop();
    }
  }

  const sign = negative ? "-" : "";
  let html = `${sign}${integerPart.toString()}.`;

  if (digits.length) {
    html += digits.join("");
  }

  if (needsEllipsis) {
    html += `<span class="oneDigitEllipsis">…</span>`;
  }

  return html;
}

function roundRationalToTenths(r) {
  let sign = 1n;
  let n = r.num;

  if (n < 0n) {
    sign = -1n;
    n = -n;
  }

  let scaled = n * 10n;
  let q = scaled / r.den;
  const rem = scaled % r.den;

  if (rem * 2n >= r.den) {
    q += 1n;
  }

  return q * sign;
}

function formatError(r) {
  if (r.num <= 0n) {
    return "0.0";

  const tenths = roundRationalToTenths(r);
  const whole = tenths / 10n;
  const dec = Rational.absBig(tenths % 10n);
  return `${whole.toString()}.${dec.toString()}`;
}

function positionCircles() {

function addDragBehavior(circle) {
  let pointerId = null;
  let offsetX = 0;
  let offsetY = 0;

  circle.addEventListener("pointerdown", event => {
    if (!questionActive || submitting) return;

    pointerId = event.pointerId;
    circle.setPointerCapture(pointerId);

    const circleRect = circle.getBoundingClientRect();
    const areaRect = playArea.getBoundingClientRect();

    offsetX = event.clientX - circleRect.left;
    offsetY = event.clientY - circleRect.top;

    circle.classList.add("dragging");

    circle.style.left = `${event.clientX - areaRect.left - offsetX}px`;
    circle.style.top = `${event.clientY - areaRect.top - offsetY}px`;
  });

  circle.addEventListener("pointermove", event => {
    if (pointerId !== event.pointerId || !circle.classList.contains("dragging")) return;

    const areaRect = playArea.getBoundingClientRect();

    circle.style.left = `${event.clientX - areaRect.left - offsetX}px`;
    circle.style.top = `${event.clientY - areaRect.top - offsetY}px`;
  });

  circle.addEventListener("pointerup", event => {
    if (pointerId !== event.pointerId) return;

    circle.releasePointerCapture(pointerId);
    pointerId = null;

    circle.classList.remove("dragging");

    if (isPointInsideCenter(event.clientX, event.clientY)) {
      applyCircle(circle);
    } else {
      positionCircles();
    }
  });

  circle.addEventListener("pointercancel", () => {
    pointerId = null;
    circle.classList.remove("dragging");
    positionCircles();
  });
}

  circle.remove();
  createCircle(slot);

function animateErrorDelta(delta, durationMs, subtract = false) {
  return new Promise(resolve => {
    const start = errorTotal;
    const signedDelta = subtract ? delta.neg() : delta;
    const started = performance.now();

    function frame(now) {
      const elapsed = Math.min(durationMs, now - started);
      const progress = new Rational(BigInt(Math.round(elapsed)), BigInt(durationMs));

      errorTotal = start.add(signedDelta.mul(progress));
      updateErrorDisplay();

      if (elapsed < durationMs) {
        requestAnimationFrame(frame);
      } else {
        errorTotal = start.add(signedDelta);
        updateErrorDisplay();
        resolve();
      }
    }

    requestAnimationFrame(frame);
  });
}

function animateIntegerCount(fromBig, toBig) {
  return new Promise(resolve => {
    const diff = toBig - fromBig;
    const absDiff = diff < 0n ? -diff : diff;

    if (absDiff === 0n) {
      currentValue = Rational.fromBigInt(toBig);
      updateMainNumberDisplay();
      resolve();
      return;
    }

    const durationMs = absDiff > 120n ? 1800 : Math.max(450, Number(absDiff) * 24);
    const startError = errorTotal;
    const deltaError = Rational.fromBigInt(absDiff);
    const started = performance.now();

    function frame(now) {
      const elapsedNumber = Math.min(durationMs, now - started);
      const elapsed = BigInt(Math.round(elapsedNumber));
      const duration = BigInt(durationMs);

      const moved = diff * elapsed / duration;
      const shown = fromBig + moved;

      currentValue = Rational.fromBigInt(shown);
      updateMainNumberDisplay();

      const progress = new Rational(elapsed, duration);
      errorTotal = startError.add(deltaError.mul(progress));
      updateErrorDisplay();

      if (elapsedNumber < durationMs) {
        requestAnimationFrame(frame);
      } else {
        currentValue = Rational.fromBigInt(toBig);
        errorTotal = startError.add(deltaError);
        updateMainNumberDisplay();
        updateErrorDisplay();
        resolve();
      }
    }

    requestAnimationFrame(frame);
  });
}

function fastDrainRemainingTime() {
  return new Promise(resolve => {
    const remaining = timerTenths;

    if (remaining <= 0) {
      timerTenths = 0;
      formatTimer();
      resolve();
      return;
    }

    const startError = errorTotal;
    const timeDelta = new Rational(BigInt(remaining), 10n);
    const durationMs = 800;
    const started = performance.now();

    function frame(now) {
      const elapsedNumber = Math.min(durationMs, now - started);
      const elapsed = BigInt(Math.round(elapsedNumber));
      const duration = BigInt(durationMs);

      const progress = new Rational(elapsed, duration);

      const remainingShown = Math.ceil(remaining * (1 - elapsedNumber / durationMs));
      timerTenths = Math.max(0, remainingShown);
      formatTimer();

      errorTotal = startError.sub(timeDelta.mul(progress));
      updateErrorDisplay();

      if (elapsedNumber < durationMs) {
        requestAnimationFrame(frame);
      } else {
        timerTenths = 0;
        errorTotal = startError.sub(timeDelta);
        formatTimer();
        updateErrorDisplay();
        resolve();
      }
    }

    requestAnimationFrame(frame);
  });
}

  submitBtn.disabled = true;

  circleLayer.innerHTML = "";

  /*
    First, the decimal part disappears. The absolute removed decimal amount is
    added to the running error total.
  */
  const truncated = currentValue.truncBigInt();
  const truncatedValue = Rational.fromBigInt(truncated);
  const removedDecimal = currentValue.sub(truncatedValue).abs();

  currentValue = truncatedValue;
  updateMainNumberDisplay();

  if (!removedDecimal.isZero()) {
    await animateErrorDelta(removedDecimal, 360, false);
  }

  /*
    Then the remaining integer counts up or down to the target integer.
    That absolute distance is added to the error total.
  */
  await animateIntegerCount(truncated, BigInt(target));

  /*
    Once the middle number equals the target, the remaining timer drains quickly
    to zero. That remaining time is subtracted from the error total.
  */
  await fastDrainRemainingTime();

  if (errorTotal.lessThanOrEqualZero()) {
    completed = true;
    errorTotal = Rational.fromInt(0);
    updateErrorDisplay();
    window.activityComplete();
    return;
  }

  await delay(450);
  showReadyPrompt();
}
