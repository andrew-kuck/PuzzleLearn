    const labels = [

    const placeValues = [10000, 1000, 100, 10, 1];

    const state = {
      dividend: 0,
      divisor: 0,
      digits: [],
      stage: 1,

    const dividendDisplay = document.getElementById("dividendDisplay");
    const divisorDisplay = document.getElementById("divisorDisplay");
    const fractionsRow = document.getElementById("fractionsRow");
    const controls = document.getElementById("controls");
    const message = document.getElementById("message");
    const instructions = document.getElementById("instructions");
    const stageLabel = document.getElementById("stageLabel");

    function randomInt(min, max) {
      return Math.floor(Math.random() * (max - min + 1)) + min;
    }

    function makeProblem() {
      state.dividend = randomInt(10000, 99999);
      state.divisor = randomInt(2, 100);
      state.digits = String(state.dividend).split("").map(Number);
      state.stage = 1;
      state.stage2Started = false;
      state.currentNumerators = [...state.digits];

      quotientInput.value = "";

      dividendDisplay.textContent = state.dividend;
      divisorDisplay.textContent = state.divisor;

      controls.style.display = "none";

    function renderStage1() {

    function checkStage1() {
      const nums = [...document.querySelectorAll(".stage1-num")];
      const dens = [...document.querySelectorAll(".stage1-den")];

      const allFilled = nums.every(input => input.value !== "") &&
                        dens.every(input => input.value !== "");

      if (!allFilled) {
        message.textContent = "";
        return;
      }

      const numeratorsCorrect = nums.every((input, i) => {
        return Number(input.value) === state.digits[i];

      const denominatorsCorrect = dens.every(input => {

    function attachStage2Events() {
      document.querySelectorAll(".work-input").forEach(input => {
        input.addEventListener("input", handleWorkInput);
      });

      document.querySelectorAll(".rem-chip").forEach(chip => {
        chip.addEventListener("dragstart", handleDragStart);
      });

      document.querySelectorAll(".num-display").forEach(target => {
        target.addEventListener("dragover", handleDragOver);
        target.addEventListener("dragleave", handleDragLeave);
        target.addEventListener("drop", handleDrop);
      });
    }

    function handleWorkInput(event) {
      const input = event.target;
      const index = Number(input.dataset.index);
      const chip = document.querySelector(`.rem-chip[data-index="${index}"]`);

      const parsed = parseWorkString(input.value);

      if (parsed) {
        chip.textContent = parsed.remainder;
        chip.dataset.remainder = parsed.remainder;
        chip.style.display = "inline-block";
        chip.classList.remove("used");
      } else {
        chip.textContent = "";
        chip.dataset.remainder = "";
        chip.style.display = "none";
      }
    }

    function parseWorkString(value) {
      const match = String(value).trim().match(/^(-?\d+)\s*r\s*(-?\d+)$/i);

      if (!match) return null;

      return {
        quotient: Number(match[1]),
        remainder: Number(match[2])
      };
    }

    function handleDragStart(event) {
      const chip = event.target;

      if (chip.classList.contains("used")) {
        event.preventDefault();
        return;
      }

      const originIndex = Number(chip.dataset.index);
      const remainder = Number(chip.dataset.remainder);

      if (!Number.isFinite(remainder)) {
        event.preventDefault();
        return;
      }

      const payload = {
        originIndex,
        remainder
      };

      event.dataTransfer.setData("application/json", JSON.stringify(payload));
      event.dataTransfer.effectAllowed = "move";
    }

    function handleDragOver(event) {
      event.preventDefault();
      event.currentTarget.classList.add("drop-hover");
      event.dataTransfer.dropEffect = "move";
    }

    function handleDragLeave(event) {
      event.currentTarget.classList.remove("drop-hover");
    }

    function handleDrop(event) {
      event.preventDefault();

      const target = event.currentTarget;
      target.classList.remove("drop-hover");

      let payload;

      try {
        payload = JSON.parse(event.dataTransfer.getData("application/json"));
      } catch {
        return;
      }

      const originIndex = Number(payload.originIndex);
      const targetIndex = Number(target.dataset.index);
      const remainder = Number(payload.remainder);

      if (
        !Number.isFinite(originIndex) ||
        !Number.isFinite(targetIndex) ||
        !Number.isFinite(remainder)
      ) {
        return;
      }

      // The prompt says the remainder is dragged to another fraction.
      // Dropping on the same numerator does nothing.
      if (originIndex === targetIndex) {
        return;
      }

      addRemainderToNumerator(originIndex, targetIndex, remainder);
      freezeOriginWork(originIndex);
      refreshNumeratorDisplays();
    }

    function addRemainderToNumerator(originIndex, targetIndex, remainder) {
      /*
        A remainder belongs to the place it came from.

        Example:
        - remainder 5 in ten-thousands represents 5 * 10000 = 50000.
        - Dropped on thousands numerator: 50000 / 1000 = 50.
        - Dropped on hundreds numerator: 50000 / 100 = 500.
        - Dropped on tens numerator: 50000 / 10 = 5000.
        - Dropped on ones numerator: 50000 / 1 = 50000.
      */
      const representedQuantity = remainder * placeValues[originIndex];
      const amountToAdd = representedQuantity / placeValues[targetIndex];

      state.currentNumerators[targetIndex] =
        cleanNumber(state.currentNumerators[targetIndex] + amountToAdd);
    }

    function freezeOriginWork(originIndex) {
      const input = document.querySelector(`.work-input[data-index="${originIndex}"]`);
      const chip = document.querySelector(`.rem-chip[data-index="${originIndex}"]`);

      if (input) {
        input.disabled = true;
      }

      if (chip) {
        chip.classList.add("used");
        chip.style.display = "none";
        chip.textContent = "";
      }
    }

    function refreshNumeratorDisplays() {
      document.querySelectorAll(".num-display").forEach(display => {
        const index = Number(display.dataset.index);
        display.textContent = formatNumber(state.currentNumerators[index]);
      });
    }

    function resetStage2() {
      state.currentNumerators = [...state.digits];

      quotientInput.value = "";
      remainderInput.value = "";

      document.querySelectorAll(".work-input").forEach(input => {
        input.value = "";
        input.disabled = false;
      });

      document.querySelectorAll(".rem-chip").forEach(chip => {
        chip.textContent = "";
        chip.dataset.remainder = "";
        chip.classList.remove("used");
        chip.style.display = "none";
      });

      refreshNumeratorDisplays();

      message.textContent = "Stage 2 has been reset.";
    }

    function checkFinalAnswer() {
      const userQuotient = Number(quotientInput.value);
      const userRemainder = Number(remainderInput.value);

      const correctQuotient = Math.floor(state.dividend / state.divisor);
      const correctRemainder = state.dividend % state.divisor;

      const correct =
        userQuotient === correctQuotient &&
        userRemainder === correctRemainder;

      if (correct) {
        showOverlay("✓", true);
        state.score++;
        updateProgress();

        if (state.score >= state.maxScore) {
          message.textContent = "Excellent work — activity complete!";
          setTimeout(() => {
            activityComplete();
          }, 650);
        } else {
          setTimeout(makeProblem, 800);
        }
      } else {
        showOverlay("×", false);
        message.textContent = "Not quite. Try again — the same problem stays on screen.";
      }
    }

    function showOverlay(symbol, isCorrect) {
      overlay.textContent = symbol;
      overlay.className = `overlay ${isCorrect ? "correct" : "incorrect"}`;
      overlay.style.display = "flex";

      setTimeout(() => {
        overlay.style.display = "none";
      }, 650);
    }

    function updateProgress() {
      const percent = Math.min(100, (state.score / state.maxScore) * 100);
      progressBar.style.width = `${percent}%`;
    }

    function cleanNumber(value) {
      // Avoids floating-point artifacts like 5.399999999999999.
      return Number(Number(value).toFixed(10));
    }

    function formatNumber(value) {
      /*
        Displays normal decimal places without unnecessary trailing zeroes.
        Examples:
          5      -> "5"
          5.4    -> "5.4"
          5.04   -> "5.04"
          5.4000 -> "5.4"
      */
      return cleanNumber(value).toString();
    }

    submitBtn.addEventListener("click", checkFinalAnswer);
    resetBtn.addEventListener("click", resetStage2);

    makeProblem();
  </script>
</body>
