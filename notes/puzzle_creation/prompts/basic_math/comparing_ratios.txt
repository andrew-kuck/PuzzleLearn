create an activity using html, css, and javascript. show two ratios on the screen, one to the left and one to the right. each ratio a:b should consist of two random integers a and b such that both a and b are between 1 and 50 inclusive. the two ratios cannot be equivalent to each other. the user must click on the ratio that is bigger. after the user clicks on a ratio, the integer b of the clicked ratio incrememtally counts down to 1, and with each increment the value a changes so that the ratio retains its value. at each change a is displayed to 2 decimal points of precision. after the clicked ratio counts down to a unit ratio in this way, the ratio that was not clicked on counts down to a unit ratio in the same way. after both ratios have been reduced to unit ratios, the value a of each ratio is highlighted blue, then after a pause the value a of the larger ratio is highlighted green and the value a of the smaller ratio is highlighted red. then after another pause the entire larger ratio is highlighted green. if this is the ratio the user clicked on, a green checkmark appears on the screen, a new problem is presented, and a progress bar is advanced one unit. if the larger ratio is not the ratio the user clicked on, a red x appears over the screen, a new problem is presented, and the progress bar goes back one unit. the progress bar extends from -10 to 10 (without labels or tick marks except for a gray vertical line in the middle indicating 0). 0 displays as nothing, negative numbers display as red fill extending to the left of 0, and positive numbers display as green fill extending to the right of 0. progress cannot go below -10. when progress reaches 10 a function activityComplete() is called.

Below is a complete single-file HTML/CSS/JavaScript activity. Save it as `ratio-activity.html` and open it in a browser.

  <script>
    const leftCard = document.getElementById("leftCard");
    const rightCard = document.getElementById("rightCard");

    const leftA = document.getElementById("leftA");
    const leftB = document.getElementById("leftB");
    const rightA = document.getElementById("rightA");
    const rightB = document.getElementById("rightB");

    const positiveFill = document.getElementById("positiveFill");
    const negativeFill = document.getElementById("negativeFill");
    const overlayMark = document.getElementById("overlayMark");

    let currentProblem = null;
    let progress = 0;
    let locked = false;

    const STEP_DELAY = 300;
    const PAUSE_AFTER_UNIT = 600;
    const PAUSE_AFTER_BLUE = 800;
    const PAUSE_AFTER_COLOR_VALUES = 900;
    const FEEDBACK_PAUSE = 900;

    function activityComplete() {
      alert("Activity complete! Great work.");
    }

    function randomInt(min, max) {
      return Math.floor(Math.random() * (max - min + 1)) + min;
    }

    function sleep(ms) {
      return new Promise(resolve => setTimeout(resolve, ms));
    }

    function makeRandomRatio() {
      return {
        originalA: randomInt(1, 50),
        originalB: randomInt(1, 50)
      };
    }

    function ratioValue(ratio) {
      return ratio.originalA / ratio.originalB;
    }

    function generateProblem() {
      let left;
      let right;

      do {
        left = makeRandomRatio();
        right = makeRandomRatio();
      } while (left.originalA * right.originalB === right.originalA * left.originalB);

      const leftValue = ratioValue(left);
      const rightValue = ratioValue(right);

      return {
        left,
        right,
        largerSide: leftValue > rightValue ? "left" : "right"
      };
    }

    function clearHighlights() {
      [leftCard, rightCard].forEach(card => {
        card.classList.remove("whole-green");
      });

      [leftA, rightA].forEach(el => {
        el.classList.remove("highlight-blue", "highlight-green", "highlight-red");
      });
    }

    function displayInitialProblem() {
      locked = false;
      clearHighlights();

      currentProblem = generateProblem();

      leftA.textContent = currentProblem.left.originalA;
      leftB.textContent = currentProblem.left.originalB;

      rightA.textContent = currentProblem.right.originalA;
      rightB.textContent = currentProblem.right.originalB;

      leftCard.classList.remove("disabled");
      rightCard.classList.remove("disabled");

      leftCard.setAttribute(
        "aria-label",
        `Left ratio ${currentProblem.left.originalA} to ${currentProblem.left.originalB}`
      );

      rightCard.setAttribute(
        "aria-label",
        `Right ratio ${currentProblem.right.originalA} to ${currentProblem.right.originalB}`
      );
    }

    function updateProgressBar() {
      const halfBarPercent = 50;

      if (progress > 0) {
        positiveFill.style.width = `${(progress / 10) * halfBarPercent}%`;
        negativeFill.style.width = "0";
      } else if (progress < 0) {
        negativeFill.style.width = `${(Math.abs(progress) / 10) * halfBarPercent}%`;
        positiveFill.style.width = "0";
      } else {
        positiveFill.style.width = "0";
        negativeFill.style.width = "0";
      }
    }

    async function countDownToUnit(side) {
      const ratio = currentProblem[side];
      const value = ratioValue(ratio);

      const aEl = side === "left" ? leftA : rightA;
      const bEl = side === "left" ? leftB : rightB;

      for (let b = ratio.originalB; b >= 1; b--) {
        const adjustedA = value * b;

        aEl.textContent = adjustedA.toFixed(2);
        bEl.textContent = b;

        await sleep(STEP_DELAY);
      }
    }

    async function handleChoice(chosenSide) {
      if (locked) return;
      locked = true;

      leftCard.classList.add("disabled");
      rightCard.classList.add("disabled");

      const otherSide = chosenSide === "left" ? "right" : "left";

      await countDownToUnit(chosenSide);
      await sleep(PAUSE_AFTER_UNIT);

      await countDownToUnit(otherSide);
      await sleep(PAUSE_AFTER_UNIT);

      await revealAnswer(chosenSide);
    }

    async function revealAnswer(chosenSide) {
      const largerSide = currentProblem.largerSide;
      const smallerSide = largerSide === "left" ? "right" : "left";

      const largerA = largerSide === "left" ? leftA : rightA;
      const smallerA = smallerSide === "left" ? leftA : rightA;

      const largerCard = largerSide === "left" ? leftCard : rightCard;

      leftA.classList.add("highlight-blue");
      rightA.classList.add("highlight-blue");

      await sleep(PAUSE_AFTER_BLUE);

      leftA.classList.remove("highlight-blue");
      rightA.classList.remove("highlight-blue");

      largerA.classList.add("highlight-green");
      smallerA.classList.add("highlight-red");

      await sleep(PAUSE_AFTER_COLOR_VALUES);

      largerCard.classList.add("whole-green");

      await sleep(PAUSE_AFTER_COLOR_VALUES);

      const correct = chosenSide === largerSide;

      if (correct) {
        progress = Math.min(10, progress + 1);
        showOverlay("✓", true);
      } else {
        progress = Math.max(-10, progress - 1);
        showOverlay("✕", false);
      }

      updateProgressBar();

      await sleep(FEEDBACK_PAUSE);
      hideOverlay();

      if (progress >= 10) {
        activityComplete();
      } else {
        displayInitialProblem();
      }
    }

    function showOverlay(symbol, correct) {
      overlayMark.textContent = symbol;
      overlayMark.classList.remove("correct", "incorrect", "show");

      void overlayMark.offsetWidth;

      overlayMark.classList.add(correct ? "correct" : "incorrect");
      overlayMark.classList.add("show");
    }

    function hideOverlay() {
      overlayMark.classList.remove("show", "correct", "incorrect");
      overlayMark.textContent = "";
    }

    leftCard.addEventListener("click", () => handleChoice("left"));
    rightCard.addEventListener("click", () => handleChoice("right"));

    updateProgressBar();
    displayInitialProblem();
  </script>
</body>
