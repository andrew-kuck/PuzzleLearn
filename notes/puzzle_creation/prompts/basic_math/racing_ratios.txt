create an activity using html, css, and javascript. show a blue car racing around a track and an orange car at rest. the blue car is moving at some speed l/m, where l is the number of laps and m is the number of minutes. l should be a random multiple of 5 between 5 and 100 (inclusive), and m should be an integer between 1 and 4 (inclusive). l and m must be chosen such that they do not have a common divisor other than 1. the user has inputs for selecting the number of laps and number of minutes for the orange car. these numbers have the same restrictions as those on the blue car (l between 5 and 100 and a multiple of 5, m between 1 and 4). the orange car will move at the speed indicated by the user inputs for laps and minutes. at any time the user can submit. if the input matches the values for the blue car exactly, a green checkmark appears over the screen and the user is given a new question. if the input is equivalent to the blue car numbers, but not in reduced form, the user is warned to reduce their answer, and the same problem remains. if the answer is incorrect, then a red x appears over the screen and the same problem remains. with each correct answer a progress bar is advanced, such that it is full after 5 correct answers, at which point the function activityComplete() is called.

keep the range of possible values for l and m the same, but the overall ratio l/m must be less than or equal to 20 (that is, 20 laps per minute)

both cars should follow the exact same path around the track. also, the finish line should perfectly span the width of the track

the finish line is now too long (along the length of the track) and it still does not span the width of the track

Below is a complete HTML/CSS/JavaScript activity in one file. It shows a blue car moving around a track at a hidden reduced speed of `l / m` laps per minute, while the user controls the orange car by selecting laps and minutes. The user must match the blue car’s exact reduced values.

  <script>
    /*************************************************************
     * Activity settings
     *************************************************************/
    const QUESTIONS_TO_COMPLETE = 5;

    /*
      Animation scale:
      The math rate is laps per minute.

      To keep the animation watchable, this activity uses a simulated minute.
      Here, 1 math minute equals 18 real seconds.
    */
    const REAL_SECONDS_PER_MATH_MINUTE = 18;

    /*************************************************************
     * DOM references
     *************************************************************/
    const blueCar = document.getElementById("blueCar");
    const orangeCar = document.getElementById("orangeCar");
    const lapsInput = document.getElementById("lapsInput");
    const minutesInput = document.getElementById("minutesInput");
    const submitBtn = document.getElementById("submitBtn");
    const statusEl = document.getElementById("status");
    const blueSpeedDisplay = document.getElementById("blueSpeedDisplay");
    const orangeSpeedDisplay = document.getElementById("orangeSpeedDisplay");
    const progressText = document.getElementById("progressText");
    const progressFill = document.getElementById("progressFill");
    const checkOverlay = document.getElementById("checkOverlay");
    const xOverlay = document.getElementById("xOverlay");

    /*************************************************************
     * State
     *************************************************************/
    let blueLaps = 5;
    let blueMinutes = 1;
    let correctCount = 0;

    let startTime = performance.now();

    /*************************************************************
     * Utility functions
     *************************************************************/
    function gcd(a, b) {
      a = Math.abs(a);
      b = Math.abs(b);

      while (b !== 0) {
        const temp = b;
        b = a % b;
        a = temp;
      }

      return a;
    }

    function randomInt(min, max) {
      return Math.floor(Math.random() * (max - min + 1)) + min;
    }

    function randomChoice(array) {
      return array[Math.floor(Math.random() * array.length)];
    }

    function isValidLaps(laps) {
      return laps >= 5 && laps <= 100 && laps % 5 === 0;
    }

    function isValidMinutes(minutes) {
      return minutes >= 1 && minutes <= 4 && Number.isInteger(minutes);
    }

    function isReduced(laps, minutes) {
      return gcd(laps, minutes) === 1;
    }

    function sameRate(l1, m1, l2, m2) {
      return l1 * m2 === l2 * m1;
    }

    function showOverlay(type) {
      const overlay = type === "check" ? checkOverlay : xOverlay;
      overlay.classList.add("show");

      setTimeout(() => {
        overlay.classList.remove("show");
      }, 650);
    }

    function setStatus(message, type = "") {
      statusEl.textContent = message;
      statusEl.className = "status";

      if (type) {
        statusEl.classList.add(type);
      }
    }

    /*************************************************************
     * Generate answer choices
     *************************************************************/
    function populateInputs() {
      for (let laps = 5; laps <= 100; laps += 5) {
        const option = document.createElement("option");
        option.value = laps;
        option.textContent = laps;
        lapsInput.appendChild(option);
      }

      for (let minutes = 1; minutes <= 4; minutes++) {
        const option = document.createElement("option");
        option.value = minutes;
        option.textContent = minutes;
        minutesInput.appendChild(option);
      }

      lapsInput.value = "5";
      minutesInput.value = "1";
    }

    /*************************************************************
     * Problem generation
     *************************************************************/
    function generateNewProblem() {
      const possiblePairs = [];

      for (let laps = 5; laps <= 100; laps += 5) {
        for (let minutes = 1; minutes <= 4; minutes++) {
          if (gcd(laps, minutes) === 1) {
            possiblePairs.push({ laps, minutes });
          }
        }
      }

      const pair = randomChoice(possiblePairs);

      blueLaps = pair.laps;
      blueMinutes = pair.minutes;

      blueSpeedDisplay.textContent = "? laps / ? minutes";

      setStatus("New race! Match the blue car’s speed.", "");

      /*
        Reset the shared animation clock so both cars begin from the start line
        on each new question.
      */
      startTime = performance.now();
    }

    /*************************************************************
     * Progress
     *************************************************************/
    function updateProgress() {
      progressText.textContent = correctCount;
      const percent = (correctCount / QUESTIONS_TO_COMPLETE) * 100;
      progressFill.style.width = `${percent}%`;
    }

    function activityComplete() {
      setStatus("Activity complete! Great racing-rate work.", "success");

      /*
        If your platform already defines activityComplete(),
        replace this function or connect this call to that system.
      */
      console.log("activityComplete() called");
    }

    /*************************************************************
     * Orange car display
     *************************************************************/
    function updateOrangeDisplay() {
      const laps = Number(lapsInput.value);
      const minutes = Number(minutesInput.value);
      orangeSpeedDisplay.textContent = `${laps} laps / ${minutes} minute${
        minutes === 1 ? "" : "s"
      }`;
    }

    lapsInput.addEventListener("change", updateOrangeDisplay);
    minutesInput.addEventListener("change", updateOrangeDisplay);

    /*************************************************************
     * Submit logic
     *************************************************************/
    submitBtn.addEventListener("click", () => {
      const userLaps = Number(lapsInput.value);
      const userMinutes = Number(minutesInput.value);

      if (!isValidLaps(userLaps) || !isValidMinutes(userMinutes)) {
        setStatus(
          "Please choose laps from 5 to 100 in multiples of 5, and minutes from 1 to 4.",
          "error"
        );
        showOverlay("x");
        return;
      }

      const exactMatch =
        userLaps === blueLaps && userMinutes === blueMinutes;

      const equivalentRate = sameRate(
        userLaps,
        userMinutes,
        blueLaps,
        blueMinutes
      );

      const userReduced = isReduced(userLaps, userMinutes);

      if (exactMatch) {
        showOverlay("check");
        setStatus(
          `Correct! ${blueLaps} laps / ${blueMinutes} minute${
            blueMinutes === 1 ? "" : "s"
          } is the exact reduced rate.`,
          "success"
        );

        correctCount++;
        updateProgress();

        if (correctCount >= QUESTIONS_TO_COMPLETE) {
          setTimeout(() => {
            activityComplete();
          }, 700);
        } else {
          setTimeout(() => {
            generateNewProblem();
          }, 900);
        }
      } else if (equivalentRate && !userReduced) {
        setStatus(
          "That rate is equivalent, but it is not in reduced form. Reduce your answer and try again.",
          "warning"
        );
      } else {
        showOverlay("x");
        setStatus("Not quite. Try to match the blue car’s rate.", "error");
      }
    });

    /*************************************************************
     * Animation
     *************************************************************/
    function positionCar(car, lapsPerMinute, laneOffset, elapsedSeconds) {
      /*
        lapsPerMinute is the mathematical speed.
        Convert real elapsed seconds into simulated math minutes.
      */
      const elapsedMathMinutes =
        elapsedSeconds / REAL_SECONDS_PER_MATH_MINUTE;

      const lapsCompleted = lapsPerMinute * elapsedMathMinutes;

      /*
        Start at the top of the track.
        Angle increases clockwise.
      */
      const angle = lapsCompleted * 2 * Math.PI - Math.PI / 2;

      /*
        Ellipse dimensions.
        The orange car is placed slightly inside the blue car's lane.
      */
      const centerX = 320;
      const centerY = 180;
      const radiusX = 250 - laneOffset;
      const radiusY = 125 - laneOffset * 0.45;

      const x = centerX + radiusX * Math.cos(angle);
      const y = centerY + radiusY * Math.sin(angle);

      /*
        Rotate the car to face along the direction of motion.
      */
      const rotationDegrees = (angle * 180) / Math.PI + 90;

      car.style.transform = `
        translate(-50%, -50%)
        translate(${x - centerX}px, ${y - centerY}px)
        rotate(${rotationDegrees}deg)
      `;
    }

    function animate(now) {
      const elapsedSeconds = (now - startTime) / 1000;

      const blueRate = blueLaps / blueMinutes;

      const orangeLaps = Number(lapsInput.value);
      const orangeMinutes = Number(minutesInput.value);
      const orangeRate = orangeLaps / orangeMinutes;

      positionCar(blueCar, blueRate, 0, elapsedSeconds);

      /*
        The orange car starts based on the user's selected speed.
        With the default choice, it is not literally "off"; it moves once
        the page starts. If you want it completely at rest until the user
        changes a value, set orangeRate to 0 initially and flip a flag on change.
      */
      positionCar(orangeCar, orangeRate, 34, elapsedSeconds);

      requestAnimationFrame(animate);
    }

    /*************************************************************
     * Initialize
     *************************************************************/
    populateInputs();
    updateOrangeDisplay();
    updateProgress();
    generateNewProblem();
    requestAnimationFrame(animate);
  </script>
</body>
