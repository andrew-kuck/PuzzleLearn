create an activity using html, css, and javascript. create an equation a^w ? b^x ? c^y ? d^z = e. a, b, c, and d should be integers between 1 and 10 (inclusive). each question mark respresents an operation from among addition (+), subtraction (-), multiplication (×), and division (÷). w, x, y, and z can be either 1 or 2. e must be an integer between 1 and u, where u is determined by a javascript constant and is initially set to 500. the user is shown the equation without any exponents and blank boxes where the operations would be. clicking on a blank box toggles between the four operations. clicking on a, b, c, or d toggles between showing the exponent 2 and showing no exponent (equivalent to exponent of 1). when the user clicks submit, the left side of the equation is evaluated. if the left side evaluates to e, then the user is correct, even if their solution is different from the original hidden equation. if the user is correct a green checkmark appears over the screen and the user is given a new question. if the user is incorrect, then a red x appears over the screen and the same question remains. with each correct answer a progress bar is advanced, such that it is full after 5 correct answers, at which point the function activityComplete() is called.

have the minus sign appear more in the vertical center of the box

  <script>
    /*
      Maximum possible target value.
      Change this constant if you want a larger or smaller target range.
    */
    const U = 500;

    const QUESTIONS_TO_COMPLETE = 5;
    const OPERATIONS = ["+", "-", "×", "÷"];
    const EPSILON = 1e-9;

    let currentQuestion = null;
    let userOperations = [null, null, null];
    let userExponents = [1, 1, 1, 1];
    let correctCount = 0;
    let completed = false;

    const termElements = document.querySelectorAll(".term");
    const operatorElements = document.querySelectorAll(".operator-box");
    const targetValueElement = document.getElementById("targetValue");
    const submitButton = document.getElementById("submitButton");
    const messageElement = document.getElementById("message");
    const overlaySymbol = document.getElementById("overlaySymbol");
    const progressFill = document.getElementById("progressFill");
    const progressText = document.getElementById("progressText");
    const completeMessage = document.getElementById("completeMessage");

    function randomInt(min, max) {
      return Math.floor(Math.random() * (max - min + 1)) + min;
    }

    function randomChoice(array) {
      return array[Math.floor(Math.random() * array.length)];
    }

    function powerValue(base, exponent) {
      return exponent === 2 ? base * base : base;
    }

    /*
      Evaluates expressions using normal arithmetic precedence:
      exponents first, then multiplication/division, then addition/subtraction.
    */
    function evaluateExpression(values, operations) {
      const nums = [values[0]];
      const ops = [];

      for (let i = 0; i < operations.length; i++) {
        const op = operations[i];
        const nextValue = values[i + 1];

        if (op === "×") {
          nums[nums.length - 1] *= nextValue;
        } else if (op === "÷") {
          nums[nums.length - 1] /= nextValue;
        } else {
          ops.push(op);
          nums.push(nextValue);
        }
      }

      let result = nums[0];

      for (let i = 0; i < ops.length; i++) {
        if (ops[i] === "+") {
          result += nums[i + 1];
        } else if (ops[i] === "-") {
          result -= nums[i + 1];
        }
      }

      return result;
    }

    function isInteger(value) {
      return Math.abs(value - Math.round(value)) < EPSILON;
    }

    /*
      Creates a hidden equation:
      a^w ? b^x ? c^y ? d^z = e

      The generated e is guaranteed to be an integer between 1 and U.
    */
    function generateQuestion() {
      while (true) {
        const bases = [
          randomInt(1, 10),
          randomInt(1, 10),
          randomInt(1, 10),
          randomInt(1, 10)
        ];

        const exponents = [
          randomChoice([1, 2]),
          randomChoice([1, 2]),
          randomChoice([1, 2]),
          randomChoice([1, 2])
        ];

        const operations = [
          randomChoice(OPERATIONS),
          randomChoice(OPERATIONS),
          randomChoice(OPERATIONS)
        ];

        const values = bases.map((base, i) => powerValue(base, exponents[i]));
        const result = evaluateExpression(values, operations);

        if (isInteger(result)) {
          const e = Math.round(result);

          if (e >= 1 && e <= U) {
            return {
              bases,
              hiddenExponents: exponents,
              hiddenOperations: operations,
              target: e
            };
          }
        }
      }
    }

    function renderEquation() {
      termElements.forEach((term, index) => {
        const base = currentQuestion.bases[index];
        const exponent = userExponents[index];

        term.innerHTML = exponent === 2
          ? `${base}<sup class="exponent">2</sup>`
          : `${base}`;
      });

      operatorElements.forEach((box, index) => {
        box.textContent = userOperations[index] || "";
      });

      targetValueElement.textContent = currentQuestion.target;
    }

    function newQuestion() {
      currentQuestion = generateQuestion();

      userOperations = [null, null, null];
      userExponents = [1, 1, 1, 1];

      messageElement.textContent = "";
      renderEquation();
    }

    function updateProgress() {
      const percentage = (correctCount / QUESTIONS_TO_COMPLETE) * 100;
      progressFill.style.width = `${percentage}%`;
      progressText.textContent = `${correctCount} / ${QUESTIONS_TO_COMPLETE}`;
    }

    function showOverlay(type) {
      overlaySymbol.className = "overlay-symbol";

      if (type === "correct") {
        overlaySymbol.textContent = "✓";
        overlaySymbol.classList.add("correct", "show");
      } else {
        overlaySymbol.textContent = "×";
        overlaySymbol.classList.add("incorrect", "show");
      }

      setTimeout(() => {
        overlaySymbol.classList.remove("show");
      }, 700);
    }

    function submitAnswer() {
      if (completed) return;

      if (userOperations.some(op => op === null)) {
        messageElement.textContent = "Choose all three operations before submitting.";
        return;
      }

      const userValues = currentQuestion.bases.map((base, index) => {
        return powerValue(base, userExponents[index]);
      });

      const userResult = evaluateExpression(userValues, userOperations);
      const target = currentQuestion.target;

      if (Math.abs(userResult - target) < EPSILON) {
        correctCount++;
        updateProgress();

        showOverlay("correct");
        messageElement.textContent = "Correct! Nice equation wrangling.";

        if (correctCount >= QUESTIONS_TO_COMPLETE) {
          completed = true;
          completeMessage.classList.add("show");
          submitButton.disabled = true;

          setTimeout(() => {
            activityComplete();
          }, 750);
        } else {
          setTimeout(() => {
            newQuestion();
          }, 800);
        }
      } else {
        showOverlay("incorrect");
        messageElement.textContent = `Not quite. Your left side equals ${formatNumber(userResult)}. Try again.`;
      }
    }

    function formatNumber(value) {
      if (isInteger(value)) {
        return String(Math.round(value));
      }

      return Number(value.toFixed(4)).toString();
    }

    /*
      This function is called when the progress bar is full.
      Replace the body with your LMS, parent-page, or completion logic.
    */
    function activityComplete() {
      console.log("Activity complete!");
    }

    termElements.forEach(term => {
      term.addEventListener("click", () => {
        if (completed) return;

        const index = Number(term.dataset.index);
        userExponents[index] = userExponents[index] === 1 ? 2 : 1;
        renderEquation();
      });
    });

    operatorElements.forEach(box => {
      box.addEventListener("click", () => {
        if (completed) return;

        const index = Number(box.dataset.index);
        const currentOperation = userOperations[index];

        if (currentOperation === null) {
          userOperations[index] = OPERATIONS[0];
        } else {
          const currentIndex = OPERATIONS.indexOf(currentOperation);
          const nextIndex = (currentIndex + 1) % OPERATIONS.length;
          userOperations[index] = OPERATIONS[nextIndex];
        }

        renderEquation();
      });
    });

    submitButton.addEventListener("click", submitAnswer);

    updateProgress();
    newQuestion();
  </script>
</body>
</html>
