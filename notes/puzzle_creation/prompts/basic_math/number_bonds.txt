create an activity using html, css, and javascript. show a circle on the screen with two lines extending below to the left and right. at the end of these "branches" are the "leaf nodes", which are two more circles of the same size. the leaf nodes each correspond to an integer between -10 and 10. the parent node or root node is the sum of the two leaf nodes. show the value for two of the 3 nodes and require the user to fill in the other node. upon loading the page or a new question, the input node should automatically be active, so that the user can type a number and hit enter without using the mouse. if the input is correct, a green checkmark appears on the screen, a new problem is presented, and a progress bar is advanced one unit. if the input is not correct, a red x appears over the screen, a new problem is presented, and the progress bar goes back one unit. the progress bar extends from -10 to 10 (without labels or tick marks except for a gray vertical line in the middle indicating 0). 0 displays as nothing, negative numbers display as red fill extending to the left of 0, and positive numbers display as green fill extending to the right of 0. progress cannot go below -10. when progress reaches 10 a function activityComplete() is called.

have the progress bar extend from -10 (10 incorrect) to 10 (10 correct), not from -5 to 5

keep everything the same, but now have the two leaf nodes multiply to equal the parent node

now keep everything the same, but change it so that one of the leaf nodes is an integer between 1 and 9 and the other leaf node is one of the following fractions: 1/2, 1/3, 1/4, 1/5, 1/6, 1/7, 1/8, 1/9. the leaf nodes are still being multiplied, and the user may still be asked to input into any of the three nodes

  <script>
    const rootNode = document.getElementById("rootNode");
    const leftNode = document.getElementById("leftNode");
    const rightNode = document.getElementById("rightNode");

    const fillPositive = document.getElementById("fillPositive");
    const fillNegative = document.getElementById("fillNegative");
    const feedback = document.getElementById("feedback");
    const completeMessage = document.getElementById("completeMessage");

    let currentAnswer = null;
    let progress = 0;
    let completed = false;

    function randomInteger(min, max) {
      return Math.floor(Math.random() * (max - min + 1)) + min;
    }

    function clearNodes() {
      rootNode.innerHTML = "";
      leftNode.innerHTML = "";
      rightNode.innerHTML = "";
    }

    function makeInput(answer) {
      const input = document.createElement("input");
      input.type = "number";
      input.inputMode = "numeric";
      input.setAttribute("aria-label", "Missing value");
      input.autocomplete = "off";

      input.addEventListener("keydown", function(event) {
        if (event.key === "Enter") {
          checkAnswer(input.value);
        }
      });

      currentAnswer = answer;
      return input;
    }

    function newProblem() {
      if (completed) return;

      clearNodes();

      const left = randomInteger(-10, 10);
      const right = randomInteger(-10, 10);
      const root = left + right;

      const missing = randomInteger(0, 2);
      let input;

      if (missing === 0) {
        input = makeInput(root);
        rootNode.appendChild(input);
        leftNode.textContent = left;
        rightNode.textContent = right;
      } else if (missing === 1) {
        rootNode.textContent = root;
        input = makeInput(left);
        leftNode.appendChild(input);
        rightNode.textContent = right;
      } else {
        rootNode.textContent = root;
        leftNode.textContent = left;
        input = makeInput(right);
        rightNode.appendChild(input);
      }

      // Automatically activate the input so the user can type immediately.
      setTimeout(() => {
        input.focus();
        input.select();
      }, 0);
    }

    function checkAnswer(value) {
      if (completed) return;

      const userAnswer = Number(value);

      if (value.trim() === "" || !Number.isInteger(userAnswer)) {
        showFeedback(false);
        updateProgress(-1);
        scheduleNextProblem();
        return;
      }

      if (userAnswer === currentAnswer) {
        showFeedback(true);
        updateProgress(1);

        if (progress >= 10) {
          completed = true;
          activityComplete();
          return;
        }

        scheduleNextProblem();
      } else {
        showFeedback(false);
        updateProgress(-1);
        scheduleNextProblem();
      }
    }

    function updateProgress(change) {
      progress += change;

      if (progress < -10) {
        progress = -10;
      }

      if (progress > 10) {
        progress = 10;
      }

      renderProgress();
    }

    function renderProgress() {
      // Each side of the progress bar represents 10 units.
      // Therefore each unit equals 10% of half the bar.
      const magnitude = Math.abs(progress) * 10;

      if (progress > 0) {
        fillPositive.style.width = magnitude + "%";
        fillNegative.style.width = "0";
      } else if (progress < 0) {
        fillNegative.style.width = magnitude + "%";
        fillPositive.style.width = "0";
      } else {
        fillPositive.style.width = "0";
        fillNegative.style.width = "0";
      }
    }

    function showFeedback(isCorrect) {
      feedback.textContent = isCorrect ? "✓" : "×";
      feedback.className = "feedback " + (isCorrect ? "correct" : "incorrect");

      setTimeout(() => {
        feedback.className = "feedback";
        feedback.textContent = "";
      }, 450);
    }

    function scheduleNextProblem() {
      setTimeout(() => {
        newProblem();
      }, 500);
    }

    // This function is called when progress reaches 10.
    // Replace the contents with whatever should happen at completion.
    function activityComplete() {
      completeMessage.style.display = "block";
      clearNodes();
      rootNode.textContent = "✓";
      leftNode.textContent = "✓";
      rightNode.textContent = "✓";
      console.log("Activity complete!");
    }

    renderProgress();
    newProblem();
  </script>
