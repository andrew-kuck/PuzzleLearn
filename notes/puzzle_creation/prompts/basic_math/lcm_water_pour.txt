create an activity using html, css, and javascript. on the left and right side of screen are rectangles that look like partially filled cups. the quantity 13 would fill the cup. each cup is filled with a random quantity between 2 and 12. closer to the middle are two larger cups. each is 3 times wider and 4 times taller than the side cups. thus, the quantity of these cups when filled is 13 * 3 * 4 = 156. below each side cup is an equation a * b = c. a is filled in with the quantity of that side cup. b is for user input: once the user inputs b that input is disabled from further input. c is initially blank. that side cup then "pours" it's quantity into the middle cup it is closest to; it does this b times, so that the quantity in that middle cup is c. at this point the value for c becomes visible. if the value for c on the the other side is also visible, and if these two values are equal, and if c is the least common multiple of the two initial quantities in the side cups (the two values for a) then a green checkmark appears over the screen and a new question is presented. there is a reset button to empty the middle cups of their fill and return the equations to their starting values. there is a progress bar at the top that fills with each correct answer and is full at 10 correct answers, at which point the function activityComplete() is called. make sure that even when the screen is wide, the cups do not get so tall that they require a vertical scroll.

don't have up and down buttons on the inputs

Below is a complete single-file HTML activity. You can copy it into an `index.html` file and open it in a browser.

It includes:

- Two small side cups with random starting amounts from 2–12.
- Two larger middle cups with capacity 156.
- Equations of the form `a × b = c`.
- User input for `b`.
- Animated pouring into the nearest middle cup.
- A reset button.
- A progress bar that completes after 10 correct answers.
- A call to `activityComplete()` when finished.
- Responsive sizing so the cups should not force vertical scrolling on wide screens.

<script>
  const CAPACITY_SMALL = 13;
  const CAPACITY_BIG = 156;
  const GOAL_COUNT = 10;

  let leftAmount = 0;
  let rightAmount = 0;
  let leftCValue = null;
  let rightCValue = null;
  let correctCount = 0;
  let locked = false;

  const els = {
    leftSideFill: document.getElementById("leftSideFill"),
    rightSideFill: document.getElementById("rightSideFill"),
    leftMiddleFill: document.getElementById("leftMiddleFill"),
    rightMiddleFill: document.getElementById("rightMiddleFill"),

    leftAmountLabel: document.getElementById("leftAmountLabel"),
    rightAmountLabel: document.getElementById("rightAmountLabel"),
    leftMiddleLabel: document.getElementById("leftMiddleLabel"),
    rightMiddleLabel: document.getElementById("rightMiddleLabel"),

    leftA: document.getElementById("leftA"),
    rightA: document.getElementById("rightA"),
    leftB: document.getElementById("leftB"),
    rightB: document.getElementById("rightB"),
    leftC: document.getElementById("leftC"),
    rightC: document.getElementById("rightC"),

    resetBtn: document.getElementById("resetBtn"),
    message: document.getElementById("message"),
    progressFill: document.getElementById("progressFill"),
    score: document.getElementById("score"),

    checkOverlay: document.getElementById("checkOverlay"),
    leftPour: document.getElementById("leftPour"),
    rightPour: document.getElementById("rightPour"),

    leftStation: document.getElementById("leftStation"),
    rightStation: document.getElementById("rightStation"),
    leftMiddleStation: document.getElementById("leftMiddleStation"),
    rightMiddleStation: document.getElementById("rightMiddleStation")
  };

  function activityComplete() {
    console.log("Activity complete!");
    els.message.textContent = "Activity complete! Excellent work.";
  }

  function randomInt(min, max) {
    return Math.floor(Math.random() * (max - min + 1)) + min;
  }

  function gcd(a, b) {
    while (b !== 0) {
      const temp = b;
      b = a % b;
      a = temp;
    }
    return a;
  }

  function lcm(a, b) {
    return Math.abs(a * b) / gcd(a, b);
  }

  function setFill(fillEl, amount, capacity) {
    const pct = Math.max(0, Math.min(100, (amount / capacity) * 100));
    fillEl.style.height = pct + "%";
  }

  function setProgress() {
    els.score.textContent = correctCount;
    els.progressFill.style.width = `${(correctCount / GOAL_COUNT) * 100}%`;
  }

  function clearMiddleCups() {
    leftCValue = null;
    rightCValue = null;

    setFill(els.leftMiddleFill, 0, CAPACITY_BIG);
    setFill(els.rightMiddleFill, 0, CAPACITY_BIG);

    els.leftMiddleLabel.textContent = "0";
    els.rightMiddleLabel.textContent = "0";

    els.leftB.value = "";
    els.rightB.value = "";
    els.leftB.disabled = false;
    els.rightB.disabled = false;

    els.leftC.textContent = "?";
    els.rightC.textContent = "?";
    els.leftC.classList.add("c-hidden");
    els.rightC.classList.add("c-hidden");

    locked = false;
  }

  function newQuestion() {
    leftAmount = randomInt(2, 12);
    rightAmount = randomInt(2, 12);

    while (rightAmount === leftAmount) {
      rightAmount = randomInt(2, 12);
    }

    els.leftA.textContent = leftAmount;
    els.rightA.textContent = rightAmount;

    els.leftAmountLabel.textContent = leftAmount;
    els.rightAmountLabel.textContent = rightAmount;

    setFill(els.leftSideFill, leftAmount, CAPACITY_SMALL);
    setFill(els.rightSideFill, rightAmount, CAPACITY_SMALL);

    els.leftB.max = Math.floor(CAPACITY_BIG / leftAmount);
    els.rightB.max = Math.floor(CAPACITY_BIG / rightAmount);

    clearMiddleCups();

    const answer = lcm(leftAmount, rightAmount);
    els.message.textContent =
      `Find multipliers so both middle cups hold the same amount: the LCM of ${leftAmount} and ${rightAmount}.`;
  }

  function resetCurrentQuestion() {
    clearMiddleCups();
    els.message.textContent = "Reset complete. Try the multipliers again.";
  }

  function positionPourStream(side) {
    const stream = side === "left" ? els.leftPour : els.rightPour;
    const source = side === "left" ? els.leftStation : els.rightStation;
    const target = side === "left" ? els.leftMiddleStation : els.rightMiddleStation;

    const sourceRect = source.getBoundingClientRect();
    const targetRect = target.getBoundingClientRect();
    const activityRect = document.getElementById("activity").getBoundingClientRect();

    const startX =
      side === "left"
        ? sourceRect.right - activityRect.left - 8
        : sourceRect.left - activityRect.left + 8;

    const endX =
      side === "left"
        ? targetRect.left - activityRect.left + targetRect.width * 0.35
        : targetRect.right - activityRect.left - targetRect.width * 0.35;

    const x = (startX + endX) / 2;

    const y = Math.min(sourceRect.top, targetRect.top) - activityRect.top + 20;

    stream.style.left = `${x}px`;
    stream.style.top = `${y}px`;
  }

  function animatePour(side, repetitions) {
    return new Promise(resolve => {
      const stream = side === "left" ? els.leftPour : els.rightPour;
      let count = 0;

      positionPourStream(side);

      function onePour() {
        stream.classList.remove("active");
        void stream.offsetWidth;
        stream.classList.add("active");

        count++;

        if (count < repetitions) {
          setTimeout(onePour, 210);
        } else {
          setTimeout(() => {
            stream.classList.remove("active");
            resolve();
          }, 700);
        }
      }

      onePour();
    });
  }

  async function submitSide(side) {
    if (locked) return;

    const isLeft = side === "left";
    const input = isLeft ? els.leftB : els.rightB;
    const amount = isLeft ? leftAmount : rightAmount;
    const cEl = isLeft ? els.leftC : els.rightC;
    const middleFill = isLeft ? els.leftMiddleFill : els.rightMiddleFill;
    const middleLabel = isLeft ? els.leftMiddleLabel : els.rightMiddleLabel;

    if (input.disabled) return;

    const b = Number(input.value);

    if (!Number.isInteger(b) || b <= 0) {
      els.message.textContent = "Please enter a positive whole number.";
      input.value = "";
      input.focus();
      return;
    }

    const c = amount * b;

    if (c > CAPACITY_BIG) {
      els.message.textContent =
        `That would overfill the middle cup. Try a smaller multiplier.`;
      input.value = "";
      input.focus();
      return;
    }

    input.disabled = true;
    els.message.textContent = `Pouring ${amount}, ${b} time${b === 1 ? "" : "s"}...`;

    await animatePour(side, Math.min(b, 12));

    setFill(middleFill, c, CAPACITY_BIG);
    middleLabel.textContent = c;

    cEl.textContent = c;
    cEl.classList.remove("c-hidden");

    if (isLeft) {
      leftCValue = c;
    } else {
      rightCValue = c;
    }

    checkAnswer();
  }

  function checkAnswer() {
    if (leftCValue === null || rightCValue === null) {
      els.message.textContent = "Now complete the other side.";
      return;
    }

    const targetLCM = lcm(leftAmount, rightAmount);

    if (
      leftCValue === rightCValue &&
      leftCValue === targetLCM
    ) {
      correctCount++;
      setProgress();
      locked = true;

      els.checkOverlay.classList.add("show");
      els.message.textContent = `Correct! ${targetLCM} is the least common multiple.`;

      setTimeout(() => {
        els.checkOverlay.classList.remove("show");

        if (correctCount >= GOAL_COUNT) {
          activityComplete();
        } else {
          newQuestion();
        }
      }, 1200);
    } else if (leftCValue === rightCValue) {
      els.message.textContent =
        `The amounts match, but ${leftCValue} is not the least common multiple. Press Reset and try smaller matching values.`;
    } else {
      els.message.textContent =
        "The middle cups do not match. Press Reset and try again.";
    }
  }

  els.leftB.addEventListener("change", () => submitSide("left"));
  els.rightB.addEventListener("change", () => submitSide("right"));

  els.leftB.addEventListener("keydown", event => {
    if (event.key === "Enter") {
      event.preventDefault();
      submitSide("left");
    }
  });

  els.rightB.addEventListener("keydown", event => {
    if (event.key === "Enter") {
      event.preventDefault();
      submitSide("right");
    }
  });

  els.resetBtn.addEventListener("click", resetCurrentQuestion);

  window.addEventListener("resize", () => {
    positionPourStream("left");
    positionPourStream("right");
  });

  setProgress();
  newQuestion();
</script>
</body>
</html>
