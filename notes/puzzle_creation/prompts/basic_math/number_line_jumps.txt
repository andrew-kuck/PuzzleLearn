create an activity using html, css, and javascript.
1. Main activity layout
- A centered container.
- A heading that says `Reach 0`.
- A horizontal progress bar near the top.
- A number line from `-15` to `15`.
- The number line should have:
  - A thick black horizontal line.
  - Arrowheads on both ends using CSS pseudo-elements.
  - Tick marks for every integer from -15 through 15.
  - Labels below each tick.
  - A blue circular dot that moves along the line.
- Below the number line, show seven circular clickable choice buttons.
2. Gameplay logic
- Use JavaScript constants:
  - `MIN = -15`
  - `MAX = 15`
- Maintain:
  - `currentPosition`
  - `progress`
- At the start of each round, randomly place the dot at a nonzero integer between -15 and 15.
- Generate seven random choice buttons.
- Each choice button should display a random integer from -10 to 10, including 0.
- Positive numbers should display with a plus sign, e.g. `+4`.
- When the user clicks a choice:
  - Subtract 2 from the progress bar, with a temporary red flash.
  - Add the clicked value to the current dot position.
  - Move the dot smoothly to the new position.
  - If the new position is greater than 15, animate the dot just past the right edge, fade it out, then reset it to 15 and fade it back in.
  - If the new position is less than -15, animate the dot just past the left edge, fade it out, then reset it to -15 and fade it back in.
  - Replace the clicked choice button with a new random choice.
- If the dot reaches exactly 0:
  - Increase progress by 10, capped at 100.
  - Show a large green checkmark overlay briefly.
  - Start a new round with a new nonzero starting position and new choices.
- If progress reaches 100:
  - Call `activityComplete();`
3. Styling
- Use CSS directly in a `<style>` block.
- Make the layout responsive and centered.
- Use `box-sizing: border-box`.
- Use Arial or a similar sans-serif font.
- Make the progress bar green by default and red briefly when progress is reduced.
- Make choice buttons circular, white, bordered, bold, and hover/active animated.
- Make the success overlay full-screen, semi-transparent white background, with a large green checkmark.

keep everything the same, but make sure the arrows at the ends of the line appear just beyond the final tick marks at -15 and 15

Paste this into a single `.html` file and open it in a browser.

<body>
  <main class="activity-container">
    <h1>Reach 0</h1>

    <div class="progress-wrapper" aria-label="Progress">
      <div class="progress-fill" id="progressFill"></div>
    </div>

    <section class="number-line-area" aria-label="Number line from negative fifteen to fifteen">
      <div class="number-axis" id="numberAxis">
        <div class="axis-line"></div>
        <div class="dot" id="dot"></div>
      </div>
    </section>

    <section class="choices" id="choices" aria-label="Choice buttons"></section>
  </main>

  <div class="success-overlay" id="successOverlay" aria-hidden="true">
    <div class="checkmark">✓</div>
  </div>

  <script>
    const MIN = -15;
    const MAX = 15;

    let currentPosition = 0;
    let progress = 0;
    let isAnimating = false;

    const numberAxis = document.getElementById("numberAxis");
    const dot = document.getElementById("dot");
    const choices = document.getElementById("choices");
    const progressFill = document.getElementById("progressFill");
    const successOverlay = document.getElementById("successOverlay");

    // Default fallback in case this activity is not inside a larger platform.
    window.activityComplete = window.activityComplete || function () {
      console.log("Activity complete!");
    };

    function randomInteger(min, max) {
      return Math.floor(Math.random() * (max - min + 1)) + min;
    }

    function randomStartingPosition() {
      let value = 0;

      while (value === 0) {
        value = randomInteger(MIN, MAX);
      }

      return value;
    }

    function randomChoiceValue() {
      return randomInteger(-10, 10);
    }

    function formatNumber(value) {
      return value > 0 ? `+${value}` : `${value}`;
    }

    function positionToPercent(position) {
      return ((position - MIN) / (MAX - MIN)) * 100;
    }

    function delay(ms) {
      return new Promise(resolve => setTimeout(resolve, ms));
    }

    function createTicks() {
      for (let value = MIN; value <= MAX; value++) {
        const tick = document.createElement("div");
        tick.className = "tick";
        tick.style.left = `${positionToPercent(value)}%`;

        const mark = document.createElement("div");
        mark.className = "tick-mark";

        const label = document.createElement("div");
        label.className = "tick-label";
        label.textContent = value;

        tick.appendChild(mark);
        tick.appendChild(label);
        numberAxis.appendChild(tick);
      }
    }

    function setDotPosition(position, instant = false) {
      if (instant) {
        dot.style.transition = "none";
      } else {
        dot.style.transition = "left 0.5s ease, opacity 0.25s ease";
      }

      dot.style.left = `${positionToPercent(position)}%`;

      if (instant) {
        dot.offsetHeight; // force reflow
        dot.style.transition = "left 0.5s ease, opacity 0.25s ease";
      }
    }

    function setDotPercent(percent, instant = false) {
      if (instant) {
        dot.style.transition = "none";
      } else {
        dot.style.transition = "left 0.5s ease, opacity 0.25s ease";
      }

      dot.style.left = `${percent}%`;

      if (instant) {
        dot.offsetHeight; // force reflow
        dot.style.transition = "left 0.5s ease, opacity 0.25s ease";
      }
    }

    function updateProgress(newProgress, flashRed = false) {
      progress = Math.max(0, Math.min(100, newProgress));
      progressFill.style.width = `${progress}%`;

      if (flashRed) {
        progressFill.classList.add("penalty");

        setTimeout(() => {
          progressFill.classList.remove("penalty");
        }, 250);
      }
    }

    function createChoiceButton(value) {
      const button = document.createElement("button");
      button.className = "choice-button";
      button.type = "button";
      button.textContent = formatNumber(value);
      button.dataset.value = value;

      button.addEventListener("click", () => handleChoiceClick(button));

      return button;
    }

    function generateChoices() {
      choices.innerHTML = "";

      for (let i = 0; i < 7; i++) {
        choices.appendChild(createChoiceButton(randomChoiceValue()));
      }
    }

    function replaceChoiceButton(button) {
      const newButton = createChoiceButton(randomChoiceValue());
      choices.replaceChild(newButton, button);
    }

    function setButtonsDisabled(disabled) {
      document.querySelectorAll(".choice-button").forEach(button => {
        button.disabled = disabled;
      });
    }

    async function animateOverflowRight() {
      setDotPercent(106);
      await delay(500);

      dot.style.opacity = "0";
      await delay(260);

      currentPosition = MAX;
      setDotPosition(MAX, true);

      dot.style.opacity = "1";
      await delay(260);
    }

    async function animateOverflowLeft() {
      setDotPercent(-6);
      await delay(500);

      dot.style.opacity = "0";
      await delay(260);

      currentPosition = MIN;
      setDotPosition(MIN, true);

      dot.style.opacity = "1";
      await delay(260);
    }

    async function showSuccess() {
      successOverlay.classList.add("show");
      await delay(700);
      successOverlay.classList.remove("show");
      await delay(250);
    }

    async function startNewRound() {
      currentPosition = randomStartingPosition();
      setDotPosition(currentPosition, true);
      generateChoices();
    }

    async function handleChoiceClick(button) {
      if (isAnimating) return;

      isAnimating = true;
      setButtonsDisabled(true);

      const value = Number(button.dataset.value);

      updateProgress(progress - 2, true);

      const newPosition = currentPosition + value;

      replaceChoiceButton(button);

      if (newPosition > MAX) {
        await animateOverflowRight();
      } else if (newPosition < MIN) {
        await animateOverflowLeft();
      } else {
        currentPosition = newPosition;
        setDotPosition(currentPosition);
        await delay(520);
      }

      if (currentPosition === 0) {
        updateProgress(progress + 10);

        await showSuccess();

        if (progress >= 100) {
          activityComplete();
          return;
        }

        await startNewRound();
      }

      setButtonsDisabled(false);
      isAnimating = false;
    }

    createTicks();
    startNewRound();
  </script>
</body>
</html>
