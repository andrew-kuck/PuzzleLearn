create an activity using html, css, and javascript. show some randomly chosen function on a coordinate grid on the screen, with -10 <= x <= 10 and -10 <= y <=10. The user is asked to find the slope at x = n, where n is an integer between -9 and 9 (inclusive). n must be a point on the graph where f is differentiable. Ask "What is f'(n): the slope of the function f at x=n?" If the user clicks on the curve of f, they are able to pull a horizontal red dotted line away from the point they clicked on. This red dotted line is labeled Δx. As this is done, a vertical blue dotted line automatically extends from the end of the red dotted line and is labeled Δy. below the coordinate grid is a one line calculator with a division symbol (÷), then an AC symbol, then an input, and then an equal symbol (=). the input accepts only input from the calculator buttons or numerical input (it will accept the . character in order to accept decimal numbers). if the user clicks on the numerical label for Δx or the numerical label for Δy, then that number is added to the calculator input. the initial coordinate grid should have each integer on the x an y axes labeled, with associated grid lines in a lighter gray color. the user can zoom in on the graph. when they do so the number of axis labels and grid lines stay the same, but the labels change accordingly to indicate the zoom. when zoomed, the user can click and drag on the screen to move to different parts of the coordinate plane, as long as they don't go outside of the initially established window. the user can zoom as far in as they like. there is a button to reset the zoom to the initial orientation of the grid. when the user submits an answer to the question, the difference between their answer and the actual value f'(n) is indicated to them and then added to a running error total. a new question is then presented, and a progress bar is advanced, such that after answering 10 questions the progress bar is full, at which point a function activityComplete() is called.

the calculator does not need digit buttons; have all the buttons appear on one line: division button, AC button, input field, then equals button, then a space for the result of the calculation to show up. there should be a separate input field below that for the actual slope f'(n). the red and blue dotted lines should disappear when the user clicks anywhere else on the graph without dragging.

      function choice(arr) {
        return arr[Math.floor(Math.random() * arr.length)];

      function fmt(value, digits = 5) {
        if (!Number.isFinite(value)) return "undefined";
        if (Math.abs(value) < 1e-12) value = 0;

        const abs = Math.abs(value);
        if (abs !== 0 && (abs >= 100000 || abs < 0.0001)) {
          return value.toExponential(3);
        }

        return Number(value.toFixed(digits)).toString();
      }

      function fmtTick(value) {
        if (Math.abs(value) < 1e-12) value = 0;

        const span = view.xmax - view.xmin;
        if (span < 0.001) return value.toExponential(2);
        if (span < 0.05) return Number(value.toFixed(5)).toString();

        if (span < 2) return Number(value.toFixed(3)).toString();
        if (span < 10) return Number(value.toFixed(2)).toString();
        return Number(value.toFixed(1)).toString();

      function makeRandomFunction() {
        const type = randInt(0, 4);

        if (type === 0) {

            df: x => m

            df: x => 2 * a * (x - h)

            df: x => 3 * a * Math.pow(x - h, 2) + b

      function clampViewToInitial() {
        const spanX = view.xmax - view.xmin;
        const spanY = view.ymax - view.ymin;

            view.xmin = INITIAL.xmin;

            view.xmin -= view.xmax - INITIAL.xmax;
            view.xmax = INITIAL.xmax;

            continue;

          if (!started) {
            ctx.moveTo(p.x, p.y);
            started = true;
          } else {
            ctx.lineTo(p.x, p.y);
          }
        }

        ctx.stroke();
        ctx.restore();

        dyLabel.style.top = `${clamp(dyMid.y, 14, H - 14)}px`;

        dxLabel.hidden = false;
        dyLabel.hidden = false;
      }

      function updateZoomReadout() {
        zoomReadout.textContent =
          `Window: x ∈ [${fmt(view.xmin, 4)}, ${fmt(view.xmax, 4)}], ` +
          `y ∈ [${fmt(view.ymin, 4)}, ${fmt(view.ymax, 4)}]`;
      }

      function render() {
        drawGrid();
        drawFunction();
        drawTargetPoint();
        drawDeltaSegments();
        updateDeltaLabels();
        updateZoomReadout();
      }

      function resizeCanvas() {
        const rect = graphWrap.getBoundingClientRect();
        W = rect.width;
        H = rect.height;

        const dpr = window.devicePixelRatio || 1;
        canvas.width = Math.round(W * dpr);
        canvas.height = Math.round(H * dpr);

        ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
        render();
      }

      function nearCurve(point) {
        if (!currentFunction) return false;

        const world = screenToWorld(point.x, point.y);
        const y = currentFunction.f(world.x);

        if (!Number.isFinite(y)) return false;

        const curvePoint = worldToScreen(world.x, y);
        return Math.abs(point.y - curvePoint.y) <= 12;
      }

      canvas.addEventListener("pointerdown", e => {
        if (completed) return;

        const p = getPointerPos(e);
        canvas.setPointerCapture(e.pointerId);

        if (nearCurve(p)) {
          const world = screenToWorld(p.x, p.y);
          const y = currentFunction.f(world.x);

          delta = {
            x0: world.x,
            y0: y,
            x1: world.x
          };

          mode = "delta";
          render();
        } else {
          mode = "pan";
          panStart = {
            sx: p.x,
            sy: p.y,
            view: { ...view }
          };
        }
      });

      canvas.addEventListener("pointermove", e => {
        if (completed || !mode) return;

        const p = getPointerPos(e);

        if (mode === "delta" && delta) {
          const clampedScreenX = clamp(p.x, 0, W);
          const world = screenToWorld(clampedScreenX, p.y);
          delta.x1 = clamp(world.x, INITIAL.xmin, INITIAL.xmax);
          render();
        }

        if (mode === "pan" && panStart) {
          const dxPixels = p.x - panStart.sx;
          const dyPixels = p.y - panStart.sy;

          const spanX = panStart.view.xmax - panStart.view.xmin;
          const spanY = panStart.view.ymax - panStart.view.ymin;

          const dxWorld = -(dxPixels / W) * spanX;
          const dyWorld = (dyPixels / H) * spanY;

          view.xmin = panStart.view.xmin + dxWorld;
          view.xmax = panStart.view.xmax + dxWorld;
          view.ymin = panStart.view.ymin + dyWorld;
          view.ymax = panStart.view.ymax + dyWorld;

          clampViewToInitial();
          render();
        }
      });

      canvas.addEventListener("pointerup", e => {
        mode = null;
        panStart = null;
        try {
          canvas.releasePointerCapture(e.pointerId);
        } catch {}
      });

      canvas.addEventListener("pointercancel", () => {
        mode = null;
        panStart = null;
      });

      canvas.addEventListener("wheel", e => {
        if (completed) return;

        e.preventDefault();

        const p = getPointerPos(e);
        const focus = screenToWorld(p.x, p.y);

        const oldSpanX = view.xmax - view.xmin;
        const oldSpanY = view.ymax - view.ymin;

        let factor = Math.exp(e.deltaY * 0.001);
        let newSpanX = oldSpanX * factor;
        let newSpanY = oldSpanY * factor;

        const maxSpan = 20;
        const minSpan = 1e-9;

        newSpanX = clamp(newSpanX, minSpan, maxSpan);
        newSpanY = clamp(newSpanY, minSpan, maxSpan);

        const rx = (focus.x - view.xmin) / oldSpanX;
        const ry = (focus.y - view.ymin) / oldSpanY;

        view.xmin = focus.x - rx * newSpanX;
        view.xmax = view.xmin + newSpanX;

        view.ymin = focus.y - ry * newSpanY;
        view.ymax = view.ymin + newSpanY;

        clampViewToInitial();
        render();
      }, { passive: false });

      resetZoomBtn.addEventListener("click", resetView);

      function sanitizeExpression(raw) {
        raw = raw
          .replace(/\//g, "÷")
          .replace(/[−–—]/g, "-");

        let beforeDivision = "";
        let afterDivision = "";
        let seenDivision = false;
        let current = "";

        function appendToCurrent(ch) {
          if (/\d/.test(ch)) {
            current += ch;
            return;
          }

          if (ch === "." && !current.includes(".")) {
            current += ch;
            return;
          }

          if (ch === "-" && current.length === 0) {
            current += ch;
          }
        }

        for (const ch of raw) {
          if (ch === "÷") {
            if (!seenDivision && current !== "" && current !== "-") {
              beforeDivision = current;
              current = "";
              seenDivision = true;
            }
            continue;
          }

          appendToCurrent(ch);
        }

        if (seenDivision) {
          afterDivision = current;
          return `${beforeDivision}÷${afterDivision}`;
        }

        return current;
      }

      function addToCalculator(value) {
        if (completed) return;
        answerInput.value = sanitizeExpression(answerInput.value + value);
        answerInput.focus();
      }

      document.querySelectorAll("[data-val]").forEach(btn => {
        btn.addEventListener("click", () => {
          addToCalculator(btn.dataset.val);
        });
      });

      divideBtn.addEventListener("click", () => {
        addToCalculator("÷");
      });

      clearBtn.addEventListener("click", () => {
        answerInput.value = "";
        answerInput.focus();
      });

      dxLabel.addEventListener("click", () => {
        addToCalculator(dxLabel.dataset.value || "");
      });

      dyLabel.addEventListener("click", () => {
        addToCalculator(dyLabel.dataset.value || "");
      });

      answerInput.addEventListener("keydown", e => {
        const allowedControlKeys = [
          "Backspace",
          "Delete",
          "ArrowLeft",
          "ArrowRight",
          "ArrowUp",
          "ArrowDown",
          "Tab",
          "Home",
          "End"
        ];

        if (e.key === "Enter") {
          e.preventDefault();
          submitAnswer();
          return;
        }

        if (allowedControlKeys.includes(e.key)) return;

        if (/[\d.\-/÷−]/.test(e.key)) return;

        e.preventDefault();
      });

      answerInput.addEventListener("input", () => {
        const cursorAtEnd = answerInput.selectionStart === answerInput.value.length;
        answerInput.value = sanitizeExpression(answerInput.value);
        if (cursorAtEnd) {
          answerInput.selectionStart = answerInput.selectionEnd = answerInput.value.length;
        }
      });

      function evaluateCalculatorExpression() {
        const expr = sanitizeExpression(answerInput.value.trim()).replace(/÷/g, "/");

        const numberPattern = "-?(?:\\d+(?:\\.\\d*)?|\\.\\d+)";
        const regex = new RegExp(`^(${numberPattern})(?:\\/(${numberPattern}))?$`);
        const match = expr.match(regex);

        if (!match) return null;

        const a = Number(match[1]);

        if (match[2] !== undefined) {
          const b = Number(match[2]);
          if (Math.abs(b) < 1e-15) return null;
          return a / b;
        }

        return a;
      }

      equalsBtn.addEventListener("click", submitAnswer);

      function submitAnswer() {
        if (completed) return;

        const answer = evaluateCalculatorExpression();

        if (answer === null || !Number.isFinite(answer)) {
          feedbackEl.innerHTML = "Please enter a valid number or division expression, such as <code>3.2÷0.8</code>.";
          return;
        }

        const actual = currentFunction.df(targetN);
        const signedDifference = answer - actual;
        const absoluteError = Math.abs(signedDifference);

        runningError += absoluteError;
        answered += 1;

        errorTotalEl.textContent = fmt(runningError, 6);
        progressFill.style.width = `${(answered / totalQuestions) * 100}%`;

        feedbackEl.innerHTML =
          `Your answer: <strong>${fmt(answer, 6)}</strong>. ` +
          `Actual value: <strong>${fmt(actual, 6)}</strong>. ` +
          `Difference, answer − actual: <strong>${fmt(signedDifference, 6)}</strong>. ` +
          `Absolute error added: <strong>${fmt(absoluteError, 6)}</strong>.`;

        if (answered >= totalQuestions) {
          completed = true;
          questionEl.innerHTML = `<span class="complete">Activity complete!</span>`;
          answerInput.disabled = true;
          equalsBtn.disabled = true;
          clearBtn.disabled = true;
          divideBtn.disabled = true;
          document.querySelectorAll("[data-val]").forEach(btn => btn.disabled = true);
          dxLabel.hidden = true;
          dyLabel.hidden = true;

          window.activityComplete();
          return;
        }

        startNewQuestion(true);
      }

      function startNewQuestion(keepFeedback = false) {
        const chosen = chooseFunctionAndN();

        currentFunction = chosen.fn;
        targetN = chosen.n;

        view = { ...INITIAL };
        delta = null;
        answerInput.value = "";

        questionEl.textContent =
          `What is f'(${targetN}): the slope of the function f at x=${targetN}?`;

        if (!keepFeedback) feedbackEl.textContent = "";

        render();
      }

      const resizeObserver = new ResizeObserver(resizeCanvas);
      resizeObserver.observe(graphWrap);

      startNewQuestion(false);
      resizeCanvas();
    })();
  </script>
</body>
