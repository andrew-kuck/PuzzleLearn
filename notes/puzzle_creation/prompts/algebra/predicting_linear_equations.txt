create an activity using html, css, and javascript. most of page filled with large coordinate plane, but without tick marks or grid lines. above the plane is an equation y = mx. y and x are left as variables and m is an integer between -10 and 10, or one of the fractions 1/2, 1/3, 1/4, -1/2, -1/3, -1/4. there is a blue equation line spanning the entire diagonal of the coordinate plane and initially corresponding to slope 1 through the origin (regardless of the equation). the user can drag the equation line so that it rotates to any slope, still passing through the origin. the slope of the equation line is not displayed. when the user releases their drag, the user's blue equation line stops there and a green equation line appears. the green equation line corresponds to the equation above the coordinate plane. the area swept out between the blue line and green line is colored red. the number of degrees between the two lines is displayed in the red region, and then that degree quantity is added to an accumulating error count at the top of the screen. after this the blue line rotate to the green line, there becomes only one line (blue), and the red disappears. then a green dot appears on either the x or y axis, corresponding to a random rational one-place decimal number between -10 and 10. -10 would appear at the far left or bottom of the corresponding axis, and 10 would appear at the far right or top. if the value is for x, it must be such that the value for y (given the equation) is also between -10 and 10. and if the value is for y, it must be such that the value for x (given the equation) is between -10 and 10. if given a dot on the x axis, the user must click a point on the y axis. and if given a dot on the y axis, the user must click a point on the x axis. the user's click becomes a blue dot, and a green dot appears where they should have clicked. a red line appears connecting the blue and green dot, and the length of that line is added to the error total. then after a short pause, a new question is presented. a progress bar at the top advances with each completed question, so that it is full after 15 completed questions, at which point the function activityComplete() is called. make sure that the plane is always a square regardless of the size of the screen. also, prevent red and blue lines from accumulating at the very bottom of the coordinate plane. after the user selects a blue dot and the correct green dot appears with red line in between the two, have the blue dot slide to the green dot before advancing to the next question.

make sure that the slope (m) cannot be 0. when placing a point on the x or y axis, make sure that it is at a point greater than or equal to 1 or less than or equal to -1. also, when selecting the corresponding point on the other axis, require the user to click on or very near the actual axis; if they don't, have nothing happen

Below is a complete single-file HTML activity. You can save it as `index.html` and open it in a browser.

It includes:

- Square coordinate plane with axes only — no tick marks or grid lines
- Random equation `y = mx`
- Draggable blue line through the origin
- Green correct line, red swept angle region, and accumulated angle error
- Axis-dot follow-up task with blue/green dots, red error line, and sliding correction animation
- Progress bar completing after 15 questions
- Calls `activityComplete()` when finished

    /************************************************************

    const TOTAL_QUESTIONS = 15;
    const SVG_SIZE = 1000;
    const COORD_MIN = -10;
    const COORD_MAX = 10;
    const COORD_SPAN = 20;

    const plane = document.getElementById("plane");
    const blueLine = document.getElementById("blueLine");
    const greenLine = document.getElementById("greenLine");
    const redLayer = document.getElementById("redLayer");
    const dotLayer = document.getElementById("dotLayer");

    const equationEl = document.getElementById("equation");
    const instructionsEl = document.getElementById("instructions");
    const errorTotalEl = document.getElementById("errorTotal");

    const questionNumberEl = document.getElementById("questionNumber");

    const slopeChoices = [];
    for (let i = -10; i <= 10; i++) {
      slopeChoices.push({ value: i, label: String(i) });
    }
    slopeChoices.push(
      { value: 1 / 2, label: "1/2" },
      { value: 1 / 3, label: "1/3" },
      { value: 1 / 4, label: "1/4" },
      { value: -1 / 2, label: "-1/2" },
      { value: -1 / 3, label: "-1/3" },
      { value: -1 / 4, label: "-1/4" }

    let state = "idle";
    let completedQuestions = 0;
    let totalError = 0;
    let completeCalled = false;

    let currentSlope = null;
    let targetAngle = 0;
    let userAngle = Math.PI / 4;
    let isDragging = false;
    let currentDotTask = null;

    const NS = "http://www.w3.org/2000/svg";

    function randInt(min, max) {
      return Math.floor(Math.random() * (max - min + 1)) + min;
    }

    function clamp(value, min, max) {
      return Math.max(min, Math.min(max, value));
    }

    function formatNumber(value) {
      const rounded = Math.round(value * 10) / 10;
      return rounded.toFixed(1);
    }

    function chooseRandomSlope() {
      return slopeChoices[randInt(0, slopeChoices.length - 1)];
    }

    function equationText(slope) {
      return `y = ${slope.label}x`;
    }

    function updateTopDisplay() {
      errorTotalEl.textContent = totalError.toFixed(1);
      questionNumberEl.textContent = Math.min(completedQuestions + 1, TOTAL_QUESTIONS);
      progressFill.style.width = `${(completedQuestions / TOTAL_QUESTIONS) * 100}%`;
    }

    function toSvgPoint(x, y) {
      return {
        x: ((x - COORD_MIN) / COORD_SPAN) * SVG_SIZE,
        y: ((COORD_MAX - y) / COORD_SPAN) * SVG_SIZE
      };
    }

    function fromSvgPoint(svgX, svgY) {
      return {
        x: (svgX / SVG_SIZE) * COORD_SPAN + COORD_MIN,
        y: COORD_MAX - (svgY / SVG_SIZE) * COORD_SPAN
      };
    }

    function eventToSvgPoint(evt) {
      const pt = plane.createSVGPoint();
      pt.x = evt.clientX;
      pt.y = evt.clientY;
      return pt.matrixTransform(plane.getScreenCTM().inverse());
    }

    function normalizeLineAngle(angle) {
      while (angle > Math.PI / 2) angle -= Math.PI;
      while (angle <= -Math.PI / 2) angle += Math.PI;
      return angle;
    }

    function shortestAngleDelta(from, to) {
      let delta = to - from;
      while (delta > Math.PI / 2) delta -= Math.PI;
      while (delta < -Math.PI / 2) delta += Math.PI;
      return delta;
    }

    function acuteAngleDifference(a, b) {
      return Math.abs(shortestAngleDelta(a, b));
    }

    function angleFromPointer(evt) {
      const svgPt = eventToSvgPoint(evt);
      const mathPt = fromSvgPoint(svgPt.x, svgPt.y);

      // Avoid unstable angle right at the origin.
      if (Math.hypot(mathPt.x, mathPt.y) < 0.15) {
        return userAngle;
      }

      return normalizeLineAngle(Math.atan2(mathPt.y, mathPt.x));
    }

    function lineEndpointsForAngle(angle) {
      const dx = Math.cos(angle);
      const dy = Math.sin(angle);

      const tx = Math.abs(dx) < 1e-9 ? Infinity : 10 / Math.abs(dx);
      const ty = Math.abs(dy) < 1e-9 ? Infinity : 10 / Math.abs(dy);
      const t = Math.min(tx, ty);

      return {
        x1: -t * dx,
        y1: -t * dy,
        x2: t * dx,
        y2: t * dy
      };
    }

    function setLineToAngle(lineEl, angle) {
      const e = lineEndpointsForAngle(angle);
      const p1 = toSvgPoint(e.x1, e.y1);
      const p2 = toSvgPoint(e.x2, e.y2);

      lineEl.setAttribute("x1", p1.x);
      lineEl.setAttribute("y1", p1.y);
      lineEl.setAttribute("x2", p2.x);
      lineEl.setAttribute("y2", p2.y);
    }

    function clearDynamicLayers() {
      redLayer.innerHTML = "";
      dotLayer.innerHTML = "";
      greenLine.style.display = "none";
    }

    function addError(amount) {
      totalError += amount;
      updateTopDisplay();
    }

    function drawAngleRegion(fromAngle, toAngle, degreeAmount) {
      redLayer.innerHTML = "";

      const delta = shortestAngleDelta(fromAngle, toAngle);

      if (Math.abs(delta) < 0.0001) {
        const label = document.createElementNS(NS, "text");
        const p = toSvgPoint(2.8, 1.2);
        label.setAttribute("class", "angle-label");
        label.setAttribute("x", p.x);
        label.setAttribute("y", p.y);
        label.setAttribute("text-anchor", "middle");
        label.setAttribute("dominant-baseline", "middle");
        label.textContent = `${degreeAmount.toFixed(1)}°`;
        redLayer.appendChild(label);
        return;
      }

      const endAngle = fromAngle + delta;
      const radiusCoord = 16;
      const radiusSvg = (radiusCoord / COORD_SPAN) * SVG_SIZE;
      const origin = toSvgPoint(0, 0);

      function pointOnAngle(angle) {
        return toSvgPoint(
          radiusCoord * Math.cos(angle),
          radiusCoord * Math.sin(angle)
        );
      }

      function sectorPath(a1, a2) {
        const p1 = pointOnAngle(a1);
        const p2 = pointOnAngle(a2);
        const sweepFlag = a2 - a1 > 0 ? 0 : 1;

        return [
          `M ${origin.x} ${origin.y}`,
          `L ${p1.x} ${p1.y}`,
          `A ${radiusSvg} ${radiusSvg} 0 0 ${sweepFlag} ${p2.x} ${p2.y}`,
          "Z"
        ].join(" ");
      }

      // Draw the swept region on both opposite sides of the two full lines.
      const pathData =
        sectorPath(fromAngle, endAngle) + " " +
        sectorPath(fromAngle + Math.PI, endAngle + Math.PI);

      const path = document.createElementNS(NS, "path");
      path.setAttribute("class", "red-sector");
      path.setAttribute("d", pathData);
      redLayer.appendChild(path);

      const mid = fromAngle + delta / 2;
      const labelPoint = toSvgPoint(
        4.2 * Math.cos(mid),
        4.2 * Math.sin(mid)
      );

      const label = document.createElementNS(NS, "text");
      label.setAttribute("class", "angle-label");
      label.setAttribute("x", labelPoint.x);
      label.setAttribute("y", labelPoint.y);
      label.setAttribute("text-anchor", "middle");
      label.setAttribute("dominant-baseline", "middle");
      label.textContent = `${degreeAmount.toFixed(1)}°`;
      redLayer.appendChild(label);
    }

    function animateLineToAngle(fromAngle, toAngle, duration = 650) {
      return new Promise(resolve => {
        const start = performance.now();
        const delta = shortestAngleDelta(fromAngle, toAngle);

        function ease(t) {
          return 1 - Math.pow(1 - t, 3);
        }

        function frame(now) {
          const rawT = clamp((now - start) / duration, 0, 1);
          const eased = ease(rawT);
          const angle = normalizeLineAngle(fromAngle + delta * eased);
          setLineToAngle(blueLine, angle);

          if (rawT < 1) {
            requestAnimationFrame(frame);
          } else {
            setLineToAngle(blueLine, toAngle);
            resolve();
          }
        }

        requestAnimationFrame(frame);
      });
    }

    function randomOneDecimalInRange(min, max) {
      const lo = Math.ceil(min * 10);
      const hi = Math.floor(max * 10);
      return randInt(lo, hi) / 10;
    }

    function createDotTask() {
      const m = currentSlope.value;

      // If m = 0, giving y on the y-axis would not produce a unique x-value.
      const canGiveY = Math.abs(m) > 1e-9;
      const giveX = canGiveY ? Math.random() < 0.5 : true;

      if (giveX) {
        const maxAbsX = Math.abs(m) < 1e-9 ? 10 : Math.min(10, 10 / Math.abs(m));
        const x = randomOneDecimalInRange(-maxAbsX, maxAbsX);
        const y = m * x;

        return {
          sourceAxis: "x",
          targetAxis: "y",
          sourceValue: x,
          correctValue: y
        };
      } else {
        const maxAbsY = Math.min(10, 10 * Math.abs(m));
        const y = randomOneDecimalInRange(-maxAbsY, maxAbsY);
        const x = y / m;

        return {
          sourceAxis: "y",
          targetAxis: "x",
          sourceValue: y,
          correctValue: x
        };
      }
    }

    function makeSvgEl(tag, attrs = {}) {
      const el = document.createElementNS(NS, tag);
      for (const [key, value] of Object.entries(attrs)) {
        el.setAttribute(key, value);
      }
      return el;
    }

    function drawDot(group, className, x, y, radius = 12) {
      const p = toSvgPoint(x, y);
      const circle = makeSvgEl("circle", {
        class: className,
        cx: p.x,
        cy: p.y,
        r: radius
      });
      group.appendChild(circle);
      return circle;
    }

    function drawDotLabel(group, text, x, y, axisType = "x") {
      const p = toSvgPoint(x, y);
      const label = makeSvgEl("text", {
        class: "dot-label",
        x: p.x,
        y: p.y,
        "text-anchor": "middle"
      });

      if (axisType === "x") {
        label.setAttribute("dy", y >= 8.7 ? "34" : "-24");
      } else {
        label.setAttribute("dx", x >= 0 ? "56" : "-56");
        label.setAttribute("dy", "10");
      }

      label.textContent = text;
      group.appendChild(label);
      return label;
    }

    function drawSourceDotTask(task) {
      dotLayer.innerHTML = "";

      if (task.sourceAxis === "x") {
        drawDot(dotLayer, "dot-green", task.sourceValue, 0);
        drawDotLabel(
          dotLayer,
          `x = ${formatNumber(task.sourceValue)}`,
          task.sourceValue,
          0,
          "x"
        );

        instructionsEl.textContent =
          `Given x = ${formatNumber(task.sourceValue)}, click the matching point on the y-axis.`;
      } else {
        drawDot(dotLayer, "dot-green", 0, task.sourceValue);
        drawDotLabel(
          dotLayer,
          `y = ${formatNumber(task.sourceValue)}`,
          0,
          task.sourceValue,
          "y"
        );

        instructionsEl.textContent =
          `Given y = ${formatNumber(task.sourceValue)}, click the matching point on the x-axis.`;
      }
    }

    function drawRedErrorLine(x1, y1, x2, y2) {
      const p1 = toSvgPoint(x1, y1);
      const p2 = toSvgPoint(x2, y2);

      const line = makeSvgEl("line", {
        class: "red-error-line",
        x1: p1.x,
        y1: p1.y,
        x2: p2.x,
        y2: p2.y
      });

      dotLayer.appendChild(line);
      return line;
    }

    function animateDotToCorrect(blueDot, redLine, task, userValue, correctValue, duration = 700) {
      return new Promise(resolve => {
        const start = performance.now();

        function ease(t) {
          return 1 - Math.pow(1 - t, 3);
        }

        function frame(now) {
          const rawT = clamp((now - start) / duration, 0, 1);
          const eased = ease(rawT);
          const currentValue = userValue + (correctValue - userValue) * eased;

          let blueX, blueY, correctX, correctY;

          if (task.targetAxis === "y") {
            blueX = 0;
            blueY = currentValue;
            correctX = 0;
            correctY = correctValue;
          } else {
            blueX = currentValue;
            blueY = 0;
            correctX = correctValue;
            correctY = 0;
          }

          const bp = toSvgPoint(blueX, blueY);
          const cp = toSvgPoint(correctX, correctY);

          blueDot.setAttribute("cx", bp.x);
          blueDot.setAttribute("cy", bp.y);

          redLine.setAttribute("x1", bp.x);
          redLine.setAttribute("y1", bp.y);
          redLine.setAttribute("x2", cp.x);
          redLine.setAttribute("y2", cp.y);

          if (rawT < 1) {
            requestAnimationFrame(frame);
          } else {
            resolve();
          }
        }

        requestAnimationFrame(frame);
      });
    }

    function startDotPhase() {
      state = "dot";
      redLayer.innerHTML = "";
      greenLine.style.display = "none";
      setLineToAngle(blueLine, targetAngle);

      currentDotTask = createDotTask();
      drawSourceDotTask(currentDotTask);
    }

    function completeQuestionAndContinue() {
      completedQuestions++;
      updateTopDisplay();

      if (completedQuestions >= TOTAL_QUESTIONS) {
        finishActivity();
      } else {
        setTimeout(beginQuestion, 450);
      }
    }

    function finishActivity() {
      if (completeCalled) return;
      completeCalled = true;

      state = "complete";
      progressFill.style.width = "100%";
      instructionsEl.textContent = "Activity complete. Nice work!";
      activityComplete();
    }

    function beginQuestion() {
      state = "drag";
      clearDynamicLayers();

      currentSlope = chooseRandomSlope();
      targetAngle = Math.atan(currentSlope.value);

      // The blue line always begins with slope 1, regardless of the equation.
      userAngle = Math.PI / 4;

      equationEl.textContent = equationText(currentSlope);
      instructionsEl.textContent = "Drag the blue line to match the equation’s slope.";

      setLineToAngle(blueLine, userAngle);
      updateTopDisplay();
    }

    async function finalizeLineGuess() {
      state = "lineFeedback";
      plane.classList.remove("dragging");

      const angleErrorRad = acuteAngleDifference(userAngle, targetAngle);
      const angleErrorDeg = angleErrorRad * 180 / Math.PI;

      setLineToAngle(greenLine, targetAngle);
      greenLine.style.display = "block";

      drawAngleRegion(userAngle, targetAngle, angleErrorDeg);
      addError(angleErrorDeg);

      instructionsEl.textContent =
        `Angle error: ${angleErrorDeg.toFixed(1)}°. Watch the blue line rotate to the correct line.`;

      await new Promise(resolve => setTimeout(resolve, 900));
      await animateLineToAngle(userAngle, targetAngle, 650);

      redLayer.innerHTML = "";
      greenLine.style.display = "none";

      await new Promise(resolve => setTimeout(resolve, 250));
      startDotPhase();
    }

    function handleDotClick(evt) {
      if (state !== "dot") return;

      state = "dotFeedback";

      const svgPt = eventToSvgPoint(evt);
      const mathPt = fromSvgPoint(svgPt.x, svgPt.y);

      const task = currentDotTask;
      let userValue;
      let blueX, blueY;
      let correctX, correctY;

      if (task.targetAxis === "y") {
        userValue = clamp(mathPt.y, -10, 10);
        blueX = 0;
        blueY = userValue;
        correctX = 0;
        correctY = task.correctValue;
      } else {
        userValue = clamp(mathPt.x, -10, 10);
        blueX = userValue;
        blueY = 0;
        correctX = task.correctValue;
        correctY = 0;
      }

      const distanceError = Math.abs(userValue - task.correctValue);
      addError(distanceError);

      const redLine = drawRedErrorLine(blueX, blueY, correctX, correctY);
      const blueDot = drawDot(dotLayer, "dot-blue", blueX, blueY);
      drawDot(dotLayer, "dot-green", correctX, correctY);

      if (task.targetAxis === "y") {
        drawDotLabel(
          dotLayer,
          `correct y = ${formatNumber(task.correctValue)}`,
          correctX,
          correctY,
          "y"
        );
      } else {
        drawDotLabel(
          dotLayer,
          `correct x = ${formatNumber(task.correctValue)}`,
          correctX,
          correctY,
          "x"
        );
      }

      instructionsEl.textContent =
        `Distance error: ${distanceError.toFixed(1)}. The blue dot will slide to the correct point.`;

      animateDotToCorrect(
        blueDot,
        redLine,
        task,
        userValue,
        task.correctValue,
        700
      ).then(() => {
        setTimeout(completeQuestionAndContinue, 550);
      });
    }

    plane.addEventListener("pointerdown", evt => {
      if (state === "drag") {
        isDragging = true;
        plane.classList.add("dragging");
        plane.setPointerCapture(evt.pointerId);

        userAngle = angleFromPointer(evt);
        setLineToAngle(blueLine, userAngle);
      } else if (state === "dot") {
        handleDotClick(evt);
      }
    });

    plane.addEventListener("pointermove", evt => {
      if (!isDragging || state !== "drag") return;

      userAngle = angleFromPointer(evt);
      setLineToAngle(blueLine, userAngle);
    });

    plane.addEventListener("pointerup", evt => {
      if (!isDragging || state !== "drag") return;

      isDragging = false;
      plane.releasePointerCapture(evt.pointerId);
      finalizeLineGuess();
    });

    plane.addEventListener("pointercancel", () => {
      isDragging = false;
      plane.classList.remove("dragging");
    });

    beginQuestion();
  </script>
