create an activity using html, css, and javascript. show a 3 x 3 grid where each square in the grid has one of three images: an apple, a banana, or an orange. assign at random a price for each of the three fruits. the price will be an integer between 1 and 20 (inclusive). in the margins to the right and bottom of the grid, show what that row or column would sum to, after adding together the costs of the fruits in that row or column. display this sum with a dollar sign ($). below the grid, have a place for the user to input the cost of each of the three fruits. if all the inputs are correct, a green checkmark appears on the screen, a new problem is presented, and a progress bar is advanced one unit. if any of the inputs are not correct, a red x appears over the screen, the same problem remains, and the progress bar goes back one unit. the progress bar extends from -10 to 10 (without labels or tick marks except for a gray vertical line in the middle indicating 0). 0 displays as nothing, negative numbers display as red fill extending to the left of 0, and positive numbers display as green fill extending to the right of 0. progress cannot go below -10. when progress reaches 10 a function activityComplete() is called.

Below is a complete single-file HTML activity. You can paste it into an `.html` file and open it in a browser.

It includes:

- 3 × 3 fruit grid
- Random apple/banana/orange prices from `$1` to `$20`
- Row and column sums
- User inputs for each fruit price
- Green check for correct answers
- Red X for incorrect answers
- Progress bar from `-10` to `10`
- Calls `activityComplete()` when progress reaches `10`

    const fruitNames = ["apple", "banana", "orange"];

    let prices = {};
    let grid = [];
    let progress = 0;

    function svgToDataUri(svg) {
      return "data:image/svg+xml;charset=UTF-8," + encodeURIComponent(svg);
    }

    const imageSources = {
      apple: svgToDataUri(fruitImages.apple),
      banana: svgToDataUri(fruitImages.banana),
      orange: svgToDataUri(fruitImages.orange)
    };

    document.getElementById("appleIcon").src = imageSources.apple;
    document.getElementById("bananaIcon").src = imageSources.banana;
    document.getElementById("orangeIcon").src = imageSources.orange;

    function randomPrice() {
      return Math.floor(Math.random() * 20) + 1;
    }

    function randomFruit() {
      return fruitNames[Math.floor(Math.random() * fruitNames.length)];
    }

    function generateProblem() {
      prices = {
        apple: randomPrice(),
        banana: randomPrice(),
        orange: randomPrice()
      };

      /*
        Generate a random grid that contains enough information
        to determine all three fruit prices.
      */
      do {
        grid = [];
        for (let r = 0; r < 3; r++) {
          const row = [];
          for (let c = 0; c < 3; c++) {
            row.push(randomFruit());
          }
          grid.push(row);
        }
      } while (!isGoodGrid(grid));

      renderBoard();
      clearInputs();
    }

    function isGoodGrid(testGrid) {
      const equations = [];

      // Row equations
      for (let r = 0; r < 3; r++) {
        equations.push(countFruits(testGrid[r]));
      }

      // Column equations
      for (let c = 0; c < 3; c++) {
        const column = [];
        for (let r = 0; r < 3; r++) {
          column.push(testGrid[r][c]);
        }
        equations.push(countFruits(column));
      }

      return matrixRank(equations) === 3;
    }

    function countFruits(items) {
      return [
        items.filter(f => f === "apple").length,
        items.filter(f => f === "banana").length,
        items.filter(f => f === "orange").length
      ];
    }

    function matrixRank(matrix) {
      const m = matrix.map(row => row.slice());
      let rank = 0;
      let row = 0;

      for (let col = 0; col < 3; col++) {
        let pivot = row;

        while (pivot < m.length && Math.abs(m[pivot][col]) < 1e-10) {
          pivot++;
        }

        if (pivot === m.length) continue;

        [m[row], m[pivot]] = [m[pivot], m[row]];

        const pivotValue = m[row][col];
        for (let c = col; c < 3; c++) {
          m[row][c] /= pivotValue;
        }

        for (let r = 0; r < m.length; r++) {
          if (r !== row) {
            const factor = m[r][col];
            for (let c = col; c < 3; c++) {
              m[r][c] -= factor * m[row][c];
            }
          }
        }

        rank++;
        row++;
      }

      return rank;
    }

    function renderBoard() {
      const board = document.getElementById("board");
      board.innerHTML = "";

      for (let r = 0; r < 3; r++) {
        for (let c = 0; c < 3; c++) {
          const fruit = grid[r][c];

          const cell = document.createElement("div");
          cell.className = "cell";

          const img = document.createElement("img");
          img.src = imageSources[fruit];
          img.alt = fruit;

          cell.appendChild(img);
          board.appendChild(cell);
        }

        const rowSum = grid[r].reduce((sum, fruit) => sum + prices[fruit], 0);
        const sumCell = document.createElement("div");
        sumCell.className = "sum";
        sumCell.textContent = "$" + rowSum;
        board.appendChild(sumCell);
      }

      for (let c = 0; c < 3; c++) {
        let colSum = 0;
        for (let r = 0; r < 3; r++) {
          colSum += prices[grid[r][c]];
        }

        const sumCell = document.createElement("div");
        sumCell.className = "sum";
        sumCell.textContent = "$" + colSum;
        board.appendChild(sumCell);
      }

      const corner = document.createElement("div");
      corner.className = "corner";
      board.appendChild(corner);
    }

    function clearInputs() {
      document.getElementById("appleInput").value = "";
      document.getElementById("bananaInput").value = "";
      document.getElementById("orangeInput").value = "";
    }

    function checkAnswers() {
      const appleGuess = Number(document.getElementById("appleInput").value);
      const bananaGuess = Number(document.getElementById("bananaInput").value);
      const orangeGuess = Number(document.getElementById("orangeInput").value);

      const correct =
        appleGuess === prices.apple &&
        bananaGuess === prices.banana &&
        orangeGuess === prices.orange;

      if (correct) {
        showSymbol("✓", true);
        progress = Math.min(10, progress + 1);
        updateProgressBar();

        if (progress >= 10) {
          activityComplete();
        } else {
          setTimeout(generateProblem, 700);
        }
      } else {
        showSymbol("✕", false);
        progress = Math.max(-10, progress - 1);
        updateProgressBar();
      }
    }

    function showSymbol(symbol, isCorrect) {
      const overlay = document.getElementById("overlaySymbol");
      overlay.textContent = symbol;
      overlay.className = "overlay-symbol " + (isCorrect ? "correct" : "incorrect");
      overlay.style.display = "flex";

      setTimeout(() => {
        overlay.style.display = "none";
      }, 650);
    }

    function updateProgressBar() {
      const fill = document.getElementById("progressFill");

      fill.className = "progress-fill";
      fill.style.width = "0";

      if (progress > 0) {
        fill.classList.add("positive");
        fill.style.width = `${progress * 5}%`;
      } else if (progress < 0) {
        fill.classList.add("negative");
        fill.style.width = `${Math.abs(progress) * 5}%`;
      }
    }

    function activityComplete() {
      alert("Activity complete! Great work!");
    }

    generateProblem();
    updateProgressBar();
  </script>
