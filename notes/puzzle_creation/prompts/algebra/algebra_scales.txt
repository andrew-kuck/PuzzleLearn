create an activity using html, css, and javascript. show a scale on the screen. the scale depicts an algebra equation of the form a + bx = c + dx, where a, b, c, and d are all integers between 0 and 9 (inclusive). ensure that solving for x yields a single solution. on the scale, a and c are represented as a small squares and c small squares, respectively. bx is represented as b medium circles, each with an x in the middle; and dx is is represented as d medium circles, each with an x in the middle. the user can drag any of these objects to a trashcan, at which point they disappear (if the user releases anywhere else the object goes back to where is was). after an object is removed, the scale tilts to that side; at that point, the subsequent object the user can remove is the type of object on the other side that will balance the scale (a square if the user removed a square and a circle if the user removed a circle). after the user has balanced the scale by removing such an object, they can once again pick any object on either side to remove. there is a button to reset the scale to its original setting. at any time the user can answer the question by filling in an input displayed as x = [input]. if the input is correct, a green checkmark appears on the screen, a new problem is presented, and a progress bar is advanced one unit. if the input is not correct, a red x appears over the screen, the same problem remains, and the progress bar goes back one unit. the progress bar extends from -10 to 10 (without labels or tick marks except for a gray vertical line in the middle indicating 0). 0 displays as nothing, negative numbers display as red fill extending to the left of 0, and positive numbers display as green fill extending to the right of 0. progress cannot go below -10. when progress reaches 10 a function activityComplete() is called.

    let problem;

    function randInt(min, max) {
      return Math.floor(Math.random() * (max - min + 1)) + min;
    }

    function generateProblem() {
      /*
        Generates a + bx = c + dx with:
        - a, b, c, d from 1 to 9, which is within 0 to 9 inclusive
        - b !== d, so there is a single solution
        - integer x solution
      */
      for (let attempt = 0; attempt < 3000; attempt++) {
        const x = randInt(-5, 5);
        const a = randInt(1, 9);
        const b = randInt(1, 9);
        const d = randInt(1, 9);

        if (b === d) continue;

        const c = a + (b - d) * x;

        if (c >= 1 && c <= 9) {
          return { a, b, c, d, x };
        }
      }

      // Reliable fallback: 2 + 3x = 8 + 1x, x = 3
      return { a: 2, b: 3, c: 8, d: 1, x: 3 };
    }

    function makeTokens(p) {
      const list = [];
      let id = 0;

      for (let i = 0; i < p.a; i++) {
        list.push({
          id: `t${id++}`,
          side: "left",
          type: "square",
          removed: false
        });
      }

      for (let i = 0; i < p.b; i++) {
        list.push({
          id: `t${id++}`,
          side: "left",
          type: "circle",
          removed: false
        });
      }

      for (let i = 0; i < p.c; i++) {
        list.push({
          id: `t${id++}`,
          side: "right",
          type: "square",
          removed: false
        });
      }

      for (let i = 0; i < p.d; i++) {
        list.push({
          id: `t${id++}`,
          side: "right",
          type: "circle",
          removed: false
        });
      }

      return list;
    }

    function formatEquation(p) {
      return `${p.a} + ${p.b}x = ${p.c} + ${p.d}x`;
    }

    function startNewProblem() {
      problem = generateProblem();
      originalTokens = makeTokens(problem);
      tokens = originalTokens.map(t => ({ ...t }));
      pendingRemoval = null;
      completed = false;
      completeMessage.textContent = "";
      answerInput.value = "";
      render();
    }

    function resetScale() {
      tokens = originalTokens.map(t => ({ ...t }));
      pendingRemoval = null;
      answerInput.value = "";
      render();
    }

    function render() {
      equationText.textContent = formatEquation(problem);

      leftPan.innerHTML = "";
      rightPan.innerHTML = "";

      for (const token of tokens) {
        if (token.removed) continue;

        const el = document.createElement("div");
        el.className = `token ${token.type}`;
        el.dataset.id = token.id;
        el.setAttribute("role", "button");
        el.setAttribute(
          "aria-label",
          `${token.type === "square" ? "constant square" : "x circle"} on the ${token.side} side`
        );

        if (token.type === "circle") {
          el.textContent = "x";
        }

        if (token.side === "left") {
          leftPan.appendChild(el);
        } else {
          rightPan.appendChild(el);
        }
      }

      updateTilt();
      updateInstructions();
      updateProgressBar();
    }

    function updateTilt() {
      scaleBoard.classList.remove("balanced", "tilt-left", "tilt-right");

      if (!pendingRemoval) {
        scaleBoard.classList.add("balanced");
      } else {
        /*
          If one object was removed from the left, the right side is now heavier,
          so the scale tips right, and vice versa.
        */
        const heavierSide = pendingRemoval.side === "left" ? "right" : "left";
        scaleBoard.classList.add(heavierSide === "left" ? "tilt-left" : "tilt-right");
      }
    }

    function updateInstructions() {
      if (!pendingRemoval) {
        instructions.textContent =
          "Drag an object to the trash. To keep balance, remove the same kind from the other side next.";
      } else {
        const kind = pendingRemoval.type === "square" ? "square" : "x-circle";
        const otherSide = pendingRemoval.side === "left" ? "right" : "left";
        instructions.textContent =
          `Now remove one ${kind} from the ${otherSide} side to balance the scale.`;
      }
    }

    function updateProgressBar() {
      const clamped = Math.max(-10, Math.min(10, progress));

      if (clamped > 0) {
        positiveFill.style.width = `${(clamped / 10) * 50}%`;
        negativeFill.style.width = "0%";
      } else if (clamped < 0) {
        negativeFill.style.width = `${(Math.abs(clamped) / 10) * 50}%`;
        positiveFill.style.width = "0%";
      } else {
        positiveFill.style.width = "0%";
        negativeFill.style.width = "0%";
      }
    }

    function visibleMatchingTokenExists(token) {
      return tokens.some(t =>
        !t.removed &&
        t.side !== token.side &&
        t.type === token.type
      );
    }

    function attemptRemove(tokenId) {
      if (completed) return false;

      const token = tokens.find(t => t.id === tokenId);
      if (!token || token.removed) return false;

      if (!pendingRemoval) {
        /*
          Prevent dead-end moves: the first removal in a pair must have
          a matching object available on the opposite side.
        */
        if (!visibleMatchingTokenExists(token)) {
          showTrashError(
            `There is no matching ${token.type === "square" ? "square" : "x-circle"} on the other side.`
          );
          return false;
        }

        token.removed = true;
        pendingRemoval = {
          side: token.side,
          type: token.type
        };
        render();
        return true;
      }

      const correctOtherSide = token.side !== pendingRemoval.side;
      const correctType = token.type === pendingRemoval.type;

      if (correctOtherSide && correctType) {
        token.removed = true;
        pendingRemoval = null;
        render();
        return true;
      }

      showTrashError("To balance the scale, remove the same kind of object from the other side.");
      return false;
    }

    function showTrashError(message) {
      instructions.textContent = message;
      trash.classList.add("flash");
      setTimeout(() => trash.classList.remove("flash"), 350);
    }

    function showFeedback(isCorrect) {
      feedback.className = `feedback show ${isCorrect ? "correct" : "incorrect"}`;
      feedback.textContent = isCorrect ? "✓" : "✕";

      setTimeout(() => {
        feedback.className = "feedback";
        feedback.textContent = "";
      }, 700);
    }

    function checkAnswer(event) {
      event.preventDefault();

      if (completed) return;

      const userValue = Number(answerInput.value);

      if (answerInput.value.trim() === "" || !Number.isFinite(userValue)) {
        showTrashError("Enter a value for x first.");
        answerInput.focus();
        return;
      }

      const isCorrect = userValue === problem.x;

      if (isCorrect) {
        progress = Math.min(10, progress + 1);
      } else {
        progress = Math.max(-10, progress - 1);
      }

      updateProgressBar();
      showFeedback(isCorrect);

      if (progress >= 10) {
        completed = true;
        completeMessage.textContent = "Great work — activity complete!";
        setTimeout(() => {
          window.activityComplete();
        }, 750);
        return;
      }

      setTimeout(() => {
        startNewProblem();
        answerInput.focus();
      }, 800);
    }

    function handlePointerDown(event) {
      if (completed) return;

      const tokenEl = event.target.closest(".token");
      if (!tokenEl) return;

      event.preventDefault();

      const rect = tokenEl.getBoundingClientRect();
      const proxy = tokenEl.cloneNode(true);
      proxy.classList.add("drag-proxy");
      proxy.style.width = `${rect.width}px`;
      proxy.style.height = `${rect.height}px`;
      proxy.style.left = `${rect.left}px`;
      proxy.style.top = `${rect.top}px`;

      document.body.appendChild(proxy);

      tokenEl.style.opacity = "0.25";

      activeDrag = {
        id: tokenEl.dataset.id,
        originalEl: tokenEl,
        proxy,
        offsetX: event.clientX - rect.left,
        offsetY: event.clientY - rect.top
      };

      moveProxy(event.clientX, event.clientY);

      document.addEventListener("pointermove", handlePointerMove);
      document.addEventListener("pointerup", handlePointerUp);
      document.addEventListener("pointercancel", handlePointerUp);
    }

    function moveProxy(clientX, clientY) {
      if (!activeDrag) return;

      activeDrag.proxy.style.left = `${clientX - activeDrag.offsetX}px`;
      activeDrag.proxy.style.top = `${clientY - activeDrag.offsetY}px`;

      const trashRect = trash.getBoundingClientRect();
      const overTrash =
        clientX >= trashRect.left &&
        clientX <= trashRect.right &&
        clientY >= trashRect.top &&
        clientY <= trashRect.bottom;

      trash.classList.toggle("over", overTrash);
    }

    function handlePointerMove(event) {
      if (!activeDrag) return;
      event.preventDefault();
      moveProxy(event.clientX, event.clientY);
    }

    function handlePointerUp(event) {
      if (!activeDrag) return;

      const trashRect = trash.getBoundingClientRect();
      const droppedOnTrash =
        event.clientX >= trashRect.left &&
        event.clientX <= trashRect.right &&
        event.clientY >= trashRect.top &&
        event.clientY <= trashRect.bottom;

      trash.classList.remove("over");

      const { id, originalEl, proxy } = activeDrag;

      if (proxy.parentNode) {
        proxy.parentNode.removeChild(proxy);
      }

      let removed = false;

      if (droppedOnTrash) {
        removed = attemptRemove(id);
      }

      if (!removed && originalEl && originalEl.isConnected) {
        originalEl.style.opacity = "";
      }

      activeDrag = null;

      document.removeEventListener("pointermove", handlePointerMove);
      document.removeEventListener("pointerup", handlePointerUp);
      document.removeEventListener("pointercancel", handlePointerUp);
    }

    document.querySelector(".scale-area").addEventListener("pointerdown", handlePointerDown);
    resetBtn.addEventListener("click", resetScale);
    answerForm.addEventListener("submit", checkAnswer);

    startNewProblem();
  </script>
