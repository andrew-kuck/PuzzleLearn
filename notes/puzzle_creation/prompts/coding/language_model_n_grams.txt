    /***********************
     * BASIC UTILITIES

    function escapeHTML(str) {
      return str.replace(/[&<>"']/g, ch => ({
        "&": "&amp;",
        "<": "&lt;",
        ">": "&gt;",
        '"': "&quot;",
        "'": "&#039;"
      }[ch]));
    }

    function tokenize(text) {
      // Whitespace-separated words, except periods are always independent tokens.
      // Examples:
      // "Hello world." -> ["Hello", "world", "."]
      // "Mr. Darcy"    -> ["Mr", ".", "Darcy"]
      return text.match(/[^\s.]+|\./g) || [];
    }

    function tokenNeedsLeadingSpace(token, previousToken) {
      if (!previousToken) return false;
      if (token === ".") return false;
      return true;
    }

    function formatTokensAsHTML(tokenArray, options = {}) {
      const {
        className = "token",
        contextIndexes = new Set(),
        targetIndex = null
      } = options;

      let html = "";

      tokenArray.forEach((token, i) => {
        const previous = tokenArray[i - 1];
        const leading = tokenNeedsLeadingSpace(token, previous) ? " " : "";

        let extraClass = "";
        if (contextIndexes.has(i)) extraClass += " context-highlight";
        if (i === targetIndex) extraClass += " target-highlight";

        html += `${leading}<span class="${className}${extraClass}" data-token-index="${i}">${escapeHTML(token)}</span>`;
      });

      return html;
    }

    function updateModelDisplay() {
      modelDisplay.textContent = JSON.stringify(model, null, 2);

    function setLeftStatus(message, isError = false) {
      leftStatus.innerHTML = isError
        ? `<span class="error">${escapeHTML(message)}</span>`
        : message;
    }

    function setRightStatus(message, isError = false) {
      rightStatus.innerHTML = isError
        ? `<span class="error">${escapeHTML(message)}</span>`
        : message;
    }

    function clearHighlights() {
      tokenSpans.forEach(span => {
        span.classList.remove("context-highlight", "target-highlight");
      });

    function highlightTrainingWindow(targetIndex) {
      clearHighlights();

        tokenSpans[targetIndex]?.classList.add("target-highlight");
        tokenSpans[targetIndex]?.scrollIntoView({
          behavior: "smooth",

      for (let i = contextStart; i <= contextEnd; i++) {
        tokenSpans[i]?.classList.add("context-highlight");
      }

      tokenSpans[targetIndex]?.classList.add("target-highlight");

        behavior: "smooth",
        block: "center"
      });
    }

    /***********************
     * GUTENBERG SEARCH / LOAD
     ***********************/
    function getPlainTextUrl(book) {
      const formats = book.formats || {};
      const entries = Object.entries(formats);

      const preferred = entries.find(([mime, url]) =>
        mime.toLowerCase().includes("text/plain") &&
        mime.toLowerCase().includes("utf-8") &&
        !url.toLowerCase().endsWith(".zip")
      );

      if (preferred) return preferred[1];

      const fallback = entries.find(([mime, url]) =>
        mime.toLowerCase().includes("text/plain") &&
        !url.toLowerCase().endsWith(".zip")
      );

      return fallback ? fallback[1] : null;
    }

    async function fetchTextWithFallback(url) {
      // Project Gutenberg text URLs usually work directly.
      // If a browser blocks the direct request, this attempts a public CORS proxy.
      try {
        const direct = await fetch(url);
        if (!direct.ok) throw new Error(`Direct text fetch failed: ${direct.status}`);
        return await direct.text();
      } catch (directError) {
        const proxy = `https://api.allorigins.win/raw?url=${encodeURIComponent(url)}`;
        const proxied = await fetch(proxy);
        if (!proxied.ok) {
          throw new Error("Could not fetch text directly or through fallback proxy.");
        }
        return await proxied.text();
      }
    }

    async function searchGutenberg() {
      const query = searchInput.value.trim();

      if (!query) {
        setLeftStatus("Please enter a search term.", true);
        return;
      }

      resetRightSide();

      rawText = "";
      tokens = [];
      tokenSpans = [];
      renderText();

      nInput.disabled = true;
      trainButton.disabled = true;

      setLeftStatus("Searching Project Gutenberg...");
      searchButton.disabled = true;

      try {
        const searchUrl = `https://gutendex.com/books/?search=${encodeURIComponent(query)}`;
        const response = await fetch(searchUrl);

        if (!response.ok) {
          throw new Error(`Search failed with status ${response.status}.`);
        }

        const data = await response.json();

        if (!data.results || data.results.length === 0) {
          throw new Error("No matching Project Gutenberg book was found.");
        }

        // Gutendex returns results ranked by relevance, so the first result is the closest match.
        const book = data.results[0];
        const textUrl = getPlainTextUrl(book);

        if (!textUrl) {
          throw new Error("The best-matching book does not have a plain-text version available.");
        }

        setLeftStatus(`
          Best match: <span class="book-title">${escapeHTML(book.title)}</span>
          ${book.authors?.length ? " by " + escapeHTML(book.authors.map(a => a.name).join(", ")) : ""}
          <br>Loading plain text...
        `);

        rawText = await fetchTextWithFallback(textUrl);
        tokens = tokenize(rawText);

        renderText();

        nInput.disabled = false;
        trainButton.disabled = false;

        setLeftStatus(`
          Loaded <span class="book-title">${escapeHTML(book.title)}</span>
          ${book.authors?.length ? " by " + escapeHTML(book.authors.map(a => a.name).join(", ")) : ""}
          <br>${tokens.length.toLocaleString()} tokens found.
          <div class="small-note">
            Tokenization rule: whitespace-separated words, with periods treated as independent tokens.
          </div>
        `);
      } catch (error) {
        rawText = "";
        tokens = [];
        renderText();
        setLeftStatus(error.message, true);
      } finally {
        searchButton.disabled = false;
      }
    }

    /***********************
     * TRAINING
     ***********************/
    function beginTraining() {
      if (!tokens.length) {
        setRightStatus("Load a text before training.", true);
        return;
      }

      const parsedN = Number(nInput.value);

      if (!Number.isInteger(parsedN) || parsedN < 1) {
        setRightStatus("n must be an integer greater than 0.", true);
        return;
      }

      resetRightSide();

      n = parsedN;
      model = {};
      scanIndex = n === 1 ? 0 : n - 1;
      trainedThroughEnd = false;
      probabilitiesReady = false;

      updateModelDisplay();

      const currentStepButton = document.getElementById("stepButton");
      const currentScanRestButton = document.getElementById("scanRestButton");

      currentStepButton.disabled = scanIndex >= tokens.length;
      currentScanRestButton.disabled = scanIndex >= tokens.length;

      if (scanIndex >= tokens.length) {
        setRightStatus(
          `This text is too short to train a ${n}-gram model. Try a smaller n.`,
          true
        );
        return;
      }

      setRightStatus(
        n === 1
          ? "Unigram model initialized as a single frequency dictionary. Click to scan tokens."
          : `${n}-gram model initialized as a dictionary of dictionaries. Click to scan tokens.`
      );
    }

    function addTrainingObservation(targetIndex) {
      const word = tokens[targetIndex];

      if (n === 1) {
        model[word] = (model[word] || 0) + 1;
      } else {
        const contextStart = targetIndex - (n - 1);
        const context = tokens.slice(contextStart, targetIndex).join(" ");

        if (!model[context]) model[context] = {};
        model[context][word] = (model[context][word] || 0) + 1;
      }
    }

    function finishScanningIfDone() {
      if (scanIndex >= tokens.length) {
        trainedThroughEnd = true;
        clearHighlights();
        showConvertOnlyControls();
        setRightStatus("Scanning complete. Convert frequencies to probabilities.");
      }
    }

    function scanOneStep() {
      if (probabilitiesReady || trainedThroughEnd) return;
      if (scanIndex >= tokens.length) {
        finishScanningIfDone();
        return;
      }

      highlightTrainingWindow(scanIndex);
      addTrainingObservation(scanIndex);
      scanIndex++;

      updateModelDisplay();

      if (scanIndex >= tokens.length) {
        finishScanningIfDone();
      } else {
        setRightStatus(
          `Scanned ${scanIndex.toLocaleString()} of ${tokens.length.toLocaleString()} tokens.`
        );
      }
    }

    function scanRemainder() {
      if (probabilitiesReady || trainedThroughEnd) return;

      clearHighlights();

      while (scanIndex < tokens.length) {
        addTrainingObservation(scanIndex);
        scanIndex++;
      }

      trainedThroughEnd = true;
      updateModelDisplay();
      showConvertOnlyControls();
      setRightStatus("Remainder scanned. Convert frequencies to probabilities.");
    }

    /***********************
     * PROBABILITIES
     ***********************/
    function showConvertOnlyControls() {
      rightControls.innerHTML = `
        <button id="convertButton" class="warning">Convert frequencies to probabilities</button>
      `;

      document.getElementById("convertButton").addEventListener("click", convertToProbabilities);
    }

    function convertToProbabilities() {
      if (probabilitiesReady) return;

      if (n === 1) {
        const total = Object.values(model).reduce((sum, count) => sum + count, 0);

        for (const word of Object.keys(model)) {
          model[word] = model[word] / total;
        }
      } else {
        for (const context of Object.keys(model)) {
          const inner = model[context];
          const total = Object.values(inner).reduce((sum, count) => sum + count, 0);

          for (const word of Object.keys(inner)) {
            inner[word] = inner[word] / total;
          }
        }
      }

      probabilitiesReady = true;
      updateModelDisplay();
      prepareGeneration();
      showGenerationControls();

      setRightStatus("Probabilities ready. Generate words from the model.");
    }

    /***********************

    function weightedChoice(probabilityDictionary) {
      const entries = Object.entries(probabilityDictionary);
      let r = Math.random();

      for (const [word, probability] of entries) {
        r -= probability;
        if (r <= 0) return word;
      }

      // Floating-point safety fallback.
      return entries[entries.length - 1]?.[0] || "";
    }

    function randomKey(obj) {
      const keys = Object.keys(obj);
      return keys[Math.floor(Math.random() * keys.length)];
    }

    function prepareGeneration() {
      generatedTokens = [];
      temporaryNewIndex = null;

      if (n > 1) {
        const seedContext = randomKey(model);

        if (seedContext) {
          generatedTokens = seedContext.split(" ");
        }
      }

      renderGenerated();
    }

    function currentContextString() {
      if (n === 1) return "";

      if (generatedTokens.length < n - 1) {
        return "";
      }

      return generatedTokens.slice(-(n - 1)).join(" ");
    }

    function generateWord() {
      if (!probabilitiesReady) return;

      let nextWord = "";

      if (n === 1) {
        nextWord = weightedChoice(model);
      } else {
        let context = currentContextString();

        // If the current generated context was never observed, restart from a random known context.
        if (!model[context]) {
          context = randomKey(model);
          generatedTokens = context.split(" ");
        }

        nextWord = weightedChoice(model[context]);
      }

      if (!nextWord) return;

      generatedTokens.push(nextWord);
      temporaryNewIndex = generatedTokens.length - 1;

      renderGenerated();

      setTimeout(() => {
        temporaryNewIndex = null;
        renderGenerated();
      }, 550);
    }

    function clearGenerated() {
      prepareGeneration();
      setRightStatus("Generated text cleared. The trained model remains unchanged.");
    }

    function renderGenerated() {
      if (!generatedTokens.length) {
        generatedDisplay.className = "empty";
        generatedDisplay.textContent =
          n === 1
            ? "Click “Generate word” to begin."
            : "No starting context is available.";
        return;
      }

      generatedDisplay.className = "";

      const contextIndexes = new Set();

      if (n > 1 && temporaryNewIndex === null) {
        const start = Math.max(0, generatedTokens.length - (n - 1));

        for (let i = start; i < generatedTokens.length; i++) {
          contextIndexes.add(i);
        }
      }

      let html = "";

      generatedTokens.forEach((token, i) => {
        const previous = generatedTokens[i - 1];
        const leading = tokenNeedsLeadingSpace(token, previous) ? " " : "";

        let classes = "gen-token";

        if (contextIndexes.has(i)) classes += " gen-context";
        if (i === temporaryNewIndex) classes += " gen-new";

        html += `${leading}<span class="${classes}">${escapeHTML(token)}</span>`;
      });

      generatedDisplay.innerHTML = html;
    }

    searchButton.addEventListener("click", searchGutenberg);

    searchInput.addEventListener("keydown", event => {
      if (event.key === "Enter") {
