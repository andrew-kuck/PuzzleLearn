create an activity using html, css, and javascript. there should be no title or instructions on the page, except for a small banner at the top which has a progress bar and that also says "Which group is taller? Sample to make your guess.". the page is broken into 4 quadrants. the items in each quadrant are shaded a different color to indicate their quadrant. each group has 50 small tiles that are blank. each tile corresponds to an integer between 48 and 96. to determine the integers for each tile: for each quadrant pick a target integer x at random between 56 and 88, then assign the tiles in that quadrant to random integers between 48 and 96, such that the average in that quadrant is x. also insure that no two target integers x are within 4 of each other. the user must guess which quadrant has the highest average, and there should be a small button in each quadrant for making this selection. if they make the correct selection, they are rewarded with j points and after an explanation animation a new problem is presented. if they make an incorrect choice, they are penalized k points and after an explanation animation a new problem is presented. the explanation animation shows all of the unrevealed tiles flipping over to reveal their numbers; then going one by one through the quadrants, the tiles of that quadrant merge together to create a single tile with the target number x of that quadrant; if the user clicks on a tile, the tile flips to reveal the integer for that tile, but they are penalized l points. j, k, and l are javascript constants and are initially set to j=30, k=0, and l=1. The points are shown via the progress bar in the top banner. the progress bar is full after m points, which is another javascript constant, initially set to m=150. the progress bar starts at n points, which is another javascript constant, initially set to n=50. within the progress bar fill is the number of the exact points the user has. the user's points cannot drop into the negative, so they just stay at 0 and the user cannot be penalized below that.

at the end of the explanation animation, after the tiles have been flipped, have translucent copies of the flipped tiles converge to the middle of the quadrant into a new large tile, which then moves to a location above the small tiles but still in the quadrant. the original small tiles (not translucent) should still be visible. also, pause for FINAL_PAUSE seconds before moving to the next problem. FINAL_PAUSE should be a javascript constant and initially set to 4 seconds

the highlighting around the quadrant that indicates an incorrect user choice (red) and the correct answer (green) should not be shown until the end of the animation. after the animation, instead of having a FINAL_PAUSE, simply cover the top banner with a question asking the user if they would like to continue. remain on that page until the user clicks the continue button, at which point a new problem is presented and the banner returns to showing the instructions and progress bar

  <script>
    const J = 30;
    const K = 0;
    const L = 1;
    const M = 150;
    const N = 50;

    const TILE_COUNT = 50;
    const MIN_VALUE = 48;
    const MAX_VALUE = 96;
    const MIN_TARGET = 56;
    const MAX_TARGET = 88;
    const TARGET_SEPARATION = 5;

    let points = N;
    let roundLocked = false;
    let quadrants = [];

    const board = document.getElementById("board");
    const progressFill = document.getElementById("progressFill");
    const continueCover = document.getElementById("continueCover");
    const continueButton = document.getElementById("continueButton");

    continueButton.addEventListener("click", () => {
      continueCover.classList.remove("show");
      createRound();
    });

    function randInt(min, max) {
      return Math.floor(Math.random() * (max - min + 1)) + min;
    }

    function sleep(ms) {
      return new Promise(resolve => setTimeout(resolve, ms));
    }

    function updateProgress() {
      points = Math.max(0, points);
      const pct = Math.max(0, Math.min(100, (points / M) * 100));
      progressFill.style.width = pct + "%";
      progressFill.textContent = points;
    }

    function changePoints(delta) {
      points = Math.max(0, points + delta);
      updateProgress();
    }

    function generateTargets() {
      const targets = [];

      while (targets.length < 4) {
        const candidate = randInt(MIN_TARGET, MAX_TARGET);
        const ok = targets.every(t => Math.abs(t - candidate) >= TARGET_SEPARATION);
        if (ok) targets.push(candidate);
      }

      return targets;
    }

    function generateValuesWithAverage(target) {
      const values = Array(TILE_COUNT).fill(target);

      for (let step = 0; step < 900; step++) {
        const a = randInt(0, TILE_COUNT - 1);
        let b = randInt(0, TILE_COUNT - 1);

        while (b === a) {
          b = randInt(0, TILE_COUNT - 1);
        }

        const upRoom = MAX_VALUE - values[a];
        const downRoom = values[b] - MIN_VALUE;
        const maxDelta = Math.min(upRoom, downRoom, 8);

        if (maxDelta > 0) {
          const delta = randInt(1, maxDelta);
          values[a] += delta;
          values[b] -= delta;
        }
      }

      for (let i = values.length - 1; i > 0; i--) {
        const j = randInt(0, i);
        [values[i], values[j]] = [values[j], values[i]];
      }

      return values;
    }

    function createRound() {
      roundLocked = false;
      board.innerHTML = "";

      const targets = generateTargets();

      quadrants = targets.map((target, index) => ({
        index,
        target,
        values: generateValuesWithAverage(target),
        element: null,
        tileElements: []
      }));

      quadrants.forEach(q => {
        const quadrant = document.createElement("section");
        quadrant.className = `quadrant q${q.index}`;

        const button = document.createElement("button");
        button.className = "pick-button";
        button.textContent = String.fromCharCode(65 + q.index);
        button.setAttribute("aria-label", `select group ${String.fromCharCode(65 + q.index)}`);
        button.addEventListener("click", () => makeGuess(q.index));
        quadrant.appendChild(button);

        const tiles = document.createElement("div");
        tiles.className = "tiles";

        q.values.forEach(value => {
          const tile = document.createElement("div");
          tile.className = "tile";
          tile.dataset.value = value;

          const inner = document.createElement("div");
          inner.className = "tile-inner";

          const front = document.createElement("div");
          front.className = "face front";

          const back = document.createElement("div");
          back.className = "face back";
          back.textContent = value;

          inner.appendChild(front);
          inner.appendChild(back);
          tile.appendChild(inner);

          tile.addEventListener("click", () => {
            if (roundLocked || tile.classList.contains("revealed")) return;
            tile.classList.add("revealed");
            changePoints(-L);
          });

          tiles.appendChild(tile);
          q.tileElements.push(tile);
        });

        const merged = document.createElement("div");
        merged.className = "merged-tile";
        merged.textContent = q.target;

        quadrant.appendChild(tiles);
        quadrant.appendChild(merged);

        q.element = quadrant;
        q.tilesContainer = tiles;
        q.mergedElement = merged;

        board.appendChild(quadrant);
      });
    }

    async function makeGuess(index) {
      if (roundLocked) return;
      roundLocked = true;

      document.querySelectorAll(".pick-button").forEach(btn => btn.disabled = true);
      document.querySelectorAll(".tile").forEach(tile => tile.classList.add("disabled"));

      const maxTarget = Math.max(...quadrants.map(q => q.target));
      const correctIndex = quadrants.find(q => q.target === maxTarget).index;

      if (index === correctIndex) {
        changePoints(J);
      } else {
        changePoints(-K);
      }

      await revealAllTiles();
      await explainWithGhostMerges();

      showAnswerHighlight(index, correctIndex);
      showContinuePrompt();
    }

    function showAnswerHighlight(userIndex, correctIndex) {
      if (userIndex === correctIndex) {
        quadrants[correctIndex].element.classList.add("correct");
      } else {
        quadrants[userIndex].element.classList.add("incorrect");
        quadrants[correctIndex].element.classList.add("correct");
      }
    }

    function showContinuePrompt() {
      continueCover.classList.add("show");
      continueButton.focus();
    }

    async function revealAllTiles() {
      const allTiles = quadrants.flatMap(q => q.tileElements);
      const unrevealed = allTiles.filter(tile => !tile.classList.contains("revealed"));

      unrevealed.forEach((tile, i) => {
        setTimeout(() => {
          tile.classList.add("revealed");
        }, i * 18);
      });

      await sleep(unrevealed.length * 18 + 520);
    }

    async function explainWithGhostMerges() {
      for (const q of quadrants) {
        await mergeGhostCopies(q);
        await sleep(500);
      }
    }

    async function mergeGhostCopies(q) {
      const quadrant = q.element;
      const quadrantRect = quadrant.getBoundingClientRect();

      const centerX = quadrantRect.width / 2;
      const centerY = quadrantRect.height / 2;

      const ghosts = q.tileElements.map(tile => {
        const tileRect = tile.getBoundingClientRect();

        const ghost = document.createElement("div");
        ghost.className = "ghost-tile";
        ghost.textContent = tile.dataset.value;

        ghost.style.left = `${tileRect.left - quadrantRect.left}px`;
        ghost.style.top = `${tileRect.top - quadrantRect.top}px`;

        quadrant.appendChild(ghost);
        return ghost;
      });

      await sleep(40);

      ghosts.forEach(ghost => {
        ghost.style.left = `${centerX - ghost.offsetWidth / 2}px`;
        ghost.style.top = `${centerY - ghost.offsetHeight / 2}px`;
        ghost.style.transform = "scale(0.18)";
        ghost.style.opacity = "0";
      });

      await sleep(700);

      ghosts.forEach(ghost => ghost.remove());

      q.mergedElement.classList.add("show");

      await sleep(450);

      q.mergedElement.classList.add("settled");

      await sleep(700);
    }

    updateProgress();
    createRound();
  </script>
