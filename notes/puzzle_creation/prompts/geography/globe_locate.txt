States', 'state': 'Florida', 'city': 'Palm Harbor', 'address': '1710 Delaware Avenue', 'addressLat': 28.0837137, 'addressLon': -82.7669778, 'cityBbox': [28.0837, 28.0838, -82.7671, -82.7668]}. at the top of the screen, show the address in the form "continent, country, state, city, address". the geography data could be from anywhere in the world. if "state" is blank, simply omit this from the shown state they guessed to the center of the correct state. the line is labeled with the distance it represents. that distance is also added to a running error total displayed in the top banner. the line and distance label then disappear and the user is advanced to the next stage. in the next stage, the city part of the address is highlighted. the user is shown a map of the state (or country if there it represents. that distance is also added to a running error total displayed in the top banner. the line and distance label then disappear and the user is advanced to the next stage. in the next and final stage, the street part of the address is highlighted. the user is shown a map of the city and asked to click the location of the street address. a red error line is drawn from location they guessed to the exact address location (indicated by addressLat and addressLon). the line is labeled with the distance it represents. that distance is also added to a running error total displayed in the top banner. the user's total running error is prominently displayed and the user is told to refresh the screen if they want to play again.

all labels should be omitted from the maps, since it is the user who must figure out what region to click on from the boundaries. also make the cursor a normal cursor with a sharp point (not a hand) so that the exact point clicked on is obvious. when moving to the level where the user must select the state, the map zooms out, when it should in fact be zooming in

the continent name displayed to the user should always be the continent label (ie "North America") and not the code such as NA

Below is a complete single-file HTML activity using **HTML, CSS, JavaScript, Leaflet, OpenStreetMap tiles, and Nominatim geocoding**.

make sure that after every automatic zoom or repositioning of the map, the point the user is supposed to click on next is visible on the screen

Save it as `geography-activity.html` and open it in a browser.

> Note: This uses the public Nominatim service for reverse-geocoding clicked countries/states. For classroom/demo use it should be fine, but for heavy or production use, use your own geocoding service.

  <script>
    /********************************************************************
     * Geography data for the activity.
     * Replace this object with any other location using the same form.
     ********************************************************************/
    const LOCATION = {
      continent: "NA",
      country: "United States",
      state: "Florida",
      city: "Palm Harbor",
      address: "1710 Delaware Avenue",
      addressLat: 28.0837137,
      addressLon: -82.7669778,
      cityBbox: [28.0837, 28.0838, -82.7671, -82.7668]
    };

    /********************************************************************
     * Approximate continent data.
     * Continent click detection is intentionally lightweight.
     ********************************************************************/
    const CONTINENTS = {
      NA: {
        label: "North America",
        aliases: ["NA", "North America", "N. America"],
        center: [54, -105],
        bounds: [[5, -170], [83, -52]]
      },
      SA: {
        label: "South America",
        aliases: ["SA", "South America", "S. America"],
        center: [-15, -60],
        bounds: [[-56, -82], [13, -34]]
      },
      EU: {
        label: "Europe",
        aliases: ["EU", "Europe"],
        center: [54, 15],
        bounds: [[35, -25], [72, 45]]
      },
      AF: {
        label: "Africa",
        aliases: ["AF", "Africa"],
        center: [1, 20],
        bounds: [[-35, -20], [38, 55]]
      },
      AS: {
        label: "Asia",
        aliases: ["AS", "Asia"],
        center: [34, 100],
        bounds: [[-12, 45], [82, 180]]
      },
      OC: {
        label: "Oceania",
        aliases: ["OC", "Oceania", "Australia"],
        center: [-22, 140],
        bounds: [[-50, 110], [10, 180]]
      },
      AN: {
        label: "Antarctica",
        aliases: ["AN", "Antarctica"],
        center: [-82, 0],
        bounds: [[-90, -180], [-60, 180]]
      }
    };

    let map;
    let currentStageIndex = 0;
    let errorTotalKm = 0;
    let isBusy = false;

    const geocodeCache = new Map();

    const stageOrder = [
      "continent",
      "country",
      ...(LOCATION.state && LOCATION.state.trim() ? ["state"] : []),
      "city",
      "address"
    ];

    const reference = {
      targetContinentCode: null,
      country: null,
      state: null,
      cityCenter: null,
      addressCenter: null
    };

    const topBanner = document.getElementById("topBanner");
    const addressLine = document.getElementById("addressLine");
    const scoreValue = document.getElementById("scoreValue");
    const instruction = document.getElementById("instruction");
    const finalPanel = document.getElementById("finalPanel");
    const finalScore = document.getElementById("finalScore");

    init();

    async function init() {
      buildAddressLine();

      map = L.map("map", {
        worldCopyJump: true,
        zoomControl: true
      });

      L.tileLayer("https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png", {
        maxZoom: 19,
        attribution:
          '&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors'
      }).addTo(map);

      map.setView([20, 0], 2);

      map.on("click", onMapClick);

      document.body.classList.add("loading");

      try {
        reference.targetContinentCode = normalizeContinentCode(LOCATION.continent);

        reference.country = await geocodePlace(LOCATION.country);

        if (LOCATION.state && LOCATION.state.trim()) {
          reference.state = await geocodePlace(`${LOCATION.state}, ${LOCATION.country}`);
        }

        reference.cityCenter = cityCenterFromBbox(LOCATION.cityBbox);

        reference.addressCenter = [
          Number(LOCATION.addressLat),
          Number(LOCATION.addressLon)
        ];
      } catch (err) {
        console.error(err);
        instruction.textContent =
          "Some map lookup data could not be loaded. The activity will still try to continue.";
      } finally {
        document.body.classList.remove("loading");
        startCurrentStage();
      }
    }

    function buildAddressLine() {
      const parts = [
        { key: "continent", text: LOCATION.continent },
        { key: "country", text: LOCATION.country },
        ...(LOCATION.state && LOCATION.state.trim()
          ? [{ key: "state", text: LOCATION.state }]
          : []),
        { key: "city", text: LOCATION.city },
        { key: "address", text: LOCATION.address }
      ];

      addressLine.innerHTML = parts
        .map((part, index) => {
          const comma = index < parts.length - 1 ? "," : "";
          return `<span class="addressPart" data-part="${part.key}">${escapeHtml(part.text)}</span>${comma}`;
        })
        .join(" ");
    }

    function highlightAddressPart(stage) {
      document.querySelectorAll(".addressPart").forEach(el => {
        el.classList.toggle("active", el.dataset.part === stage);
      });
    }

    function startCurrentStage() {
      if (currentStageIndex >= stageOrder.length) {
        finishGame();
        return;
      }

      const stage = stageOrder[currentStageIndex];
      highlightAddressPart(stage);
      isBusy = false;

      if (stage === "continent") {
        instruction.textContent = `Click the correct continent for ${LOCATION.continent}.`;
        map.setView([20, 0], 2);
      }

      if (stage === "country") {
        instruction.textContent = `Click the correct country: ${LOCATION.country}.`;
        const continent = CONTINENTS[reference.targetContinentCode];
        if (continent) {
          map.fitBounds(continent.bounds, { padding: [20, 20] });
        } else {
          map.setView([20, 0], 2);
        }
      }

      if (stage === "state") {
        instruction.textContent = `Click the correct state/province/region: ${LOCATION.state}.`;
        if (reference.country?.bbox) {
          map.fitBounds(reference.country.bbox, { padding: [20, 20] });
        }
      }

      if (stage === "city") {
        instruction.textContent = `Click where you think the city is: ${LOCATION.city}.`;
        if (reference.state?.bbox) {
          map.fitBounds(reference.state.bbox, { padding: [20, 20] });
        } else if (reference.country?.bbox) {
          map.fitBounds(reference.country.bbox, { padding: [20, 20] });
        }
      }

      if (stage === "address") {
        instruction.textContent = `Click where you think the street address is: ${LOCATION.address}.`;
        const cityBounds = bboxToLeafletBounds(LOCATION.cityBbox);
        map.fitBounds(cityBounds, {
          padding: [50, 50],
          maxZoom: 16
        });
      }
    }

    async function onMapClick(e) {
      if (isBusy) return;

      const stage = stageOrder[currentStageIndex];
      isBusy = true;

      try {
        if (stage === "continent") {
          await handleContinentGuess(e.latlng);
        } else if (stage === "country") {
          await handleCountryGuess(e.latlng);
        } else if (stage === "state") {
          await handleStateGuess(e.latlng);
        } else if (stage === "city") {
          await handlePointGuess(
            [e.latlng.lat, e.latlng.lng],
            reference.cityCenter,
            "City location checked."
          );
        } else if (stage === "address") {
          await handlePointGuess(
            [e.latlng.lat, e.latlng.lng],
            reference.addressCenter,
            "Address location checked.",
            true
          );
        }
      } catch (err) {
        console.error(err);
        instruction.textContent = "Something went wrong with that guess. Try again.";
        isBusy = false;
      }
    }

    async function handleContinentGuess(latlng) {
      const guessedCode = getContinentAtLatLng(latlng);
      const correctCode = reference.targetContinentCode;

      if (guessedCode && guessedCode === correctCode) {
        flashCorrectAndAdvance();
        return;
      }

      const guessedCenter = guessedCode
        ? CONTINENTS[guessedCode].center
        : [latlng.lat, latlng.lng];

      const correctCenter = CONTINENTS[correctCode]?.center;

      if (!correctCenter) {
        advanceStage();
        return;
      }

      const km = haversineKm(guessedCenter, correctCenter);
      addError(km);

      await showTemporaryErrorLine(
        guessedCenter,
        correctCenter,
        km,
        guessedCode
          ? `${CONTINENTS[guessedCode].label} → ${CONTINENTS[correctCode].label}`
          : `Unknown area → ${CONTINENTS[correctCode].label}`
      );

      advanceStage();
    }

    async function handleCountryGuess(latlng) {
      const reverse = await reverseGeocode(latlng.lat, latlng.lng);
      const guessedCountry = reverse?.address?.country || "";
      const correctCountry = LOCATION.country;

      if (sameName(guessedCountry, correctCountry)) {
        flashCorrectAndAdvance();
        return;
      }

      let guessedCenter = [latlng.lat, latlng.lng];

      if (guessedCountry) {
        try {
          const guessedGeo = await geocodePlace(guessedCountry);
          guessedCenter = guessedGeo.center;
        } catch {
          // Fallback to clicked point.
        }
      }

      const correctCenter = reference.country?.center || [
        LOCATION.addressLat,
        LOCATION.addressLon
      ];

      const km = haversineKm(guessedCenter, correctCenter);
      addError(km);

      await showTemporaryErrorLine(
        guessedCenter,
        correctCenter,
        km,
        guessedCountry
          ? `${guessedCountry} → ${correctCountry}`
          : `Unknown country → ${correctCountry}`
      );

      advanceStage();
    }

    async function handleStateGuess(latlng) {
      const reverse = await reverseGeocode(latlng.lat, latlng.lng);
      const guessedState = extractAdminRegion(reverse?.address || {});
      const correctState = LOCATION.state;

      if (sameName(guessedState, correctState)) {
        flashCorrectAndAdvance();
        return;
      }

      let guessedCenter = [latlng.lat, latlng.lng];

      if (guessedState) {
        try {
          const guessedCountry = reverse?.address?.country || LOCATION.country;
          const guessedGeo = await geocodePlace(`${guessedState}, ${guessedCountry}`);
          guessedCenter = guessedGeo.center;
        } catch {
          // Fallback to clicked point.
        }
      }

      const correctCenter = reference.state?.center || reference.country?.center || [
        LOCATION.addressLat,
        LOCATION.addressLon
      ];

      const km = haversineKm(guessedCenter, correctCenter);
      addError(km);

      await showTemporaryErrorLine(
        guessedCenter,
        correctCenter,
        km,
        guessedState
          ? `${guessedState} → ${correctState}`
          : `Unknown region → ${correctState}`
      );

      advanceStage();
    }

    async function handlePointGuess(guessedPoint, correctPoint, message, isFinal = false) {
      const km = haversineKm(guessedPoint, correctPoint);
      addError(km);

      await showTemporaryErrorLine(
        guessedPoint,
        correctPoint,
        km,
        message
      );

      if (isFinal) {
        finishGame();
      } else {
        advanceStage();
      }
    }

    function flashCorrectAndAdvance() {
      topBanner.classList.add("correctFlash");
      instruction.textContent = "Correct!";
      setTimeout(() => {
        topBanner.classList.remove("correctFlash");
        advanceStage();
      }, 750);
    }

    function advanceStage() {
      currentStageIndex += 1;
      startCurrentStage();
    }

    function finishGame() {
      isBusy = true;
      highlightAddressPart("address");
      instruction.textContent =
        "Finished! Your total running error is displayed below. Refresh the screen to play again.";
      scoreValue.textContent = formatDistance(errorTotalKm);
      finalScore.textContent = formatDistance(errorTotalKm);
      finalPanel.classList.add("show");
    }

    function addError(km) {
      errorTotalKm += km;
      scoreValue.textContent = formatDistance(errorTotalKm);
    }

    async function showTemporaryErrorLine(from, to, km, labelText) {
      const line = L.polyline([from, to], {
        color: "#d62828",
        weight: 5,
        opacity: 0.9
      }).addTo(map);

      const mid = midpoint(from, to);

      const label = L.marker(mid, {
        interactive: false,
        icon: L.divIcon({
          className: "distance-label",
          html: `${formatDistance(km)}`,
          iconSize: null
        })
      }).addTo(map);

      instruction.textContent = `${labelText} — error: ${formatDistance(km)}`;

      return new Promise(resolve => {
        setTimeout(() => {
          map.removeLayer(line);
          map.removeLayer(label);
          resolve();
        }, 1900);
      });
    }

    async function geocodePlace(query) {
      const key = `search:${query.toLowerCase()}`;
      if (geocodeCache.has(key)) return geocodeCache.get(key);

      const url =
        "https://nominatim.openstreetmap.org/search?" +
        new URLSearchParams({
          q: query,
          format: "jsonv2",
          limit: "1",
          addressdetails: "1"
        });

      const res = await fetch(url);
      if (!res.ok) throw new Error(`Geocoding failed for ${query}`);

      const data = await res.json();
      if (!data.length) throw new Error(`No geocoding result for ${query}`);

      const item = data[0];

      const result = {
        name: item.display_name,
        center: [Number(item.lat), Number(item.lon)],
        bbox: item.boundingbox
          ? bboxToLeafletBounds(item.boundingbox.map(Number))
          : null
      };

      geocodeCache.set(key, result);
      return result;
    }

    async function reverseGeocode(lat, lon) {
      const url =
        "https://nominatim.openstreetmap.org/reverse?" +
        new URLSearchParams({
          lat,
          lon,
          format: "jsonv2",
          addressdetails: "1",
          zoom: "10"
        });

      const res = await fetch(url);
      if (!res.ok) return null;

      return await res.json();
    }

    function extractAdminRegion(address) {
      return (
        address.state ||
        address.province ||
        address.region ||
        address.state_district ||
        address.county ||
        ""
      );
    }

    function normalizeContinentCode(value) {
      const raw = String(value || "").trim().toLowerCase();

      for (const [code, data] of Object.entries(CONTINENTS)) {
        if (code.toLowerCase() === raw) return code;

        if (data.aliases.some(alias => alias.toLowerCase() === raw)) {
          return code;
        }

        if (data.label.toLowerCase() === raw) return code;
      }

      return value;
    }

    function getContinentAtLatLng(latlng) {
      const lat = latlng.lat;
      let lon = latlng.lng;

      while (lon < -180) lon += 360;
      while (lon > 180) lon -= 360;

      if (lat < -60) return "AN";

      // North America, including Central America and much of the Caribbean.
      if (lat >= 5 && lat <= 83 && lon >= -170 && lon <= -50) return "NA";

      // South America.
      if (lat >= -56 && lat <= 13 && lon >= -82 && lon <= -34) return "SA";

      // Africa.
      if (lat >= -35 && lat <= 38 && lon >= -20 && lon <= 55) return "AF";

      // Europe.
      if (lat >= 35 && lat <= 72 && lon >= -25 && lon <= 45) return "EU";

      // Asia.
      if (lat >= -12 && lat <= 82 && lon >= 45 && lon <= 180) return "AS";

      // Oceania.
      if (lat >= -50 && lat <= 10 && lon >= 110 && lon <= 180) return "OC";

      return null;
    }

    function bboxToLeafletBounds(bbox) {
      // bbox form: [south, north, west, east]
      const south = Number(bbox[0]);
      const north = Number(bbox[1]);
      const west = Number(bbox[2]);
      const east = Number(bbox[3]);

      return [
        [south, west],
        [north, east]
      ];
    }

    function cityCenterFromBbox(bbox) {
      const south = Number(bbox[0]);
      const north = Number(bbox[1]);
      const west = Number(bbox[2]);
      const east = Number(bbox[3]);

      return [
        (south + north) / 2,
        (west + east) / 2
      ];
    }

    function haversineKm(a, b) {
      const lat1 = Number(a[0]);
      const lon1 = Number(a[1]);
      const lat2 = Number(b[0]);
      const lon2 = Number(b[1]);

      const R = 6371.0088;
      const toRad = deg => deg * Math.PI / 180;

      const dLat = toRad(lat2 - lat1);
      const dLon = toRad(lon2 - lon1);

      const p1 = toRad(lat1);
      const p2 = toRad(lat2);

      const h =
        Math.sin(dLat / 2) ** 2 +
        Math.cos(p1) * Math.cos(p2) * Math.sin(dLon / 2) ** 2;

      return 2 * R * Math.atan2(Math.sqrt(h), Math.sqrt(1 - h));
    }

    function midpoint(a, b) {
      return [
        (Number(a[0]) + Number(b[0])) / 2,
        (Number(a[1]) + Number(b[1])) / 2
      ];
    }

    function formatDistance(km) {
      if (km < 1) {
        return `${Math.round(km * 1000)} m`;
      }

      if (km < 100) {
        return `${km.toFixed(1)} km`;
      }

      return `${Math.round(km).toLocaleString()} km`;
    }

    function sameName(a, b) {
      return normalizeName(a) === normalizeName(b);
    }

    function normalizeName(value) {
      return String(value || "")
        .normalize("NFD")
        .replace(/[\u0300-\u036f]/g, "")
        .toLowerCase()
        .replace(/[^a-z0-9]/g, "");
    }

    function escapeHtml(value) {
      return String(value)
        .replaceAll("&", "&amp;")
        .replaceAll("<", "&lt;")
        .replaceAll(">", "&gt;")
        .replaceAll('"', "&quot;")
        .replaceAll("'", "&#039;");
    }
  </script>
